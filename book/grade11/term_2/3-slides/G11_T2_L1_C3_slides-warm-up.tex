% Instructional source (unchanged):
% book/grade11/term_2/3-slides/G11_T2_L1_C3_slides.tex
% Anchors: "The confidence interval stays centred at 84" (centre and bounds);
% "Task 2: Fixed SE, changing multiplier" (uncertainty and confidence choice);
% "What changes, and what stays fixed?" (precision/reliability synthesis).
% Central idea: a shared estimate can support different interval widths;
% greater coverage requires a wider interval when the data/model stay fixed.
% Purpose: compare how cautiously two reports describe a mean journey time,
% and why a more cautious range may be less useful for precise planning.
% Accessible reasoning: both ranges share a centre; the wider one allows
% more room for an estimate to miss the true mean. No formal CI calculation.
% Later refinement: centre, standard error, critical value and margin explain
% the width and quantify the precision/confidence trade-off under assumptions.
% No practice used. Fresh context/data; no lesson example answers reproduced.
% Use after the lesson's complete opening legend, before its first example.
% Timing: reading 20 s + thinking/discussion 60 s + check/bridge 40 s = 120 s.
% Frame 1 has no answer; frame 2 is the check, with bridge on overlay 2.
\begin{frame}[t]{Planning the school journey}
A school surveys a random sample of \CtwoSize{60} students about their
journey times. Their average is \CtwoPoint{30 minutes}.

\medskip
Using the same survey, two reports suggest ranges for the
\textbf{true average journey time of all students}:

\medskip
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\textbf{Report A} & \CtwoEndpoint{28--32 minutes}\\[4pt]
\textbf{Report B} & \CtwoEndpoint{24--36 minutes}\\
\bottomrule
\end{tabularx}

\medskip
\textbf{Which report would you choose if your priority is to avoid leaving
the true average outside the range? Explain your reasoning.}

\medskip
{\color{GETeal}\textbf{Think and discuss: 60 seconds.}}
\end{frame}

\begin{frame}[t]{Check and connection}
\textbf{An intuitive check.}
Report B allows more room on either side of the same
\CtwoPoint{30-minute estimate}. Every average inside A is also inside B.
Neither range is guaranteed to contain the true average.

\medskip
A gives a tighter estimate for planning; B leaves more room for uncertainty.
These ranges describe the \textbf{average}, not every student's journey.

\medskip
\visible<2->{\textbf{Where the lesson takes us.}
We will connect the \CtwoPoint{centre} and \CtwoMoE{margin of error}
to the data and the chosen \CtwoConfidence{confidence level},
so we can explain how much wider a more cautious interval needs to be.}
\end{frame}
