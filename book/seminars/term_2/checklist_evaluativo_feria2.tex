\newcommand{\pageheading}[1]{%
  {\fontsize{15}{17}\selectfont\bfseries Coevaluación de la feria\hfill
    \fontsize{10}{12}\selectfont Grupos #1\par}
  \vspace{3pt}
  \begin{minipage}{0.99\linewidth}\field{Integrantes grupo evaluador}\end{minipage}\hfill
  \vspace{3pt}
  Escribe E, P, A o N en cada casilla. Anota una fortaleza y una mejora concreta.\par
  \textbf{E} = Se evidencia; \textbf{P} = Parcialmente;
  \textbf{A} = Aún no se evidencia; \textbf{N} = No se observa.\par
  \vspace{8pt}
}

\newcommand{\assessmentquestion}[2]{%
  #1. #2 & \\\hline}

% Fixed, unbreakable blocks: three rows per letter-size page, at 10 pt.
\newcommand{\groupblock}[1]{%
  \begin{minipage}[t]{0.488\linewidth}
  \begin{tcolorbox}[
    colback=white,colframe=gridgray,colbacktitle=black!12,
    coltitle=black,fonttitle=\bfseries,title={Grupo #1},
    sharp corners,boxrule=0.6pt,boxsep=0pt,
    left=5pt,right=5pt,top=3pt,bottom=5pt,
    toptitle=3pt,bottomtitle=3pt,
    height=211pt,before skip=0pt,after skip=0pt]
    \field{Grupo/proyecto}\par
    \vspace{4pt}
    \begin{tabularx}{\linewidth}{|>{\raggedright\arraybackslash}X|p{16pt}|}
      \hline
      \assessmentquestion{1}{¿El grupo explica con claridad qué problema busca resolver y a quiénes afecta?}
      \assessmentquestion{2}{¿El grupo presenta una solución pertinente y explica qué la hace diferente?}
      \assessmentquestion{3}{¿El grupo justifica con ejemplos, datos o respuestas cómo funcionaría su solución?}
      \assessmentquestion{4}{¿El grupo explica qué recursos necesita y qué beneficios podría generar su propuesta?}
    \end{tabularx}\par
    \vspace{4pt}
    \textbf{Fortaleza}\par
    \vspace{20pt}\hrule height 0.3pt
    \vspace{4pt}
    \textbf{Mejora concreta}\par
    \vspace{20pt}\hrule height 0.3pt
  \end{tcolorbox}
  \end{minipage}%
}
\newcommand{\grouprow}[2]{%
  \groupblock{#1}\hfill\groupblock{#2}\par\vspace{7pt}}
