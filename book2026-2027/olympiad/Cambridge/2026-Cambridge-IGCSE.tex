		\textbf{Cambridge IGCSE Mathematics 0580}\\
		\textbf{Extended Mathematics -- Grade 10 Practice Paper}
	\end{center}
	
	\vspace{5mm}
	
	\noindent
	\textbf{Time: 1 hour 30 minutes}
	
	\vspace{3mm}
	
	\noindent
	\textbf{Total marks: 60}
	
	\vspace{5mm}
	
	\noindent
	\textbf{Instructions}
	\begin{itemize}
		\item Answer all questions.
		\item Show all necessary working.
		\item Give non-exact numerical answers correct to 3 significant figures unless otherwise stated.
		\item A calculator may be used.
		\item Diagrams are not necessarily drawn to scale.
	\end{itemize}
	
	\vspace{5mm}
	
	\section*{Question 1 -- Number \score{6}}
	
	\begin{enumerate}
		
		\item[(a)]
		Express \(0.000384\) in standard form.
		\score{1}
		
		\vspace{15mm}
		
		\item[(b)]
		Find the highest common factor of \(84\) and \(126\).
		\score{1}
		
		\vspace{15mm}
		
		\item[(c)]
		Simplify
		\[
		\frac{3}{\sqrt{5}+1}.
		\]
		Give your answer in the form \(a\sqrt{5}+b\), where \(a\) and \(b\) are rational numbers.
		\score{2}
		
		\vspace{25mm}
		
		\item[(d)]
		A quantity \(P\) is increased by \(12\%\) and then decreased by \(12\%\).
		
		Determine whether the final value is equal to the original value. Justify your answer.
		\score{2}
		
	\end{enumerate}

	\section*{Question 2 -- Algebra \score{8}}
	
	\begin{enumerate}
		
		\item[(a)]
		Expand and simplify
		\[
		(2x-3)(x+5).
		\]
		\score{2}
		
		\vspace{15mm}
		
		\item[(b)]
		Factorise completely
		\[
		6x^2-13x-5.
		\]
		\score{2}
		
		\vspace{15mm}
		
		\item[(c)]
		Solve
		\[
		\frac{2x-1}{3}+\frac{x+4}{2}=7.
		\]
		\score{2}
		
		\vspace{20mm}
		
		\item[(d)]
		Solve
		\[
		x^2-7x+10=0.
		\]
		\score{2}
		
	\end{enumerate}
	
	\vspace{5mm}
	
	\section*{Question 3 -- Simultaneous equations and sequences \score{7}}
	
	\begin{enumerate}
		
		\item[(a)]
		Solve the simultaneous equations
		\[
		\begin{aligned}
			2x+3y&=17,\\
			4x-y&=11.
		\end{aligned}
		\]
		\score{3}
		
		\vspace{30mm}
		
		\item[(b)]
		The first four terms of a sequence are
		\[
		5,\quad 9,\quad 13,\quad 17.
		\]
		
		\begin{enumerate}
			\item[(i)]
			Write down the next term.
			\score{1}
			
			\item[(ii)]
			Find an expression for the \(n\)th term.
			\score{2}
			
			\item[(iii)]
			Determine which term of the sequence is \(101\).
			\score{1}
		\end{enumerate}
		
	\end{enumerate}

	\section*{Question 4 -- Coordinate geometry \score{7}}
	
	The points \(A(-2,5)\) and \(B(6,-3)\) are shown on a coordinate grid.
	
	\begin{enumerate}
		
		\item[(a)]
		Find the gradient of \(AB\).
		\score{2}
		
		\vspace{20mm}
		
		\item[(b)]
		Find the midpoint of \(AB\).
		\score{2}
		
		\vspace{20mm}
		
		\item[(c)]
		Find the equation of the straight line passing through \(A\) and \(B\).
		\score{2}
		
		\vspace{20mm}
		
		\item[(d)]
		A third point \(C\) has coordinates \((4,7)\).
		
		Determine whether \(AC\) is perpendicular to \(AB\).
		You must justify your answer.
		\score{1}
		
	\end{enumerate}

	\section*{Question 5 -- Geometry and mensuration \score{8}}
	
	A sector of a circle has radius \(8\) cm and angle \(135^\circ\).
	
	\begin{enumerate}
		
		\item[(a)]
		Calculate the length of the arc of the sector.
		\score{2}
		
		\vspace{25mm}
		
		\item[(b)]
		Calculate the area of the sector.
		\score{2}
		
		\vspace{25mm}
		
		\item[(c)]
		The sector is joined to a rectangle measuring \(8\) cm by \(12\) cm.
		
		Calculate the total area of the resulting shape.
		\score{2}
		
		\vspace{25mm}
		
		\item[(d)]
		Explain why the perimeter of the resulting shape cannot be found simply by adding the perimeter of the rectangle to the circumference of the full circle.
		\score{2}
		
	\end{enumerate}

	\section*{Question 6 -- Trigonometry \score{8}}
	
	A ladder of length \(6.5\) m rests against a vertical wall.
	
	The bottom of the ladder is \(2.8\) m from the wall.
	
	\begin{enumerate}
		
		\item[(a)]
		Calculate the height reached by the ladder on the wall.
		\score{3}
		
		\vspace{30mm}
		
		\item[(b)]
		Calculate the angle between the ladder and the ground.
		\score{2}
		
		\vspace{25mm}
		
		\item[(c)]
		The ladder is moved so that its bottom is \(3.5\) m from the wall, while its length remains \(6.5\) m.
		
		Calculate the new height reached by the ladder.
		\score{3}
		
	\end{enumerate}

	\section*{Question 7 -- Probability \score{7}}
	
	A box contains \(5\) red counters, \(3\) blue counters and \(2\) green counters.
	
	Two counters are selected at random without replacement.
	
	\begin{enumerate}
		
		\item[(a)]
		Find the probability that the first counter selected is red.
		\score{1}
		
		\vspace{15mm}
		
		\item[(b)]
		Find the probability that both counters are red.
		\score{2}
		
		\vspace{20mm}
		
		\item[(c)]
		Find the probability that exactly one of the two counters is blue.
		\score{2}
		
		\vspace{25mm}
		
		\item[(d)]
		Find the probability that neither counter is green.
		\score{2}
		
	\end{enumerate}

	\section*{Question 8 -- Statistics \score{9}}
	
	The times, in minutes, taken by \(12\) students to complete a mathematical task are
	
	\[
	\begin{array}{cccccccccccc}
		18 & 21 & 17 & 24 & 19 & 22 &
		26 & 21 & 20 & 18 & 23 & 25
	\end{array}
	\]
	
	\begin{enumerate}
		
		\item[(a)]
		Find the mean time.
		\score{2}
		
		\vspace{20mm}
		
		\item[(b)]
		Find the median time.
		\score{2}
		
		\vspace{20mm}
		
		\item[(c)]
		Find the range.
		\score{1}
		
		\vspace{15mm}
		
		\item[(d)]
		The student who took \(26\) minutes is removed from the data.
		
		Explain whether the mean will increase, decrease, or remain unchanged.
		\score{2}
		
		\vspace{20mm}
		
		\item[(e)]
		Another student takes \(35\) minutes to complete the task.
		
		Explain why the median is likely to be a better measure of the typical completion time than the mean.
		\score{2}
		
	\end{enumerate}

	\section*{Question 9 -- Functions and graphs \score{8}}
	
	The function \(f\) is defined by
	
	\[
	f(x)=x^2-4x+1.
	\]
	
	\begin{enumerate}
		
		\item[(a)]
		Find \(f(3)\).
		\score{1}
		
		\vspace{15mm}
		
		\item[(b)]
		Solve
		\[
		f(x)=1.
		\]
		\score{2}
		
		\vspace{20mm}
		
		\item[(c)]
		Write \(f(x)\) in the form
		\[
		(x-a)^2+b.
		\]
		\score{2}
		
		\vspace{20mm}
		
		\item[(d)]
		State the coordinates of the turning point of the graph of \(y=f(x)\).
		\score{1}
		
		\vspace{15mm}
		
		\item[(e)]
		Find the inverse function \(f^{-1}(x)\).
		
		\textit{State any restriction required for the inverse to be a function.}
		\score{2}
		
	\end{enumerate}

	\section*{Question 10 -- Problem solving \score{10}}
	
	A company produces cylindrical containers.
	
	Each container has radius \(4\) cm and height \(12\) cm.
	
	The company wants to reduce the amount of material used while keeping the volume unchanged.
	
	\begin{enumerate}
		
		\item[(a)]
		Calculate the volume of the original container.
		\score{2}
		
		\vspace{25mm}
		
		\item[(b)]
		The company proposes a new container with radius \(6\) cm.
		
		Calculate the height required for the new container to have the same volume as the original.
		\score{3}
		
		\vspace{30mm}
		
		\item[(c)]
		The surface area of a closed cylinder is given by
		\[
		A=2\pi r^2+2\pi rh.
		\]
		
		Calculate the surface area of the original container.
		\score{2}
		
		\vspace{25mm}
		
		\item[(d)]
		Calculate the surface area of the proposed new container.
		\score{2}
		
		\vspace{25mm}
		
		\item[(e)]
		Determine which container uses less material. Give a reason for your answer.
		\score{1}
		
	\end{enumerate}
