		\item[(d)] Round \(7.38649\) to three significant figures.

	\section*{Question 3 --- Sequences and Functions}
	
	The first five terms of a sequence are
	
	\[
	5,\quad 9,\quad 13,\quad 17,\quad 21.
	\]
	
	\begin{enumerate}
		\item[(a)] Write down the next two terms.

		\vspace{12mm}
		
		\item[(b)] Find an expression for the \(n\)-th term of the sequence.
		\score{2}
		
		\vspace{15mm}
		
		\item[(c)] The function \(f\) is defined by
		\[
		f(x)=3x^2-2.
		\]
		
		Find \(f(4)\).
		\score{1}
		
		\vspace{15mm}
		
		\item[(d)] Solve
		\[
		f(x)=25.

		\score{1}

	\section*{Question 4 --- Ratio, Proportion and Percentages}
	
	A school has \(840\) students.
	
	The ratio of students in Year 10 to students in Year 11 is
	
	\[
	5:3.
	\]
	
	\begin{enumerate}
		\item[(a)] Calculate the number of Year 10 students.
		\score{2}
		
		\vspace{15mm}
		
		\item[(b)] \(40\%\) of the Year 10 students study French.
		
		Calculate the number of Year 10 students who study French.
		\score{2}
		
		\vspace{15mm}
		
		\item[(c)] The number of Year 11 students increases by \(12\%\) the following year.
		
		Calculate the new number of Year 11 students.
		\score{2}
		
		\vspace{15mm}
		
		\item[(d)] Explain why the answer in part (c) must be a whole number in this context.
		\score{1}
	\end{enumerate}
	
	\vfill
	
	\noindent
	\textbf{[7 marks]}

	\section*{Question 5 --- Geometry}
	
	A triangle has side lengths \(7\text{ cm}\), \(9\text{ cm}\), and \(12\text{ cm}\).
	
	\begin{enumerate}
		\item[(a)] Determine whether the triangle is right-angled.

		\score{2}
		
		\vspace{20mm}

		Calculate the length of the hypotenuse.
		\score{2}
		
		\vspace{15mm}
		
		\item[(c)] The angle between two sides of a triangle is \(42^\circ\).
		The sides surrounding this angle have lengths \(8\text{ cm}\) and
		\(11\text{ cm}\).
		
		Calculate the area of the triangle.
		\score{3}
	\end{enumerate}

	\section*{Question 7 --- Statistics}
	
	The following data show the number of minutes that 10 students spent
	completing a mathematical investigation.
	
	\[
	\begin{array}{cccccccccc}
		18 & 21 & 15 & 24 & 19 & 27 & 21 & 16 & 23 & 26
	\end{array}
	\]
	
	\begin{enumerate}
		\item[(a)] Find the median.
		\score{2}
		
		\vspace{15mm}
		
		\item[(b)] Find the range.
		\score{1}
		
		\vspace{10mm}
		
		\item[(c)] Calculate the mean.
		\score{2}
		
		\vspace{15mm}
		
		\item[(d)] A new student takes \(60\) minutes to complete the
		investigation.
		
		Explain how adding this value affects the mean and the median.
		\score{2}
	\end{enumerate}

	\section*{Question 8 --- Probability}
	
	A bag contains \(5\) red counters, \(3\) blue counters and \(2\) green
	counters.
	
	A counter is selected at random.
	
	\begin{enumerate}
		\item[(a)] Find the probability that the counter is blue.
		\score{1}
		
		\vspace{12mm}
		
		\item[(b)] Find the probability that the counter is not green.
		\score{2}
		
		\vspace{15mm}
		
		Two counters are selected without replacement.
		
		\item[(c)] Find the probability that both counters are red.
		\score{2}
		
		\vspace{15mm}
		
		\item[(d)] Find the probability that one counter is red and the other
		counter is blue.
		\score{2}
	\end{enumerate}

	\section*{Question 9 --- Problem Solving}
	
	A company manufactures rectangular storage boxes.
	
	Each box has length \(x\) cm, width \(8\) cm and height \(5\) cm.
	
	\begin{enumerate}
		\item[(a)] Write an expression for the volume \(V\) of the box.
		\score{1}
		
		\vspace{12mm}
		
		\item[(b)] The volume of one box is \(360\text{ cm}^3\).
		
		Find the value of \(x\).
		\score{2}
		
		\vspace{15mm}
		
		\item[(c)] The company wants to increase the length by \(25\%\)
		while keeping the width and height unchanged.
		
		Find the new volume of the box.
		\score{2}
		
		\vspace{15mm}
		
		\item[(d)] The boxes are packed into a larger rectangular container.
		The container has dimensions \(60\text{ cm}\), \(40\text{ cm}\) and
		\(30\text{ cm}\).
		
		Calculate the volume of the container.
		\score{1}
		
		\vspace{12mm}
		
		\item[(e)] The company claims that \(40\) of the boxes can fit inside
		the container.
		
		Use the volumes to determine whether this claim is possible.
		Explain your answer.
		\score{1}
	\end{enumerate}
