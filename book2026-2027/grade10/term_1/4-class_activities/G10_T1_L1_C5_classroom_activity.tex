		\section*{Learning Criterion}
		
		C4: Applies the Minimax Regret criterion to make decisions without
		probabilities, selecting the alternative with the smallest possible
		maximum regret. \vspace{-5mm}
		
		% ============================================================
		% SUMMARY
		% ============================================================
		
		\section{Minimax Regret}
		
		When probabilities are unavailable, the Minimax Regret criterion compares
		the regret associated with each alternative under each possible state of
		nature.

		\begin{center}
			\renewcommand{\arraystretch}{1.3}
			\small
			\begin{tabular}{p{0.20\columnwidth}p{0.68\columnwidth}}
				\toprule
				\textbf{Focus}
				& Maximum possible regret \\
				\textbf{First step}
				& Find the best payoff in each state of nature \\
				\textbf{Second step}
				& Construct the regret table \\
				\textbf{Third step}
				& Find the maximum regret for each alternative \\
				\textbf{Final step}
				& Choose the smallest maximum regret \\
				\bottomrule
			\end{tabular}
		\end{center}
		
		First, identify the best payoff in each state of nature:
		\[
		\boxed{
			b(S_j)
			=
			\max_i P_{ay}(A_i,S_j)
		}
		\]
		
		Then calculate the regret for each alternative and state:
		\[
		\boxed{
			r(A_i,S_j)
			=
			b(S_j)-P_{ay}(A_i,S_j)
		}
		\]
		
		For each alternative, find its maximum regret:
		\[
		\boxed{
			R(A_i)
			=
			\max_j r(A_i,S_j)
		}
		\]
		
		Finally, choose the alternative with the smallest maximum regret:
		\[
		\boxed{
			A^*_{\mathrm{Minimax\ Regret}}
			=
			\arg\min_i R(A_i)
		}
		\]
		
		% ============================================================
		% SOLVED EXAMPLE
		% ============================================================
		
		
		\section{Solved Example: Clothing Store}
		
		A local clothing store is deciding whether to launch a new advertising
		campaign before the holiday season or continue with its current marketing
		strategy. The manager knows that customer response to the campaign is
		uncertain.
		
		\[ S_1 : \text{Strong Sales} \qquad S_2 : \text{Weak Sales} \]
		\begin{center}
			\renewcommand{\arraystretch}{1.3}
			\small
			\begin{tabular}{lcc}
				\toprule
				Alternative & $S_1$ & $S_2$ \\
				\midrule
				Launch new campaign ($A_1$) & 55 & 5 \\
				Keep current strategy ($A_2$) & 35 & 22 \\
				\bottomrule
			\end{tabular}
		\end{center}
		
		All payoffs are expressed in thousands of dollars. What decision should the manager make using the Minimax Regret criterion?		

		\subsection{Solution}
		\textbf{Step 1: Identify the best payoff in each state of nature}
		\[ \text{Under strong sales: } \qquad
		b(S_1)
		=
		\max\{55,35\}
		=
		55
		\]
		\[ \text{Under weak sales: } \qquad
		b(S_2)
		=
		\max\{5,22\}
		=
		22
		\]
		
		\textbf{Step 2: Construct the opportunity loss table}
		
		Regret is calculated as
		\[
		r(A_i,S_j)
		=
		b(S_j)-P_{ay}(A_i,S_j).
		\]
		\[
		\begin{array}{|lll|}
			\toprule
			\text{For } A_1: & r(A_1,S_1) &= 55-55 = 0 \\[3pt]
			& r(A_1,S_2) &= 22-5 = 17 \\[6pt]
			\midrule
			\text{For } A_2: & r(A_2,S_1) &= 55-35 = 20 \\[3pt]
			& r(A_2,S_2) &= 22-22 = 0 \\
			\bottomrule
		\end{array}
		\quad
		\begin{array}{|lcc|}
			\toprule
			& S_1 & S_2\\
			\midrule
			A_1 & 0 & 17\\
			A_2 & 20 & 0\\
			\bottomrule
		\end{array}
		\]
	
		
		\textbf{Step 3: Find the maximum regret for each alternative}
		\[
		\begin{array}{lll}
			R(A_1) &= \max\{0,17\} &= 17 \\[3pt]
			R(A_2) &= \max\{20,0\} &= 20
		\end{array}
		\]
		
		\textbf{Step 4: Apply the Minimax Regret criterion}
		
		Choose the alternative with the smallest maximum regret:
		\[
		R^*
		=
		\min\{17,20\}
		=
		17
		\]
		Thus,
		\[
		A^*_{\mathrm{Minimax\ Regret}}
		=
		\arg\min_i R(A_i)
		=
		A_1
		\]
		
		The Minimax Regret criterion recommends launching the new campaign
		because its maximum possible regret is \$17,000, compared with
		\$20,000 for keeping the current strategy.
		
		\setlength{\columnseprule}{0pt}

	\[
	\begin{array}{|ll|}
		\toprule
		\text{For } A_1: &
		r(A_1,S_1) = 90-90=0
		\\
		&
		r(A_1,S_2) = 45-30=15
		\\[6pt]
		&
		r(A_1,S_3) = 28-(-15)=43
		\\
		\midrule
		\text{For } A_2: &
		r(A_2,S_1) = 90-65=25
		\\
		&
		r(A_2,S_2) = 45-45=0
		\\[6pt]
		&
		r(A_2,S_3) = 28-20=8
		\\
		\midrule
		\text{For } A_3: &
		r(A_3,S_1) = 90-45=45
		\\
		&
		r(A_3,S_2) = 45-38=7
		\\[6pt]
		&
		r(A_3,S_3) = 28-28=0
		\\
		\midrule
		\text{For } A_4: &
		r(A_4,S_1) = 90-20=70
		\\
		&
		r(A_4,S_2) = 45-20=25
		\\[6pt]
		&
		r(A_4,S_3) = 28-20=8
		\\
		\bottomrule
	\end{array}
	\]
		
		Therefore, the opportunity loss table is:\vspace{-3mm}
		\begin{center}
			\renewcommand{\arraystretch}{1.5}
			\small
			\begin{tabular}{lccc}
				\toprule
				Alternative & $S_1$ & $S_2$ & $S_3$ \\
				\midrule

				$A_2$ & $25$ & $0$ & $8$ \\
				$A_3$ & $45$ & $7$ & $0$ \\
				$A_4$ & $70$ & $25$ & $8$ \\
				\bottomrule
			\end{tabular}
		\end{center}
		
		\subsection*{Step 3: Find the maximum regret for each alternative}\vspace{-3mm}
		\[
		\begin{array}{lll}
			R(A_1) &= \max\{0,15,43\} &= 43 \\[6pt]
			R(A_2) &= \max\{25,0,8\} &= 25 \\[6pt]
			R(A_3) &= \max\{45,7,0\} &= 45 \\[6pt]
			R(A_4) &= \max\{70,25,8\} &= 70
		\end{array}
		\]
		
		\subsection*{Step 4: Apply the Minimax Regret criterion}
		
		Choose the alternative with the smallest maximum regret:
		\[
		R^*
		=
		\min\{43,25,45,70\}
		=
		25
		\]
		
		Thus,
		\[
		A^*_{\mathrm{Minimax\ Regret}}
		=
		\arg\min_i R(A_i)
		=
		A_2
		\]
