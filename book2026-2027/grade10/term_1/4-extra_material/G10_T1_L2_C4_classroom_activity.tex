	\section*{Learning Criterion}
	
	C4: Applies the Maximax and Maximin criteria to make optimistic or
	conservative decisions without probabilities, selecting the alternative
	with the highest possible payoff or the best worst-case outcome.
	
	% ============================================================
	% SUMMARY
	% ============================================================
	
	\section{Maximax and Maximin}
	
	When probabilities are unavailable, decisions can be made by comparing
	the payoffs for each alternative under each possible state of nature.
	
	\begin{center}
		\renewcommand{\arraystretch}{1.4}
		\begin{tabular}{p{0.20\textwidth}p{0.32\textwidth}p{0.32\textwidth}}
			\toprule
			& \textbf{Maximax} & \textbf{Maximin} \\
			\midrule
			Focus & Best possible payoff & Worst possible payoff \\
			Attitude & Optimistic & Conservative \\
			First step & Find the maximum in each row & Find the minimum in each row \\
			Final step & Choose the largest maximum & Choose the largest minimum \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	\[
	\boxed{
		M_i=\max_j P_{ay}(A_i,S_j)
		\qquad
		A^*_{\mathrm{Maximax}}=\arg\max_i M_i
	}
	\]
	
	\[
	\boxed{
		m_i=\min_j P_{ay}(A_i,S_j)
		\qquad
		A^*_{\mathrm{Maximin}}=\arg\max_i m_i
	}
	\]

	\section{Solved Example: Coffee Shop}
	
	A coffee shop must decide whether to launch a dessert line or keep its
	current menu. Customer traffic could be high or low.
	
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lcc}
			\toprule
			Alternative & High traffic ($S_1$) & Low traffic ($S_2$) \\
			\midrule
			Launch dessert line ($A_1$) & 42 & 8 \\
			Keep current menu ($A_2$) & 28 & 20 \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	All payoffs are in thousands of dollars.
	
	\subsection*{Step 1: Identify}
	
	\[
	A_1=\text{Launch dessert line}
	\qquad
	A_2=\text{Keep current menu}
	\]
	
	\[
	S_1=\text{High traffic}
	\qquad
	S_2=\text{Low traffic}
	\]
	
	\subsection*{Steps 2--3: Complete the analysis}
	
	\begin{center}
		\renewcommand{\arraystretch}{1.6}
		\begin{tabular}{lcccc}
			\toprule
			Alternative & $S_1$ & $S_2$ & Max & Min \\
			\midrule
			$A_1$ & 42 & 8 & $\max\{42,8\}=42$ & $\min\{42,8\}=8$ \\
			$A_2$ & 28 & 20 & $\max\{28,20\}=28$ & $\min\{28,20\}=20$ \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	\begin{multicols}{2}
		
		\textbf{Maximax}
		
		\[
		42>28
		\]
		
		\[
		\boxed{
			A^*_{\mathrm{Maximax}}=A_1
		}
		\]
		
		Choose the alternative with the highest possible payoff.
		
		\columnbreak
		
		\textbf{Maximin}
		
		\[
		20>8
		\]
		
		\[
		\boxed{
			A^*_{\mathrm{Maximin}}=A_2
		}
		\]
		
		Choose the alternative with the highest worst-case payoff.
		
	\end{multicols}

		\[

		\]
		
		\[
		A^*_{\mathrm{Maximin}}
		=\underline{\hspace{2cm}}
		\]
		
	\end{multicols}
