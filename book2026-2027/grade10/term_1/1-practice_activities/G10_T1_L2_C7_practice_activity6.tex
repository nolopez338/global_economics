	% ============================================================
	% INTRODUCTION
	% ============================================================
	
	\section{Expected Value and Sensitivity Analysis}
	
	When probabilities are subjective or estimated, a decision-maker can
	examine how changes in the probability assumptions affect the preferred
	alternative. The Expected Value criterion allows each alternative to be
	represented as a function of the probability assigned to the possible
	states of nature. For a payoff table with alternatives in rows and states of nature in columns, let
	$
	P_{ay}(A_i,S_j)
	$
	represent the payoff obtained when alternative $A_i$ is selected and
	state of nature $S_j$ occurs.
	
	Let $p$ represent the subjective probability that state of nature $S_1$
	occurs. Since there are two possible states,
	\begin{center}
		$P(S_1)=p$ \qquad and \qquad $P(S_2)=1-p$ \qquad with \quad $0\leq p\leq 1$
	\end{center}
	
	For each alternative, calculate its Expected Value as a function of $p$:
	$$
	EV(A_i)
	=
	pP_{ay}(A_i,S_1)
	+
	(1-p)P_{ay}(A_i,S_2).
	$$
	
	This expression shows how the expected payoff changes when the
	subjective probability of $S_1$ changes. The preferred alternative depends on the value of $p$. To determine
	which alternative is preferred, compare their expected-value functions.	For two alternatives $A_1$ and $A_2$, the threshold probability is found
	by solving
	$$
	EV(A_1)=EV(A_2).
	$$
	
	At this threshold, the decision-maker is indifferent between the two
	alternatives. The preferred alternative on each side of the threshold
	is determined by comparing the corresponding expected values. Therefore, sensitivity analysis identifies how changes in the subjective
	probability affect the preferred decision.

	\section{General 2x2 Problem}
	
	Consider the following payoff table, where the probability of state
	$S_1$ is $p$ and the probability of state $S_2$ is $1-p$:
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lcc}
			\toprule
			Alternative
			& \begin{tabular}{c}
				$S_1$\\
				$(p)$
			\end{tabular}
			& \begin{tabular}{c}
				$S_2$\\
				$(1-p)$
			\end{tabular}
			\\
			\midrule
			$A_1$
			& $P_{ay}(A_1,S_1)$
			& $P_{ay}(A_1,S_2)$
			\\
			$A_2$
			& $P_{ay}(A_2,S_1)$
			& $P_{ay}(A_2,S_2)$
			\\
			\bottomrule
		\end{tabular}
	\end{center}
	
	Determine the preferred alternative as a function of $p$, including the
	threshold probability at which the decision-maker is indifferent.
	
	% ============================================================
	% SOLUTION 2x2
	% ============================================================
	
	\subsection{Solution}
	
	\textbf{Step 1: Identify the alternatives, states of nature, and probabilities}
	
	Alternatives: $A_1, A_2$ \qquad
	States of nature: $S_1, S_2$
	
	Probabilities: $P(S_1) = p$	\qquad	$P(S_2) = 1-p$
	
	\noindent
	\setlength{\columnseprule}{0.4pt}
	
	\begin{multicols}{2}
		\vspace{0pt}
		
		\textbf{Step 2: Calculate the expected value of $A_1$}
		
		For alternative $A_1$:
		
		\[
		EV(A_1)
		=
		pP_{ay}(A_1,S_1)
		+
		(1-p)P_{ay}(A_1,S_2).
		\]
		
		This expression shows how the expected payoff of $A_1$ changes as
		$p$ changes.
		
		\columnbreak
		
		\textbf{Step 3: Calculate the expected value of $A_2$}
		
		For alternative $A_2$:
		
		\[
		EV(A_2)
		=
		pP_{ay}(A_2,S_1)
		+
		(1-p)P_{ay}(A_2,S_2).
		\]
		
		This expression shows how the expected payoff of $A_2$ changes as
		$p$ changes.
		
	\end{multicols}
	
	\setlength{\columnseprule}{0pt}
	
	\textbf{Step 4: Find the threshold probability}
	
	The threshold occurs when the two alternatives have the same expected
	value:
	$$
	EV(A_1)=EV(A_2).
	$$
	
	Solve this equation for $p$. The resulting value is the probability at
	which the decision-maker is indifferent between the two alternatives.
	
	\textbf{Step 5: Perform the sensitivity analysis}
	
	Compare the expected values on each side of the threshold.
	
	\begin{center}
		\begin{minipage}{0.4\textwidth}
			\[
			\begin{aligned}
				EV(A_1)>EV(A_2) &\quad\Rightarrow\quad \text{choose } A_1,\\
				EV(A_2)>EV(A_1) &\quad\Rightarrow\quad \text{choose } A_2,\\
				EV(A_1)=EV(A_2) &\quad\Rightarrow\quad \text{indifferent}.
			\end{aligned}
			\]
		\end{minipage}
		$ \qquad \longrightarrow \qquad$
		\begin{minipage}{0.4\textwidth}
			\textbf{Decision rule}
			\[
			\begin{cases}
				A_1, & \text{if } EV(A_1)>EV(A_2),\\
				A_2, & \text{if } EV(A_2)>EV(A_1),\\
				\text{indifferent}, & \text{if } EV(A_1)=EV(A_2).
			\end{cases}
			\]
		\end{minipage}
	\end{center}
	
	Thus, the preferred alternative can be expressed as a function of the
	subjective probability $p$.

	\section{Contextualized 2x2 Problem - Coffee shop}
	
	A small coffee shop is considering whether to introduce a new dessert line
	or keep its current menu. The manager is uncertain about the level of
	customer traffic during the promotional weekend. Let $p$ represent the
	subjective probability of high customer traffic. Therefore, the probability
	of low customer traffic is $1-p$. All payoffs are expressed in thousands of
	dollars.
	Using the Expected Value criterion, determine how the preferred decision
	changes as the subjective probability $p$ changes. Identify the threshold
	probability at which the manager is indifferent between the two alternatives.
	\vspace{-4mm}
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lcc}
			\toprule
			Alternative
			& \begin{tabular}{c}High traffic $S_1$ \\ $P(S_1)=p$\end{tabular}
			& \begin{tabular}{c}Low traffic $S_2$ \\ $P(S_2)=1-p$\end{tabular}
			\\
			\midrule
			Launch dessert line ($A_1$) & 40 & 10 \\
			Keep current menu ($A_2$) & 30 & 20 \\
			\bottomrule
		\end{tabular}
	\end{center}
	\vspace{-5mm}
	
	% ============================================================
	% SOLUTION 2x2
	% ============================================================
	
	\subsection{Solution}
	
	\textbf{Step 1: Identify the alternatives, states of nature, and probabilities}
	
	Alternatives:
	$A_1$ = Launch dessert line,
	$A_2$ = Keep current menu
	
	States of nature:
	$S_1$ = High traffic,
	$S_2$ = Low traffic
	
	Probabilities:
	$
	P(S_1)=p,
	P(S_2)=1-p
	$
	with
	$
	0\leq p\leq1.
	$
	
	\vspace{-3mm}
	
	\noindent
	\setlength{\columnseprule}{0.4pt} 
	\begin{multicols}{2}
		\vspace{0pt}
		
		\textbf{Step 2: Calculate the expected value of $A_1$}
		
		For the Launch dessert line alternative:
		\[
		EV(A_1)
		=
		p(40)
		+
		(1-p)(10)
		\]
		\[
		=
		40p+10-10p
		=
		30p+10.
		\]
		
		Thus, the expected payoff from launching the dessert line is
		$30p+10$ thousand dollars.
		
		\columnbreak
		
		\textbf{Step 3: Calculate the expected value of $A_2$}
		
		For the Keep current menu alternative:
		\[
		EV(A_2)
		=
		p(30)
		+
		(1-p)(20)
		\]
		\[
		=
		30p+20-20p
		=
		10p+20.
		\]
		
		Thus, the expected payoff from keeping the current menu is
		$10p+20$ thousand dollars.
		
	\end{multicols}
	
	\EMVGraph{
		{30}{10}{blue}{$EMV(A_1)=30p+10$},
		{10}{20}{red}{$EMV(A_2)=10p+20$}
	}
	
	\setlength{\columnseprule}{0pt}
	
	\textbf{Step 4: Find the threshold probability}
	
	The threshold occurs when the two alternatives have the same expected
	value:
	\[
	EV(A_1)=EV(A_2)
	\qquad \longrightarrow \qquad
	30p+10=10p+20
	\]
	\[
	\qquad \longrightarrow \qquad
	30p-10p=20-10
	\qquad \longrightarrow \qquad
	20p=10
	\qquad \longrightarrow \qquad
	p=\frac{10}{20}
	=\frac{1}{2}
	=0.5
	\]
	
	Thus, the manager is indifferent between the two alternatives when the
	subjective probability of high customer traffic is $50\%$.
	
	\textbf{Step 5: Perform the sensitivity analysis}
	
	Compare the expected values on each side of the threshold.
	
	\[
	\begin{aligned}
		p<0.5 &\quad\Rightarrow\quad EV(A_1)<EV(A_2) \quad\Rightarrow\quad \text{choose } A_2,\\
		p>0.5 &\quad\Rightarrow\quad EV(A_1)>EV(A_2) \quad\Rightarrow\quad \text{choose } A_1,\\
		p=0.5 &\quad\Rightarrow\quad EV(A_1)=EV(A_2) \quad\Rightarrow\quad \text{indifferent}.
	\end{aligned}
	\qquad \longrightarrow \qquad
	\boxed{
		\begin{cases}
			A_2, & 0\leq p<0.5,\\
			\text{indifferent}, & p=0.5,\\
			A_1, & 0.5<p\leq1.
	\end{cases}}
	\]
	
	The sensitivity analysis shows that the coffee shop should keep its current
	menu when the subjective probability of high traffic is below $50\%$.
	When that probability exceeds $50\%$, launching the dessert line has the
	higher expected value.

	value:
	$$
	EV(A_1)=EV(A_2)
	\qquad \longrightarrow \qquad
	50p+5=45
	$$
	$$
	\qquad \longrightarrow \qquad
	50p=40
	\qquad \longrightarrow \qquad
	p=0.8.
	$$
	
	Thus, the manager is indifferent between the two alternatives when the
	subjective probability of strong sales is $80\%$.
	
	\textbf{Step 5: Perform the sensitivity analysis}
	
	Compare the expected values on each side of the threshold.
	$$
	\begin{aligned}
		p<0.800 &\quad\Rightarrow\quad EV(A_1)<EV(A_2) \quad\Rightarrow\quad \text{choose } A_2,\\
		p>0.800 &\quad\Rightarrow\quad EV(A_1)>EV(A_2) \quad\Rightarrow\quad \text{choose } A_1,\\
		p=0.800 &\quad\Rightarrow\quad EV(A_1)=EV(A_2) \quad\Rightarrow\quad \text{indifferent}.
	\end{aligned}
	\qquad \longrightarrow \qquad 
	\boxed{
		\begin{cases}
			A_2, & 0\leq p<0.800,\\
			\text{indifferent}, & p=0.800,\\
			A_1, & 0.800<p\leq1.
	\end{cases}}
	$$
	
	The sensitivity analysis shows that the clothing store should keep its current
	marketing strategy when the subjective probability of strong sales is below
	$80\%$. Only when that probability exceeds $80\%$ does launching the new
	advertising campaign have the higher expected value.

	\section{Contextualized 3x2 Problem - Technology company}
	
	A technology company is deciding which product strategy to adopt for the launch
	of a new device. The manager is uncertain about the level of demand during the
	launch period. Let $p$ represent the subjective probability of high demand.
	Therefore, the probability of low demand is $1-p$. All payoffs are expressed
	in thousands of dollars.
	
	Using the Expected Value criterion, determine how the preferred decision
	changes as the subjective probability $p$ changes. Identify the threshold
	probabilities at which the manager is indifferent between the alternatives.
	
	\vspace{-2mm}
	
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lcc}
			\toprule
			Alternative
			& \begin{tabular}{c}
				High demand $S_1$ \\
				$P(S_1)=p$
			\end{tabular}
			& \begin{tabular}{c}
				Low demand $S_2$ \\
				$P(S_2)=1-p$
			\end{tabular}
			\\
			\midrule
			Premium version ($A_1$) & 80 & -20 \\
			Standard version ($A_2$) & 60 & 18 \\
			Budget version ($A_3$) & 40 & 25 \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	\vspace{-5mm}
	
	% ============================================================
	% SOLUTION 3x2
	% ============================================================
	
	\subsection{Solution}
	
	\textbf{Step 1: Identify the alternatives, states of nature, and probabilities}
	
	Alternatives:
	$A_1$ = Premium version,
	$A_2$ = Standard version,
	$A_3$ = Budget version
	
	States of nature:
	$S_1$ = High demand,
	$S_2$ = Low demand
	
	Probabilities:
	$
	P(S_1)=p,
	P(S_2)=1-p
	$
	with
	$
	0\leq p\leq1.
	$
	
	\vspace{-3mm}
	
	\noindent
	\setlength{\columnseprule}{0.4pt}
	\begin{multicols}{2}
		\vspace{0pt}
		
		\textbf{Step 2: Calculate the expected value of $A_1$}
		
		For the Premium version alternative:
		\[
		EV(A_1)
		=
		p(80)
		+
		(1-p)(-20)
		\]
		\[
		=
		80p-20+20p
		\]
		\[
		=
		100p-20.
		\]
		
		Thus, the expected payoff from the Premium version is
		$100p-20$ thousand dollars.
		
		\columnbreak
		
		\textbf{Step 3: Calculate the expected value of $A_2$}
		
		For the Standard version alternative:
		\[
		EV(A_2)
		=
		p(60)
		+
		(1-p)(18)
		\]
		\[
		=
		60p+18-18p
		\]
		\[
		=
		42p+18.
		\]
		
		Thus, the expected payoff from the Standard version is
		$42p+18$ thousand dollars.
		
	\end{multicols}
	
	\setlength{\columnseprule}{0pt}
	
	\begin{multicols}{2}
		\vspace{0pt}
		
		\textbf{Step 4: Calculate the expected value of $A_3$}
		
		For the Budget version alternative:
		\[
		EV(A_3)
		=
		p(40)
		+
		(1-p)(25)
		\]
		\[
		=
		40p+25-25p
		\]
		\[
		=
		15p+25.
		\]
		
		Thus, the expected payoff from the Budget version is
		$15p+25$ thousand dollars.
		
		\columnbreak
		
		\textbf{Step 5: Find the threshold probabilities}
		
		The thresholds occur when two alternatives have the same expected
		value.
		
		For $A_2$ and $A_3$:
		\[
		EV(A_2)=EV(A_3)
		\]
		\[
		42p+18=15p+25
		\]
		\[
		27p=7
		\]
		\[
		p=\frac{7}{27}
		\approx0.259.
		\]
		
	\end{multicols}
	
	\textbf{Step 6: Find the second threshold probability}
	
	For $A_1$ and $A_2$:
	\[
	EV(A_1)=EV(A_2)
	\]
	\[
	100p-20=42p+18
	\]
	\[
	58p=38
	\]
	\[
	p=\frac{38}{58}
	=\frac{19}{29}
	\approx0.655.
	\]
	
	Thus, the manager is indifferent between $A_2$ and $A_3$ when the
	subjective probability of high demand is approximately $25.9\%$,
	and indifferent between $A_1$ and $A_2$ when it is approximately
	$65.5\%$.
	
	\textbf{Step 7: Perform the sensitivity analysis}
	
	Compare the expected values on each side of the thresholds.
	
	$$
	\begin{aligned}
		p<0.259
		&\quad\Rightarrow\quad
		EV(A_3)>EV(A_2),EV(A_1)
		\quad\Rightarrow\quad
		\text{choose }A_3,
		\\
		p=0.259
		&\quad\Rightarrow\quad
		EV(A_2)=EV(A_3)
		\quad\Rightarrow\quad
		\text{indifferent between }A_2\text{ and }A_3,
		\\
		0.259<p<0.655
		&\quad\Rightarrow\quad
		EV(A_2)>EV(A_1),EV(A_3)
		\quad\Rightarrow\quad
		\text{choose }A_2,
		\\
		p=0.655
		&\quad\Rightarrow\quad
		EV(A_1)=EV(A_2)
		\quad\Rightarrow\quad
		\text{indifferent between }A_1\text{ and }A_2,
		\\
		p>0.655
		&\quad\Rightarrow\quad
		EV(A_1)>EV(A_2),EV(A_3)
		\quad\Rightarrow\quad
		\text{choose }A_1.
	\end{aligned}
	$$
	
	Therefore, the preferred decision depends on the subjective probability
	of high demand:
	
	$$
	\boxed{
		\begin{cases}
			A_3, & 0\leq p<0.259,\\
			\text{indifferent between }A_2\text{ and }A_3,
			& p=0.259,\\
			A_2, & 0.259<p<0.655,\\
			\text{indifferent between }A_1\text{ and }A_2,
			& p=0.655,\\
			A_1, & 0.655<p\leq1.
	\end{cases}}
	$$
	
	The sensitivity analysis shows that the company should choose the Budget
	version when the subjective probability of high demand is below approximately
	$25.9\%$. Between $25.9\%$ and $65.5\%$, the Standard version has the highest
	expected value. When the probability of high demand exceeds $65.5\%$, the
	Premium version becomes the preferred decision.
