	C5: Constructs opportunity loss tables and applies the Minimax regret criterion to select the alternative that minimises the maximum opportunity loss.
	
	% ============================================================
	% INTRODUCTION
	% ============================================================
	\section{Minimax Regret Criterion}
	
	When probabilities are not available, a decision-maker can compare
	alternatives by considering the opportunity loss, or regret, associated
	with each possible state of nature. Regret measures the payoff that could have been obtained by choosing the
	best alternative for the state of nature that actually occurs, minus the
	payoff obtained from the selected alternative.
	
	For a payoff table with alternatives in rows and states of nature in
	columns, let
	$
	P_{ay}(A_i,S_j)
	$
	represent the payoff obtained when alternative $A_i$ is selected and
	state of nature $S_j$ occurs.

	For each state of nature, first identify the best available payoff:
	\[
	\begin{aligned}
		b(S_j)
		&=
		\max_i P_{ay}(A_i,S_j)
		\\
		&=
		\max
		\left\{
		P_{ay}(A_1,S_j),
		P_{ay}(A_2,S_j),
		\ldots,
		P_{ay}(A_m,S_j)
		\right\}.
	\end{aligned}
	\]
	
	The regret associated with alternative $A_i$ under state $S_j$ is then
	\[
	r(A_i,S_j)
	=
	b(S_j)-P_{ay}(A_i,S_j).
	\]
	
	For each alternative, determine its maximum possible regret:
	\[
	\begin{aligned}
		R(A_i)
		&=
		\max_j r(A_i,S_j)
		=
		\max_j
		\left\{
		b(S_j)-P_{ay}(A_i,S_j)
		\right\}
		\\
		&=
		\max
		\left\{
		r(A_i,S_1),
		r(A_i,S_2),
		\ldots,
		r(A_i,S_n)
		\right\}.
	\end{aligned}
	\]
	
	The Minimax Regret criterion selects the alternative whose maximum regret
	is as small as possible:
	
	\[
	A^*_{\mathrm{Minimax\ Regret}}
	=
	\arg\min_i R(A_i).
	\]
	
	Thus, the procedure is:
	
	\[
	\boxed{
		\begin{aligned}
			&\text{Best payoff in each state}
			\;\longrightarrow\;
			\text{Regret table}
			\;\longrightarrow\;
			\text{Maximum regret for each alternative}
			\\
			&\text{Minimum of those maximum regrets}
		\end{aligned}
	}
	\]
	
	The selected alternative minimizes the largest opportunity loss that
	could result from the decision.

	\section{General 2x2 Problem}
	
	Consider the following payoff table:
	
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lcc}
			\toprule
			Alternative & $S_1$ & $S_2$ \\
			\midrule
			$A_1$
			& $P_{ay}(A_1,S_1)$
			& $P_{ay}(A_1,S_2)$
			\\
			
			$A_2$
			& $P_{ay}(A_2,S_1)$
			& $P_{ay}(A_2,S_2)$
			\\
			\bottomrule
		\end{tabular}
	\end{center}
	
	Construct the opportunity loss table and determine the decision using
	the Minimax Regret criterion.
	
	% ============================================================
	% SOLUTION 2x2
	% ============================================================
	
	\noindent
	\setlength{\columnseprule}{0.4pt}
	
	\begin{multicols}{2}
		
	\subsection{Solution}
	
	\textbf{Step 1: Identify the best payoff in each state of nature}
	
	Under $S_1$, compare the payoffs of $A_1$ and $A_2$:
	\[
	b(S_1)
	=
	\max
	\left\{
	P_{ay}(A_1,S_1),
	P_{ay}(A_2,S_1)
	\right\}.
	\]
	
	Under $S_2$, compare the payoffs of $A_1$ and $A_2$
	\[
	b(S_2)
	=
	\max
	\left\{
	P_{ay}(A_1,S_2),
	P_{ay}(A_2,S_2)
	\right\}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		b(S_1)
		&=
		\max
		\left\{
		P_{ay}(A_1,S_1),
		P_{ay}(A_2,S_1)
		\right\},
		\\
		b(S_2)
		&=
		\max
		\left\{
		P_{ay}(A_1,S_2),
		P_{ay}(A_2,S_2)
		\right\}.
	\end{aligned}
	\]
	
		\vspace{0pt}
		
		\textbf{Step 2: Construct the opportunity loss table}
		
		For each alternative and state of nature, subtract the payoff from
		the best payoff available under that state:
		\[
		r(A_i,S_j)
		=
		b(S_j)-P_{ay}(A_i,S_j).
		\]
		
		Thus,
		\[
		r(A_1,S_1)
		=
		b(S_1)-P_{ay}(A_1,S_1),
		\]
		\[
		r(A_1,S_2)
		=
		b(S_2)-P_{ay}(A_1,S_2),
		\]
		\[
		r(A_2,S_1)
		=
		b(S_1)-P_{ay}(A_2,S_1),
		\]
		\[
		r(A_2,S_2)
		=
		b(S_2)-P_{ay}(A_2,S_2).
		\]
		
				\columnbreak
				
		\textbf{Step 3: Find the maximum regret for each alternative}
		
		\[
		\text{For} A_1,
		\qquad
		R(A_1)
		=
		\max
		\left\{
		r(A_1,S_1),r(A_1,S_2)
		\right\}.
		\]
		
		For $A_2$,
		\[
		R(A_2)
		=
		\max
		\left\{
		r(A_2,S_1),r(A_2,S_2)
		\right\}.
		\]
		
		Therefore,
		
		\[
		\begin{aligned}
			R(A_1)
			&=
			\max
			\left\{
			b(S_1)-P_{ay}(A_1,S_1),
			b(S_2)-P_{ay}(A_1,S_2)
			\right\},
			\\
			R(A_2)
			&=
			\max
			\left\{
			b(S_1)-P_{ay}(A_2,S_1),
			b(S_2)-P_{ay}(A_2,S_2)
			\right\}.
		\end{aligned}
		\]
		
		\textbf{Step 4: Apply the Minimax Regret criterion}
		
		Compare the maximum regrets:
		
		\[
		R^*
		=
		\min
		\left\{
		R(A_1),R(A_2)
		\right\}.
		\]
		
		The Minimax Regret decision is
		
		\[
		A^*_{\mathrm{Minimax\ Regret}}
		=
		\arg\min_i
		\left\{
		R(A_i)
		\right\}.
		\]
		
		Thus, choose $A_1$ if $R(A_1)<R(A_2)$, choose $A_2$ if
		$R(A_2)<R(A_1)$, and report a tie if $R(A_1)=R(A_2)$.
		
		\textbf{Opportunity loss table}
		
		\begin{center}
			\small
			\renewcommand{\arraystretch}{1.5}
			\setlength{\tabcolsep}{5pt}
			
			\begin{tabular}{lcc|c}
				\toprule
				Alternative & $S_1$ & $S_2$ & Maximum Regret \\
				\midrule
				
				$A_1$
				&
				$r(A_1,S_1)$
				&
				$r(A_1,S_2)$
				&
				$R(A_1)$
				\\
				
				$A_2$
				&
				$r(A_2,S_1)$
				&
				$r(A_2,S_2)$
				&
				$R(A_2)$
				\\
				
				\bottomrule
			\end{tabular}
		\end{center}
		
	\end{multicols}

	\section{Contextualized 2x2 Problem - Coffee shop}
	
	A small coffee shop is considering whether to introduce a new dessert line or keep its current menu. The manager knows that the level of customer traffic during the promotional weekend is uncertain and has no reliable information about the probability of high or low traffic.
	
	What decision should the manager make using the Minimax Regret criterion?
	
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lcc}
			\toprule
			Alternative & High traffic ($S_1$) & Low traffic ($S_2$) \\
			\midrule
			Launch dessert line ($A_1$) & 42 & 8 \\
			Keep current menu ($A_2$) & 28 & 20 \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	All payoffs are expressed in thousands of dollars.
	
	Determine the decision using the Minimax Regret criterion.
	
	\subsection{Solution}
	
	\textbf{Step 1: Identify the alternatives and states of nature}
	
	Alternatives: $A_1$ = Launch dessert line, $A_2$ = Keep current menu
	\qquad
	States of nature: $S_1$ = High traffic, $S_2$ = Low traffic.
	
	\noindent
	\setlength{\columnseprule}{0.4pt}
	
	\begin{multicols}{2}
		\vspace{0pt}
		
		\textbf{Step 2: Identify the best payoff in each state of nature}
		
		Under high traffic, compare the payoffs of both alternatives:
		
		\[
		b(S_1)
		=
		\max\{42,28\}
		=
		42.
		\]
		
		Under low traffic:
		
		\[
		b(S_2)
		=
		\max\{8,20\}
		=
		20.
		\]
		
		Thus, the best payoff in each state is
		
		\[
		\begin{aligned}
			b(S_1)&=42,\\
			b(S_2)&=20.
		\end{aligned}
		\]
		
		\textbf{Step 3: Construct the opportunity loss table}
		
		Regret is calculated as
		
		\[
		r(A_i,S_j)
		=
		b(S_j)-P_{ay}(A_i,S_j).
		\]
		
		For $A_1$:
		
		\[
		r(A_1,S_1)
		=
		42-42
		=
		0,
		\]
		
		\[
		r(A_1,S_2)
		=
		20-8
		=
		12.
		\]
		
		For $A_2$:
		
		\[
		r(A_2,S_1)
		=
		42-28
		=
		14,
		\]
		
		\[
		r(A_2,S_2)
		=
		20-20
		=
		0.
		\]
		
		Therefore, the opportunity loss table is
		
		\[
		\begin{array}{lcc}
			\toprule
			& S_1 & S_2\\
			\midrule
			A_1 & 0 & 12\\
			A_2 & 14 & 0\\
			\bottomrule
		\end{array}
		\]
		
		\columnbreak
		
		\textbf{Step 4: Find the maximum regret for each alternative}
		
		For $A_1$:
		
		\[
		R(A_1)
		=
		\max\{0,12\}
		=
		12.
		\]
		
		For $A_2$:
		
		\[
		R(A_2)
		=
		\max\{14,0\}
		=
		14.
		\]
		
		Therefore,
		
		\[
		\begin{aligned}
			R(A_1)&=12,\\
			R(A_2)&=14.
		\end{aligned}
		\]
		
		\textbf{Step 5: Apply the Minimax Regret criterion}
		
		Choose the alternative with the smallest maximum regret:
		
		\[
		R^*
		=
		\min\{12,14\}
		=
		12.
		\]
		
		Thus,
		
		\[
		A^*_{\mathrm{Minimax\ Regret}}
		=
		\arg\min_i R(A_i)
		=
		A_1.
		\]
		
		The Minimax Regret criterion recommends launching the dessert line because its maximum possible regret is \$12,000, compared with \$14,000 for keeping the current menu.
		
	\end{multicols}
	
	\setlength{\columnseprule}{0pt}
	
	\textbf{Extended opportunity loss table}
	
	\begin{center}
		\small
		\renewcommand{\arraystretch}{1.5}
		\setlength{\tabcolsep}{5pt}
		
		\begin{tabular}{lccc}
			\toprule
			Alternative & $S_1$ & $S_2$ & Maximum Regret \\
			\midrule
			
			$A_1$: Launch dessert line
			& $0$
			& $12$
			& $\max\{0,12\}=12$
			\\
			
			$A_2$: Keep current menu

	\section{Contextualized 4x3 Problem - Technology company}
	
	A technology company is deciding which product strategy to adopt for the launch of a new device. The manager knows that market conditions are uncertain and has no reliable information about the probability of each possible level of demand. The company is considering four possible strategies: launching a premium version, launching a standard version, launching a budget version, or delaying the launch.
	
	The level of market demand could be high, medium, or low. The estimated profit for each strategy under each possible market condition is shown in the table below.
	
	What decision should the company make using the Minimax Regret criterion?
	
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lccc}
			\toprule
			Alternative & High demand ($S_1$) & Medium demand ($S_2$) & Low demand ($S_3$) \\
			\midrule
			Premium version ($A_1$) & 80 & 35 & -20 \\
			Standard version ($A_2$) & 60 & 42 & 18 \\
			Budget version ($A_3$) & 40 & 32 & 25 \\
			Delay launch ($A_4$) & 15 & 15 & 15 \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	All payoffs are expressed in thousands of dollars.
	
	Determine the decision using the Minimax Regret criterion.
	
	% ============================================================
	% SOLUTION 4x3
	% ============================================================
	
	\subsection{Solution}
	
	\textbf{Step 1: Identify the alternatives and states of nature}
	
	Alternatives: $A_1$ = Premium version, $A_2$ = Standard version,
	$A_3$ = Budget version, $A_4$ = Delay launch
	
	\qquad
	
	States of nature: $S_1$ = High demand, $S_2$ = Medium demand,
	$S_3$ = Low demand.
	
	\noindent
	\setlength{\columnseprule}{0.4pt}
	
	\begin{multicols}{2}
		\vspace{0pt}
		
		\textbf{Step 2: Identify the best payoff in each state of nature}
		
		Under high demand, compare the payoffs of all alternatives:
		
		\[
		b(S_1)
		=
		\max\{80,60,40,15\}
		=
		80.
		\]
		
		Under medium demand:
		
		\[
		b(S_2)
		=
		\max\{35,42,32,15\}
		=
		42.
		\]
		
		Under low demand:
		
		\[
		b(S_3)
		=
		\max\{-20,18,25,15\}
		=
		25.
		\]
		
		Thus,
		
		\[
		\begin{aligned}
			b(S_1)&=80,\\
			b(S_2)&=42,\\
			b(S_3)&=25.
		\end{aligned}
		\]
		
		\textbf{Step 3: Construct the opportunity loss table}
		
		Regret is calculated as
		
		\[
		r(A_i,S_j)
		=
		b(S_j)-P_{ay}(A_i,S_j).
		\]
		
		For $A_1$:
		
		\[
		r(A_1,S_1)=80-80=0,
		\]
		\[
		r(A_1,S_2)=42-35=7,
		\]
		\[
		r(A_1,S_3)=25-(-20)=45.
		\]
		
		For $A_2$:
		
		\[
		r(A_2,S_1)=80-60=20,
		\]
		\[
		r(A_2,S_2)=42-42=0,
		\]
		\[
		r(A_2,S_3)=25-18=7.
		\]
		
		For $A_3$:
		
		\[
		r(A_3,S_1)=80-40=40,
		\]
		\[
		r(A_3,S_2)=42-32=10,
		\]
		\[
		r(A_3,S_3)=25-25=0.
		\]
		
		For $A_4$:
		
		\[
		r(A_4,S_1)=80-15=65,
		\]
		\[
		r(A_4,S_2)=42-15=27,
		\]
		\[
		r(A_4,S_3)=25-15=10.
		\]
		
		Therefore, the opportunity loss table is
		
		\[
		\begin{array}{lccc}
			\toprule
			& S_1 & S_2 & S_3\\
			\midrule
			A_1 & 0 & 7 & 45\\
			A_2 & 20 & 0 & 7\\
			A_3 & 40 & 10 & 0\\
			A_4 & 65 & 27 & 10\\
			\bottomrule
		\end{array}
		\]
		
		\columnbreak
		
		\textbf{Step 4: Find the maximum regret for each alternative}
		
		For $A_1$:
		
		\[
		R(A_1)
		=
		\max\{0,7,45\}
		=
		45.
		\]
		
		For $A_2$:
		
		\[
		R(A_2)
		=
		\max\{20,0,7\}
		=
		20.
		\]
		
		For $A_3$:
		
		\[
		R(A_3)
		=
		\max\{40,10,0\}
		=
		40.
		\]
		
		For $A_4$:
		
		\[
		R(A_4)
		=
		\max\{65,27,10\}
		=
		65.
		\]
		
		Therefore,
		
		\[
		\begin{aligned}
			R(A_1)&=45,\\
			R(A_2)&=20,\\
			R(A_3)&=40,\\
			R(A_4)&=65.
		\end{aligned}
		\]
		
		\textbf{Step 5: Apply the Minimax Regret criterion}
		
		Choose the alternative with the smallest maximum regret:
		
		\[
		R^*
		=
		\min\{45,20,40,65\}
		=
		20.
		\]
		
		Thus,
		
		\[
		A^*_{\mathrm{Minimax\ Regret}}
		=
		\arg\min_i R(A_i)
		=
		A_2.
		\]
		
		The Minimax Regret criterion recommends launching the standard version because its maximum possible regret is \$20,000, which is lower than the maximum regret of every other strategy.
		
	\end{multicols}
