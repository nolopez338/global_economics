	\section{Introduction}
	\label{sec:introduction}
	
	This section presents the general procedure for analysing a payoff table under
	uncertainty. The alternatives are denoted by \(A_i\), the states of nature by
	\(S_j\), and the payoff obtained by choosing alternative \(A_i\) when state
	\(S_j\) occurs by
	\[
	P(A_i,S_j).
	\]
	
	The analysis uses three decision criteria:
	
	\begin{enumerate}
		\item \textbf{Maximax:} choose the alternative with the largest possible
		payoff.
		\item \textbf{Maximin:} choose the alternative with the largest worst-case
		payoff.
		\item \textbf{Minimax Regret:} choose the alternative whose maximum
		opportunity loss is the smallest.
	\end{enumerate}
	
	Because the payoffs in this section are represented symbolically, the final
	choice depends on the numerical values assigned to the expressions
	\(P(A_i,S_j)\).

	\subsection{General \(2\times 2\) Problem}
	\label{sec:general-problem-2x2}
	
	A decision maker must choose between two alternatives, \(A_1\) and \(A_2\).
	Two states of nature, \(S_1\) and \(S_2\), may occur. The payoff associated
	with each alternative--state combination is represented by \(P(A_i,S_j)\),
	where \(i\) identifies the alternative and \(j\) identifies the state of
	nature.
	
	\begin{center}
		\textit{General payoff table} \\
		\begin{tabular}{l c c}
			\toprule
			State of nature & Alternative \(A_1\) & Alternative \(A_2\) \\
			\midrule
			State \(S_1\) & \(P(A_1,S_1)\) & \(P(A_2,S_1)\) \\
			State \(S_2\) & \(P(A_1,S_2)\) & \(P(A_2,S_2)\) \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	Use Maximax, Maximin, and Minimax Regret to determine the decision rules for
	this general payoff table.

	\subsubsection*{C4 - Applies the Maximin and Maximax criteria}
	\phantomsection
	\label{subsec:general-problem-2x2-c4}
	
	\textbf{Step 1: Find the largest payoff for each alternative.}
	
	For \(A_1\), compare its payoffs under \(S_1\) and \(S_2\):
	\[
	M_1=\max\{P(A_1,S_1),P(A_1,S_2)\}.
	\]
	
	For \(A_2\), compare its payoffs under \(S_1\) and \(S_2\):
	\[
	M_2=\max\{P(A_2,S_1),P(A_2,S_2)\}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		M_1&=\max\{P(A_1,S_1),P(A_1,S_2)\},\\
		M_2&=\max\{P(A_2,S_1),P(A_2,S_2)\}.
	\end{aligned}
	\]
	
	\textbf{Step 2: Apply the Maximax criterion.}
	
	Compare the largest payoffs \(M_1\) and \(M_2\):
	\[
	M^*=\max\{M_1,M_2\}.
	\]
	
	Equivalently,
	\[
	M^*
	=
	\max\left\{
	\max\{P(A_1,S_1),P(A_1,S_2)\},
	\max\{P(A_2,S_1),P(A_2,S_2)\}
	\right\}.
	\]
	
	The Maximax decision is
	\[
	A_{\text{Maximax}}
	=
	\operatorname*{arg\,max}_{i\in\{1,2\}}
	\left\{
	\max_{j\in\{1,2\}}P(A_i,S_j)
	\right\}.
	\]
	
	Thus, choose \(A_1\) if \(M_1>M_2\), choose \(A_2\) if \(M_2>M_1\), and
	report a tie if \(M_1=M_2\).
	
	\textbf{Step 3: Find the smallest payoff for each alternative.}
	
	For \(A_1\),
	\[
	m_1=\min\{P(A_1,S_1),P(A_1,S_2)\}.
	\]
	
	For \(A_2\),
	\[
	m_2=\min\{P(A_2,S_1),P(A_2,S_2)\}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		m_1&=\min\{P(A_1,S_1),P(A_1,S_2)\},\\
		m_2&=\min\{P(A_2,S_1),P(A_2,S_2)\}.
	\end{aligned}
	\]
	
	\textbf{Step 4: Apply the Maximin criterion.}
	
	Compare the smallest payoffs \(m_1\) and \(m_2\):
	\[
	m^*=\max\{m_1,m_2\}.
	\]
	
	Equivalently,
	\[
	m^*
	=
	\max\left\{
	\min\{P(A_1,S_1),P(A_1,S_2)\},
	\min\{P(A_2,S_1),P(A_2,S_2)\}
	\right\}.
	\]
	
	The Maximin decision is
	\[
	A_{\text{Maximin}}
	=
	\operatorname*{arg\,max}_{i\in\{1,2\}}
	\left\{
	\min_{j\in\{1,2\}}P(A_i,S_j)
	\right\}.
	\]
	
	Thus, choose \(A_1\) if \(m_1>m_2\), choose \(A_2\) if \(m_2>m_1\), and
	report a tie if \(m_1=m_2\).
	
	
	\subsubsection*{C5 - Constructs opportunity loss tables and applies the Minimax regret criterion}
	\phantomsection
	\label{subsec:general-problem-2x2-c5}
	
	\textbf{Step 1: Find the best payoff in each state of nature.}
	
	Under \(S_1\), compare the payoffs of \(A_1\) and \(A_2\):
	\[
	b_1=\max\{P(A_1,S_1),P(A_2,S_1)\}.
	\]
	
	Under \(S_2\), compare the payoffs of \(A_1\) and \(A_2\):
	\[
	b_2=\max\{P(A_1,S_2),P(A_2,S_2)\}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		b_1&=\max\{P(A_1,S_1),P(A_2,S_1)\},\\
		b_2&=\max\{P(A_1,S_2),P(A_2,S_2)\}.
	\end{aligned}
	\]
	
	\textbf{Step 2: Calculate the regret for every alternative--state
		combination.}
	
	The regret associated with alternative \(A_i\) under state \(S_j\) is
	\[
	r_{ij}=b_j-P(A_i,S_j).
	\]
	
	Consequently,
	\[
	\begin{aligned}
		r_{11}&=b_1-P(A_1,S_1),&
		r_{12}&=b_2-P(A_1,S_2),\\
		r_{21}&=b_1-P(A_2,S_1),&
		r_{22}&=b_2-P(A_2,S_2).
	\end{aligned}
	\]
	
	The general regret table is therefore
	
	\begin{center}
		\begin{tabular}{l c c}
			\toprule
			Alternative & State \(S_1\) & State \(S_2\) \\
			\midrule
			\(A_1\)
			& \(b_1-P(A_1,S_1)\)
			& \(b_2-P(A_1,S_2)\) \\
			\(A_2\)
			& \(b_1-P(A_2,S_1)\)
			& \(b_2-P(A_2,S_2)\) \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	\textbf{Step 3: Find the maximum regret for each alternative.}
	
	For \(A_1\),
	\[
	R_1=
	\max\{
	b_1-P(A_1,S_1),
	b_2-P(A_1,S_2)
	\}.
	\]
	
	For \(A_2\),
	\[
	R_2=
	\max\{
	b_1-P(A_2,S_1),
	b_2-P(A_2,S_2)
	\}.
	\]
	
	Thus,
	\[
	\begin{aligned}
		R_1&=\max\{r_{11},r_{12}\},\\
		R_2&=\max\{r_{21},r_{22}\}.
	\end{aligned}
	\]
	
	\textbf{Step 4: Apply the Minimax Regret criterion.}
	
	Compare the maximum regrets \(R_1\) and \(R_2\):
	\[
	R^*=\min\{R_1,R_2\}.
	\]
	
	The Minimax Regret decision is
	\[
	A_{\text{Minimax Regret}}
	=
	\operatorname*{arg\,min}_{i\in\{1,2\}}
	\left\{
	\max_{j\in\{1,2\}}
	\left[b_j-P(A_i,S_j)\right]
	\right\}.
	\]
	
	Thus, choose \(A_1\) if \(R_1<R_2\), choose \(A_2\) if \(R_2<R_1\), and
	report a tie if \(R_1=R_2\).

	\subsubsection*{C4 - Applies the Maximin and Maximax criteria}
	\phantomsection
	\label{subsec:general-problem-4x3-c4}
	
	\textbf{Step 1: Find the largest payoff for each alternative.}
	
	For the three alternatives,
	\[
	\begin{aligned}
		M_1&=\max\{
		P(A_1,S_1),P(A_1,S_2),P(A_1,S_3),P(A_1,S_4)
		\},\\
		M_2&=\max\{
		P(A_2,S_1),P(A_2,S_2),P(A_2,S_3),P(A_2,S_4)
		\},\\
		M_3&=\max\{
		P(A_3,S_1),P(A_3,S_2),P(A_3,S_3),P(A_3,S_4)
		\}.
	\end{aligned}
	\]
	
	\textbf{Step 2: Apply the Maximax criterion.}
	
	Compare the three largest payoffs:
	\[
	M^*=\max\{M_1,M_2,M_3\}.
	\]
	
	Equivalently,
	\[
	\begin{aligned}
		M^*=\max\{&
		\max\{
		P(A_1,S_1),P(A_1,S_2),P(A_1,S_3),P(A_1,S_4)
		\},\\
		&
		\max\{
		P(A_2,S_1),P(A_2,S_2),P(A_2,S_3),P(A_2,S_4)
		\},\\
		&
		\max\{
		P(A_3,S_1),P(A_3,S_2),P(A_3,S_3),P(A_3,S_4)
		\}
		\}.
	\end{aligned}
	\]
	
	The Maximax decision is
	\[
	A_{\text{Maximax}}
	=
	\operatorname*{arg\,max}_{i\in\{1,2,3\}}
	\left\{
	\max_{j\in\{1,2,3,4\}}P(A_i,S_j)
	\right\}.
	\]
	
	Choose the alternative whose value \(M_i\) is the largest. If two or more
	alternatives have the same largest value, the Maximax criterion produces a
	tie.
	
	\textbf{Step 3: Find the smallest payoff for each alternative.}
	
	For the three alternatives,
	\[
	\begin{aligned}
		m_1&=\min\{
		P(A_1,S_1),P(A_1,S_2),P(A_1,S_3),P(A_1,S_4)
		\},\\
		m_2&=\min\{
		P(A_2,S_1),P(A_2,S_2),P(A_2,S_3),P(A_2,S_4)
		\},\\
		m_3&=\min\{
		P(A_3,S_1),P(A_3,S_2),P(A_3,S_3),P(A_3,S_4)
		\}.
	\end{aligned}
	\]
	
	\textbf{Step 4: Apply the Maximin criterion.}
	
	Compare the three smallest payoffs:
	\[
	m^*=\max\{m_1,m_2,m_3\}.
	\]
	
	Equivalently,
	\[
	\begin{aligned}
		m^*=\max\{&
		\min\{
		P(A_1,S_1),P(A_1,S_2),P(A_1,S_3),P(A_1,S_4)
		\},\\
		&
		\min\{
		P(A_2,S_1),P(A_2,S_2),P(A_2,S_3),P(A_2,S_4)
		\},\\
		&
		\min\{
		P(A_3,S_1),P(A_3,S_2),P(A_3,S_3),P(A_3,S_4)
		\}
		\}.
	\end{aligned}
	\]
	
	The Maximin decision is
	\[
	A_{\text{Maximin}}
	=
	\operatorname*{arg\,max}_{i\in\{1,2,3\}}
	\left\{
	\min_{j\in\{1,2,3,4\}}P(A_i,S_j)
	\right\}.
	\]
	
	Choose the alternative whose value \(m_i\) is the largest. If two or more
	alternatives have the same largest value, the Maximin criterion produces a
	tie.
	
	
	\subsubsection*{C5 - Constructs opportunity loss tables and applies the Minimax regret criterion}
	\phantomsection
	\label{subsec:general-problem-4x3-c5}
	
	\textbf{Step 1: Find the best payoff in each state of nature.}
	
	For the four states,
	\[
	\begin{aligned}
		b_1&=\max\{P(A_1,S_1),P(A_2,S_1),P(A_3,S_1)\},\\
		b_2&=\max\{P(A_1,S_2),P(A_2,S_2),P(A_3,S_2)\},\\

	\textbf{Step 2: Calculate the regret for every alternative--state
		combination.}

	r_{ij}=b_j-P(A_i,S_j).

		\begin{tabular}{l c c c c}

			Alternative
			& State \(S_1\)
			& State \(S_2\)
			& State \(S_3\)
			& State \(S_4\) \\
			\midrule
			\(A_1\)
			& \(b_1-P(A_1,S_1)\)

			& \(b_1-P(A_3,S_1)\)
			& \(b_2-P(A_3,S_2)\)
			& \(b_3-P(A_3,S_3)\)
			& \(b_4-P(A_3,S_4)\) \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	\textbf{Step 3: Find the maximum regret for each alternative.}
	
	For \(A_1\),
	\[
	R_1
	=
	\max\{
	b_1-P(A_1,S_1),
	b_2-P(A_1,S_2),
	b_3-P(A_1,S_3),
	b_4-P(A_1,S_4)
	\}.
	\]
	
	For \(A_2\),
	\[
	R_2
	=
	\max\{
	b_1-P(A_2,S_1),
	b_2-P(A_2,S_2),
	b_3-P(A_2,S_3),
	b_4-P(A_2,S_4)
	\}.
	\]
	
	For \(A_3\),
	\[
	R_3
	=
	\max\{
	b_1-P(A_3,S_1),
	b_2-P(A_3,S_2),
	b_3-P(A_3,S_3),
	b_4-P(A_3,S_4)
	\}.
	\]
	
	Equivalently,
	\[
	\begin{aligned}
		R_1&=\max\{r_{11},r_{12},r_{13},r_{14}\},\\
		R_2&=\max\{r_{21},r_{22},r_{23},r_{24}\},\\
		R_3&=\max\{r_{31},r_{32},r_{33},r_{34}\}.
	\end{aligned}
	\]
	
	\textbf{Step 4: Apply the Minimax Regret criterion.}
	
	Compare the maximum regrets:
	\[
	R^*=\min\{R_1,R_2,R_3\}.
	\]
	
	The Minimax Regret decision is
	\[
	A_{\text{Minimax Regret}}
	=
	\operatorname*{arg\,min}_{i\in\{1,2,3\}}
	\left\{
	\max_{j\in\{1,2,3,4\}}
	\left[b_j-P(A_i,S_j)\right]
	\right\}.
	\]
	
	Choose the alternative whose value \(R_i\) is the smallest. If two or more
	alternatives have the same smallest value, the Minimax Regret criterion
	produces a tie.

	\subsubsection*{C4 - Applies the Maximin and Maximax criteria}
	\phantomsection
	\label{subsec:food-truck-location-c4}
	
	\[
	\begin{aligned}
		\max(\text{Location A}) &= \max\{100,0\}=100,\\
		\max(\text{Location B}) &= \max\{60,50\}=60.
	\end{aligned}
	\]
	\[
	\max\{\max(\text{Location A}),\max(\text{Location B})\}
	=\max\{100,60\}
	=100.
	\]
	The Maximax choice is Location A because it has the highest possible payoff.
	
	\[
	\begin{aligned}
		\min(\text{Location A}) &= \min\{100,0\}=0,\\
		\min(\text{Location B}) &= \min\{60,50\}=50.
	\end{aligned}
	\]
	\[
	\max\{\min(\text{Location A}),\min(\text{Location B})\}
	=\max\{0,50\}
