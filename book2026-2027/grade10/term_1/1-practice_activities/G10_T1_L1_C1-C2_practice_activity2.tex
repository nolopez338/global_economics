\begin{multicols}{2}
	\begin{enumerate}[label=\arabic*.]
	 \item \textbf{Alternatives}
	 \begin{enumerate}[label=\roman*)]
	  \item Fruit cups
	  \item Baked snacks
	 \end{enumerate}
	 \item \textbf{States of nature}
	 \begin{enumerate}[label=\roman*)]
	  \item Busy market
	  \item Quiet market
	 \end{enumerate}
	\end{enumerate}
 \end{multicols}
 \textbf{Numerical information}
 \begin{enumerate}[label=\roman*)]

 \end{enumerate}

\[
 \text{Payoff}=\text{sales revenue}-\text{applicable cost or costs}.
