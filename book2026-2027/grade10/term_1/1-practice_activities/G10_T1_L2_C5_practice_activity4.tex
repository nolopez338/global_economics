	% ============================================================
	% INTRODUCTION
	% ============================================================
	\section{Minimax Regret Criterion}
	
	When probabilities are not available, a decision-maker can compare
	alternatives by considering the opportunity loss, or regret, associated
	with each possible state of nature. Regret measures the payoff that could have been obtained by choosing the
	best alternative for the state of nature that actually occurs, minus the
	payoff obtained from the selected alternative.
	
	For a payoff table with alternatives in rows and states of nature in
	columns, let
	$
	P_{ay}(A_i,S_j)
	$
	represent the payoff obtained when alternative $A_i$ is selected and
	state of nature $S_j$ occurs.

	For each state of nature, first identify the best available payoff:
	\[
	\begin{aligned}
		b(S_j)
		&=
		\max_i P_{ay}(A_i,S_j)
		\\
		&=
		\max
		\left\{
		P_{ay}(A_1,S_j),
		P_{ay}(A_2,S_j),
		\ldots,
		P_{ay}(A_m,S_j)
		\right\}.
	\end{aligned}
	\]
	
	The regret associated with alternative $A_i$ under state $S_j$ is then
	\[
	r(A_i,S_j)
	=
	b(S_j)-P_{ay}(A_i,S_j).
	\]
	
	For each alternative, determine its maximum possible regret:
	\[
	\begin{aligned}
		R(A_i)
		&=
		\max_j r(A_i,S_j)
		=
		\max_j
		\left\{
		b(S_j)-P_{ay}(A_i,S_j)
		\right\}
		\\
		&=
		\max
		\left\{
		r(A_i,S_1),
		r(A_i,S_2),
		\ldots,
		r(A_i,S_n)
		\right\}.
	\end{aligned}
	\]
	
	The Minimax Regret criterion selects the alternative whose maximum regret
	is as small as possible:
	
	\[
	A^*_{\mathrm{Minimax\ Regret}}
	=
	\arg\min_i R(A_i).
	\]
	
	Thus, the procedure is:
	
	\[
	\boxed{
		\begin{aligned}
			&\text{Best payoff in each state}
			\;\longrightarrow\;
			\text{Regret table}
			\;\longrightarrow\;
			\text{Maximum regret for each alternative}
			\\
			&\text{Minimum of those maximum regrets}
		\end{aligned}
	}
	\]
	
	The selected alternative minimizes the largest opportunity loss that
	could result from the decision.

	\section{General 2x2 Problem}
	
	Consider the following payoff table:
	
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lcc}
			\toprule
			Alternative & $S_1$ & $S_2$ \\
			\midrule
			$A_1$
			& $P_{ay}(A_1,S_1)$
			& $P_{ay}(A_1,S_2)$
			\\
			
			$A_2$
			& $P_{ay}(A_2,S_1)$
			& $P_{ay}(A_2,S_2)$
			\\
			\bottomrule
		\end{tabular}
	\end{center}
	
	Construct the opportunity loss table and determine the decision using
	the Minimax Regret criterion.
	
	% ============================================================
	% SOLUTION 2x2
	% ============================================================
	
	\subsection{Solution}
	
	\textbf{Step 1: Identify the best payoff in each state of nature}
	
	Under $S_1$, compare the payoffs of $A_1$ and $A_2$:
	
	\[
	b(S_1)
	=
	\max
	\left\{
	P_{ay}(A_1,S_1),
	P_{ay}(A_2,S_1)
	\right\}.
	\]
	
	Under $S_2$, compare the payoffs of $A_1$ and $A_2$:
	
	\[
	b(S_2)
	=
	\max
	\left\{
	P_{ay}(A_1,S_2),
	P_{ay}(A_2,S_2)
	\right\}.
	\]
	
	Therefore,
	
	\[
	\begin{aligned}
		b(S_1)
		&=
		\max
		\left\{
		P_{ay}(A_1,S_1),
		P_{ay}(A_2,S_1)
		\right\},
		\\
		b(S_2)
		&=
		\max
		\left\{
		P_{ay}(A_1,S_2),
		P_{ay}(A_2,S_2)
		\right\}.
	\end{aligned}
	\]
	
	\noindent
	\setlength{\columnseprule}{0.4pt}
	
	\begin{multicols}{2}
		\vspace{0pt}
		
		\textbf{Step 2: Construct the opportunity loss table}
		
		For each alternative and state of nature, subtract the payoff from
		the best payoff available under that state:
		
		\[
		r(A_i,S_j)
		=
		b(S_j)-P_{ay}(A_i,S_j).
		\]
		
		Thus,
		
		\[
		r(A_1,S_1)
		=
		b(S_1)-P_{ay}(A_1,S_1),
		\]
		
		\[
		r(A_1,S_2)
		=
		b(S_2)-P_{ay}(A_1,S_2),
		\]
		
		\[
		r(A_2,S_1)
		=
		b(S_1)-P_{ay}(A_2,S_1),
		\]
		
		\[
		r(A_2,S_2)
		=
		b(S_2)-P_{ay}(A_2,S_2).
		\]
		
		\columnbreak
		
		\textbf{Step 3: Find the maximum regret for each alternative}
		
		For $A_1$,
		
		\[
		R(A_1)
		=
		\max
		\left\{
		r(A_1,S_1),r(A_1,S_2)
		\right\}.
		\]
		
		For $A_2$,
		
		\[
		R(A_2)
		=
		\max
		\left\{
		r(A_2,S_1),r(A_2,S_2)
		\right\}.
		\]
		
		Therefore,
		
		\[
		\begin{aligned}
			R(A_1)
			&=
			\max
			\left\{
			b(S_1)-P_{ay}(A_1,S_1),
			b(S_2)-P_{ay}(A_1,S_2)
			\right\},
			\\
			R(A_2)
			&=
			\max
			\left\{
			b(S_1)-P_{ay}(A_2,S_1),
			b(S_2)-P_{ay}(A_2,S_2)
			\right\}.
		\end{aligned}
		\]
		
	\end{multicols}
	
	\setlength{\columnseprule}{0pt}
	
	\textbf{Step 4: Apply the Minimax Regret criterion}
	
	Compare the maximum regrets:
	
	\[
	R^*
	=
	\min
	\left\{
	R(A_1),R(A_2)
	\right\}.
	\]
	
	The Minimax Regret decision is
	
	\[
	A^*_{\mathrm{Minimax\ Regret}}
	=
	\arg\min_i
	\left\{
	R(A_i)
	\right\}.
	\]
	
	Thus, choose $A_1$ if $R(A_1)<R(A_2)$, choose $A_2$ if
	$R(A_2)<R(A_1)$, and report a tie if $R(A_1)=R(A_2)$.
	
	\textbf{Opportunity loss table}
	
	\begin{center}
		\small
		\renewcommand{\arraystretch}{1.5}
		\setlength{\tabcolsep}{5pt}
		
		\begin{tabular}{lcc|c}
			\toprule
			Alternative & $S_1$ & $S_2$ & Maximum Regret \\
			\midrule
			
			$A_1$
			&
			$r(A_1,S_1)$
			&
			$r(A_1,S_2)$
			&
			$R(A_1)$
			\\
			
			$A_2$
			&
			$r(A_2,S_1)$
			&
			$r(A_2,S_2)$
			&
			$R(A_2)$
			\\
			
			\bottomrule
		\end{tabular}
	\end{center}
	
	\section{General 4x3 Problem}

	States of nature: $S_1, S_2, S_3$.

		\textbf{Step 2: Calculate the Maximax values}
		
		For each alternative, identify its highest possible payoff:
		
		\[
		M_1
		=
		\max
		\left\{
		P_{ay}(A_1,S_1),
		P_{ay}(A_1,S_2),
		P_{ay}(A_1,S_3)
		\right\},
		\]
		
		\[
		M_2
		=
		\max
		\left\{
		P_{ay}(A_2,S_1),
		P_{ay}(A_2,S_2),
		P_{ay}(A_2,S_3)
		\right\},
		\]
		
		\[
		M_3
		=
		\max
		\left\{
		P_{ay}(A_3,S_1),
		P_{ay}(A_3,S_2),
		P_{ay}(A_3,S_3)
		\right\},
		\]
		
		\[
		M_4
		=
		\max
		\left\{
		P_{ay}(A_4,S_1),
		P_{ay}(A_4,S_2),
		P_{ay}(A_4,S_3)
		\right\}.
		\]
		
		Compare the four maximum values:
		
		\[
		A^*_{\mathrm{Maximax}}
		=
		\arg\max_i M_i.
		\]
		
		Thus, choose the alternative with the highest possible payoff.
		
		\columnbreak
		
		\textbf{Step 3: Calculate the Maximin values}
		
		For each alternative, identify its lowest possible payoff:
		
		\[
		m_1
		=
		\min
		\left\{
		P_{ay}(A_1,S_1),
		P_{ay}(A_1,S_2),
		P_{ay}(A_1,S_3)
		\right\},
		\]
		
		\[
		m_2
		=
		\min
		\left\{
		P_{ay}(A_2,S_1),
		P_{ay}(A_2,S_2),
		P_{ay}(A_2,S_3)
		\right\},
		\]
		
		\[
		m_3
		=
		\min
		\left\{
		P_{ay}(A_3,S_1),
		P_{ay}(A_3,S_2),
		P_{ay}(A_3,S_3)
		\right\},
		\]
		
		\[
		m_4
		=
		\min
		\left\{
		P_{ay}(A_4,S_1),
		P_{ay}(A_4,S_2),
		P_{ay}(A_4,S_3)
		\right\}.
		\]
		
		Compare the four minimum values:
		
		\[
		A^*_{\mathrm{Maximin}}
		=
		\arg\max_i m_i.
		\]
	
	Thus, choose the alternative with the highest worst-case payoff.
	
	\end{multicols}

	\section{Comparing the Two Criteria}
	
	The two criteria use the same payoff table but emphasize different
	aspects of the decision.
	\begin{center}
		\renewcommand{\arraystretch}{1.5}
		\begin{tabular}{p{0.20\textwidth}p{0.32\textwidth}p{0.32\textwidth}}
			\toprule
			& \textbf{Maximax} & \textbf{Maximin} \\
			\midrule
			
			Focus
			&
			Best possible payoff
			&
			Worst possible payoff
			\\
			
			Decision rule
			&
			Choose the alternative with the highest possible payoff
			&
			Choose the alternative with the highest worst-case payoff
			\\
			
			Attitude
			&
			Optimistic
			&
			Conservative
			\\
			
			Risk
			&
			Accepts greater uncertainty in pursuit of a high payoff
			&
			Places greater emphasis on protection against poor outcomes
			\\
			
			Use when
			&
			The decision-maker focuses on favorable possibilities
			&
			The decision-maker wants protection against unfavorable possibilities
			\\
			
			\bottomrule
		\end{tabular}
	\end{center}
	
	A decision-maker using \textbf{Maximax} asks:
	\[
	\text{``What is the best that could happen?''}
	\]
	A decision-maker using \textbf{Maximin} asks:
	\[
	\text{``What is the worst that could happen, and which alternative
		handles it best?''}
	\]
	
	\section{Contextualized 2x2 Problem - Coffee shop}
	
	A small coffee shop is considering whether to introduce a new dessert line or keep its current menu. The manager knows that the level of customer traffic during the promotional weekend is uncertain and has no reliable information about the probability of high or low traffic.

			\bottomrule
		\end{tabular}

		M_1

		\max\{42,8\}

		42,

		M_2

		\[
		m_2
		=
		\min\{28,20\}

		20.
		\]
		
		Compare the two minimum values:
		
		\[
		A^*_{\mathrm{Maximin}}
		=
		\arg\max_i m_i

		A_2.

	\section{Contextualized 4x3 Problem - Technology company}
	
	A technology company is deciding which product strategy to adopt for the launch of a new device. The manager knows that market conditions are uncertain and has no reliable information about the probability of each possible level of demand. The company is considering four possible strategies: launching a premium version, launching a standard version, launching a budget version, or delaying the launch.
	
	The level of market demand could be high, medium, or low. The estimated profit for each strategy under each possible market condition is shown in the table below.
	
	What would a perfectly logical and optimistic decision-maker choose? What would a perfectly logical and pessimistic decision-maker choose?
	
	\begin{center}
		\renewcommand{\arraystretch}{1.3}
		\begin{tabular}{lccc}
			\toprule
			Alternative & High demand ($S_1$) & Medium demand ($S_2$) & Low demand ($S_3$) \\
			\midrule
			Premium version ($A_1$) & 80 & 35 & -20 \\
			Standard version ($A_2$) & 60 & 42 & 18 \\
			Budget version ($A_3$) & 40 & 32 & 25 \\
			Delay launch ($A_4$) & 15 & 15 & 15 \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	All payoffs are expressed in thousands of dollars.
	
	Determine the decision using both the Maximax and Maximin criteria.
	
	% ============================================================
	% SOLUTION 4x3
	% ============================================================
	
	\subsection{Solution}
	
	\textbf{Step 1: Identify the alternatives and states of nature}
	
	Alternatives: $A_1$ = Premium version, $A_2$ = Standard version,
	$A_3$ = Budget version, $A_4$ = Delay launch
	
	\qquad
	
	States of nature: $S_1$ = High demand, $S_2$ = Medium demand,
	$S_3$ = Low demand.
	
	\noindent
	\setlength{\columnseprule}{0.4pt}
	
	\begin{multicols}{2}
		\vspace{0pt}
		
		\textbf{Step 2: Calculate the Maximax values}
		
		For each alternative, identify its highest possible payoff:
		
		\[
		M_1
		=
		\max\{80,35,-20\}
		=
		80,
		\]
		
		\[
		M_2
		=
		\max\{60,42,18\}
		=
		60,
		\]
		
		\[
		M_3
		=
		\max\{40,32,25\}
		=
		40,
		\]
		
		\[
		M_4
		=
		\max\{15,15,15\}
		=
		15.
		\]
		
		Compare the four maximum values:
		
		\[
		A^*_{\mathrm{Maximax}}
		=
		\arg\max_i M_i
		=
		A_1.
		\]
		
		Thus, the Maximax criterion recommends launching the premium version because it offers the highest possible payoff.
		
		\columnbreak
		
		\textbf{Step 3: Calculate the Maximin values}
		
		For each alternative, identify its lowest possible payoff:
		
		\[
		m_1
		=
		\min\{80,35,-20\}
		=
		-20,
		\]
		
		\[
		m_2
		=
		\min\{60,42,18\}
		=
		18,
		\]
		
		\[
		m_3
		=
		\min\{40,32,25\}
		=
		25,
		\]
		
		\[
		m_4
		=
		\min\{15,15,15\}
		=
		15.
		\]
		
		Compare the four minimum values:
		
		\[
		A^*_{\mathrm{Maximin}}
		=
		\arg\max_i m_i
		=
		A_3.
		\]
		
		Thus, the Maximin criterion recommends launching the budget version because it provides the highest worst-case payoff.
		
	\end{multicols}
