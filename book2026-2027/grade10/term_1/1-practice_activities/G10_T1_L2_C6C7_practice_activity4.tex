\paragraph{Evaluated criteria.}
    \begin{enumerate}[label=\textbf{C\arabic*.}, start=6]
        \item Calculates Expected Monetary Value using stated or estimated probabilities and applies the Bayes decision criterion to select the alternative with the highest expected payoff.
        \item Uses subjective probabilities and sensitivity analysis to examine how changes in probability assumptions affect the preferred decision.
    \end{enumerate}

\subsection{Expected Value Comparison of Marketing Options}
\label{sec:minimal-marketing-options}

\[
    EV(A)=12p+12,\quad EV(B)=2p+16,\quad 10p-4>0.
\]
A if $p>0.4$; B if $p<0.4$; indifferent at $p=0.4$.

\subsection{Expected Value Comparison of Production Plans}
\label{sec:minimal-production-plans}

\[
    EV(A)=28p+4,\quad EV(B)=6p+14,\quad EV(C)=2p+10.
\]
A if $p>\frac{5}{11}$; B if $p<\frac{5}{11}$; A and B tie at $p=\frac{5}{11}$; C is never optimal.
