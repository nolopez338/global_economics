		\section{Algebraic Problem Solving}
		
		For each problem, define a variable, construct an equation or inequality, and solve it algebraically.
		
		\begin{enumerate}
			\item
			A number increased by $7$ is equal to $25$. Find the number.
			
			Let $x$ be the number.
			
			\[
			x+7=25 \longrightarrow x=18
			\]
			
			Therefore, the number is $18$.
			
			\item
			Three times a number decreased by $8$ is equal to $19$. Find the number.
			
			Let $x$ be the number.
			
			\[
			3x-8=19 \longrightarrow 3x=27 \longrightarrow x=9
			\]
			
			Therefore, the number is $9$.
			
			\item
			The sum of two consecutive integers is $57$. Find the two integers.
			
			Let $x$ be the first integer. The next integer is $x+1$.
			
			\[
			x+(x+1)=57
			\longrightarrow 2x+1=57
			\longrightarrow 2x=56
			\longrightarrow x=28
			\]
			
			The two integers are $28$ and $29$.
			
			\item
			A rectangle has a perimeter of $54$ cm. Its length is $3$ cm greater than its width. Find its dimensions.
			
			Let $x$ be the width. Then the length is $x+3$.
			
			\[
			2x+2(x+3)=54
			\longrightarrow 4x+6=54
			\longrightarrow 4x=48
			\longrightarrow x=12
			\]
			
			The width is $12$ cm and the length is
			\[
			12+3=15\text{ cm}.
			\]
			
			Therefore, the dimensions are $12$ cm by $15$ cm.
			
			\item
			A taxi charges a fixed fee of \$5 plus \$2.50 per kilometre.
			A passenger pays \$32.50. How many kilometres did the passenger travel?
			
			Let $x$ be the number of kilometres travelled.
			
			\[
			5+2.50x=32.50
			\longrightarrow 2.50x=27.50
			\longrightarrow x=11
			\]
			
			Therefore, the passenger travelled $11$ km.
			\end{enumerate}

	\section{Hard Challenge}

	\begin{enumerate}
		
		\item
		Determine all values of $a$ for which the equation
		$(a-1)x+2a=3x+5$
		has a solution satisfying $x>1$.
		
		\textbf{Solution.}\vspace{-3mm}
		\[
		(a-1)x+2a=3x+5
		\quad\longrightarrow\quad
		(a-4)x=5-2a
		\quad\longrightarrow\quad
		x=\frac{5-2a}{a-4}.
		\]
		For $x>1$, \vspace*{-3mm}
		\[
		\frac{5-2a}{a-4}>1
		\quad\longrightarrow\quad
		\frac{5-2a-(a-4)}{a-4}>0
		\quad\longrightarrow\quad
		\frac{9-3a}{a-4}>0
		\quad\longrightarrow\quad
		\frac{3-a}{a-4}>0.
		\]
		The critical values are $a=3$ and $a=4$. Since $a=4$ is excluded,
		we analyze the signs: \vspace{-2mm}
		\[
		\begin{array}{c|ccc}
			a & (-\infty,3) & (3,4) & (4,\infty)\\
			\hline
			3-a & + & - & -\\
			a-4 & - & - & +\\
			\hline
			\dfrac{3-a}{a-4} & - & + & -
		\end{array}
		\qquad \text{Therefore,}\qquad
		\frac{3-a}{a-4}>0
		\quad\longrightarrow\quad
		\boxed{3<a<4}.
		\]
		
		\item
		Find all positive integers $x$ and $y$ satisfying
		$x+y=20$ and $x$ divides $y$.
		
		\textbf{Solution.}
		
		Since $x\mid y$, $y$ is a positive integer multiple of $x$. Therefore,
		there exists $n\in\mathbb Z^+$ such that
		$
		y=nx.
		$
		Substituting $y=nx$ into $x+y=20$ gives
		\[
		x+y=20
		\quad\longrightarrow\quad
		x+nx=20
		\quad\longrightarrow\quad
		x(n+1)=20.
		\]
		
		Since $x$ and $n+1$ are positive integers and their product is $20$,
		both $x$ and $n+1$ must be positive divisors of $20$.
		The positive factor pairs of $20$ are
		\[
		1\cdot20,\qquad
		2\cdot10,\qquad
		4\cdot5,\qquad
		5\cdot4,\qquad
		10\cdot2,\qquad
		20\cdot1.
		\quad\longrightarrow\quad
		n+1\in\{2,4,5,10,20\},
		\]
		\begin{minipage}[t]{0.38\textwidth}
			Then,
			\[
			n\in\{1,3,4,9,19\}.
			\]
			
			For each value of $n$, we use
			\[
			x=\frac{20}{n+1}
			\qquad\text{and}\qquad
			y=nx.
			\]
		\end{minipage}
		\hfill
		\begin{minipage}[t]{0.58\textwidth}
			\vspace{-2mm}
			\[
			\begin{array}{c|c|c|c}
				\hline
				n & n+1 & x=\dfrac{20}{n+1} & y=nx\\
				\hline
				1  & 2  & 10 & 10\\[-4pt]
				3  & 4  & 5  & 15\\[-4pt]
				4  & 5  & 4  & 16\\[-4pt]
				9  & 10 & 2  & 18\\[-4pt]
				19 & 20 & 1  & 19
			\end{array}
			\]
		\end{minipage}
		
		Therefore,
		\[
		\boxed{
			(x,y)\in
			\{(10,10),(5,15),(4,16),(2,18),(1,19)\}
		}.
		\]

		\item
		Find all positive integers $x$ and $y$ satisfying $
		xy-2x-3y=10.
		$
		
		\textbf{Solution.}
		\[
		xy-2x-3y=10
		\quad\longrightarrow\quad
		xy-2x-3y+6=16
		\quad\longrightarrow\quad
		(x-3)(y-2)=16.
		\]
		
		The positive factor pairs of $16$ are
		\[
		(x-3,y-2)
		\quad\longrightarrow\quad
		(1,16),(2,8),(4,4),(8,2),(16,1).
		\]
		
		Therefore,
		\[
		\boxed{
			(x,y)\in
			\{(4,18),(5,10),(7,6),(11,4),(19,3)\}
		}.
		\]
		
		
		\item
		Solve over the real numbers
		\[
		\frac{x+1}{x-1}+\frac{x-1}{x+1}=3.
		\]
		
		\textbf{Solution.}
		
		First,
		$
		x\neq\pm1.
		$
		Then
		\[
		\frac{x+1}{x-1}+\frac{x-1}{x+1}=3
		\quad\longrightarrow\quad
		\frac{(x+1)(x+1)+(x-1)(x-1)}
		{(x-1)(x+1)}=3
		\quad\longrightarrow\quad
		\frac{(x+1)^2+(x-1)^2}{x^2-1}=3
		\]
		\[
		\quad\longrightarrow\quad
		\frac{x^2+2x+1+x^2-2x+1}{x^2-1}=3
		\quad\longrightarrow\quad
		\frac{2x^2+2}{x^2-1}=3
		\quad\longrightarrow\quad
		2x^2+2=3(x^2-1)
		\]
		
		\[
		\quad\longrightarrow\quad
		2x^2+2=3x^2-3
		\quad\longrightarrow\quad
		x^2=5
		\quad\longrightarrow\quad
		\boxed{x=\pm\sqrt5}.
		\]

		
		\item
		Which sets of three consecutive integers have a sum equal to their product? Prove that these are the only such sets.
		
		\textbf{Solution.}
		
		Let the three consecutive integers be
		$
		(n-1,\quad n,\quad n+1).
		$
		Then
		\[
		\quad\longrightarrow\quad
		3n=n(n-1)(n+1)
		\quad\longrightarrow\quad
		3n=n(n^2-1)
		\quad\longrightarrow\quad
		3n=n^3-n
		\quad\longrightarrow\quad
		n^3-4n=0
		\]
		\[
		\quad\longrightarrow\quad
		n(n^2-4)=0
		\quad\longrightarrow\quad
		n(n-2)(n+2)=0
		\quad\longrightarrow\quad
		\boxed{n=0,\,-2,\,2}.
		\]
		
		Therefore, the corresponding triples are
		
		\begin{minipage}[t]{0.48\textwidth}
			\[
			n=-2\longrightarrow(-3,-2,-1),
			\]
			\[
			n=0\longrightarrow(-1,0,1),
			\]
			\[
			n=2\longrightarrow(1,2,3).
			\]
		\end{minipage}
		\hfill
		\begin{minipage}[t]{0.48\textwidth}
			\[
			(n-1,\quad n,\quad n+1).
			\]
			\[
			\boxed{(-3,-2,-1),\ (-1,0,1),\ (1,2,3)}
			\]
		\end{minipage}
		
		\vspace{5mm}
		\item
		Determine all values of $k$ for which
		$
		|x-1|+|x+1|=k
		$
		has exactly two solutions.
		
		\begin{minipage}[t]{0.55\textwidth}
			
			
			\textbf{Solution.}
			
			The expressions change form at $x=-1$ and $x=1$:
			\[
			|x-1|+|x+1|=
			\begin{cases}
				(1-x)+(-x-1)=-2x, & x<-1,\\
				(1-x)+(x+1)=2, & -1\leq x\leq1,\\
				(x-1)+(x+1)=2x, & x>1.
			\end{cases}
			\]
			
		Thus,
			\[
			|x-1|+|x+1|\geq2.
			\]
			

		\end{minipage}
		\hfill
		\begin{minipage}[t]{0.36\textwidth}
					
						If $k<2$,
			\[
			\text{no solutions}.
			\]
			
			If $k=2$,
			\[
			x\in[-1,1]
			\quad\longrightarrow\quad
			\text{infinitely many solutions}.
			\]
			
			If $k>2$, the middle interval gives no solutions, while
			\[
			-2x=k
			\quad\longrightarrow\quad
			x=-\frac{k}{2},
			\]\[
			\qquad
			2x=k
			\quad\longrightarrow\quad
			x=\frac{k}{2}.
			\]
			
			Since $k>2$, these satisfy $x<-1$ and $x>1$, respectively.
			
			Therefore, exactly two solutions occur when
			\[
			\boxed{k>2}.
			\]
		\end{minipage}
		
	\end{enumerate}
