 {
  \prop_if_exist:cTF {g_hl_#1_#2_prop}
   {\msg_error:nnn {highlights}{duplicate}{#1/#2}}
   {
    \prop_clear:N \l_hl_record_prop
    \keys_set:nn {highlights/record}{#3}
    \prop_put:Nnn \l_hl_record_prop {id}{#2}
    \prop_new:c {g_hl_#1_#2_prop}
    \prop_gset_eq:cN {g_hl_#1_#2_prop} \l_hl_record_prop
    \seq_gput_right:cn {g_hl_#1s_seq}{#2}
   }
 }
