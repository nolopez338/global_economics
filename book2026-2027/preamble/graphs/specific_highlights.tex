% Keys include grade, term and every applicable criterion, e.g. G11-T3-C1-C2-C3.
% Add a lesson or variant suffix only to distinguish legends for the same key.
\newcommand{\PracticeHighlightLegend}[1]{%
  \ifcsname practicehighlightlegend@#1\endcsname
    \csname practicehighlightlegend@#1\endcsname
  \else
    \PackageError{specific_highlights}{Unknown practice highlight legend key '#1'}%
      {Use a key registered in graphs/specific_highlights.tex.}%
  \fi}

% Legends stay readable in monochrome: names and units carry the meaning.
\newcommand{\GTenHighlightLegend}{%
  \par\medskip\noindent
  \begin{tabularx}{\linewidth}{@{}l>{\raggedright\arraybackslash}X@{}}
    \toprule
    \textbf{Highlighted item} & \textbf{What it means} \\
    \midrule
    \GTenHLOutcome{Outcome} & A possible result of the experiment. \\
    \GTenHLProbability{Probability} & The chance of an outcome or event, expressed without units. \\
    \GTenHLValue{First variable} & The first variable, its numerical values, and any units. \\
    \GTenHLSecond{Second variable} & The second variable, its numerical values, and any units. \\
    \GTenHLThird{Third variable} & The third variable, its numerical values, and any units. \\
    \GTenHLParameter{Unknown parameter} & A quantity to determine from the information given. \\
    \bottomrule
  \end{tabularx}\par\medskip}
\expandafter\let\csname practicehighlightlegend@G10-T2-C1\endcsname\GTenHighlightLegend

\expandafter\newcommand\csname practicehighlightlegend@G10-T2-C1-C2-C3-C4-C5-C6-C7-C8-C9\endcsname{%
  \par\smallskip\noindent
  {\small\textbf{Reading the markers.}
  \GTenHLOutcome{Outcome labels},
  \GTenHLValue{primary variable},
  \GTenHLSecond{second variable},
  \GTenHLThird{third variable}
  (each variable includes its values and units),
  \GTenHLProbability{unitless probabilities}, and
  \GTenHLParameter{quantities to determine}.%
  }\par\smallskip}
