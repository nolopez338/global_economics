% Representative amounts below use one illustrative interval in generic units.
% Each row connects highlighted names, complete symbols, meanings and values.
\newcommand{\CtwoLegendMeans}{%
    \CtwoMean{Population mean}, $\CtwoMean{\mu}$ & The fixed population parameter being estimated. \\
    \addlinespace
    \CtwoPoint{Point estimate}, $\CtwoPoint{\bar{x}}$ & Sample mean and interval centre; for example, $\CtwoPoint{60}$ units. \\
    \addlinespace}
\newcommand{\CtwoLegendInterval}{%
    \CtwoSE{Standard error}, $\CtwoSE{\mathrm{SE}}$ & Spread of sample means across repeated samples, or its estimate; for example, $\CtwoSE{2}$ units. Distinct from individual-observation spread. \\
    \addlinespace
    \CtwoMoE{Margin of error}, $\CtwoMoE{\mathrm{MoE}}$ & Distance from centre to an endpoint: $\CtwoMoE{\mathrm{MoE}}=\CtwoCritical{z^*}\CtwoSE{\mathrm{SE}}$. For example, $\CtwoCritical{1.96}(\CtwoSE{2})=\CtwoMoE{3.92}$ units. \\
    \addlinespace
    \CtwoEndpoint{Endpoints}, $\CtwoPoint{\bar{x}}\pm\CtwoMoE{\mathrm{MoE}}$ & Lower and upper bounds: $\CtwoPoint{60}\pm\CtwoMoE{3.92}$ gives $\CtwoEndpoint{56.08}$ and $\CtwoEndpoint{63.92}$ units. \\
    \addlinespace
    \CtwoWidth{Full width}, $2\CtwoMoE{\mathrm{MoE}}$ & Distance between endpoints, twice the margin of error; here $2(\CtwoMoE{3.92})=\CtwoWidth{7.84}$ units. \\
    \addlinespace
    \CtwoConfidence{Confidence level}, $\CtwoConfidence{C}$ & Nominal long-run coverage under the model; $\CtwoConfidence{95\%}=\CtwoConfidence{0.95}$ is the central normal area. \\
    \addlinespace
    \CtwoCritical{Standard normal} \CtwoCritical{critical value}, $\CtwoCritical{z^*}$ & Unitless multiplier selected by $\CtwoConfidence{C}$; $\CtwoCritical{1.96}$ bounds the central area $\CtwoConfidence{0.95}$ between $-\CtwoCritical{z^*}$ and $\CtwoCritical{z^*}$. \\
}
