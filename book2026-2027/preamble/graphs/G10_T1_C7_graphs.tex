\newcommand{\EMVGraph}[1]{%
	\begin{tikzpicture}
		\begin{axis}[
			width=0.85\textwidth,
			height=7cm,
			xmin=0,
			xmax=1,
			ymin=-25,
			ymax=85,
			axis lines=left,
			xlabel={Subjective probability $p$},
			ylabel={Expected Monetary Value},
			grid=both,
			legend style={
				at={(0.5,1.02)},
				anchor=north,
				legend columns=-1
			}
			]
			
			\foreach \item in {#1} {
				\expandafter\EMVPlot\item
			}
			
		\end{axis}
	\end{tikzpicture}
}

\newcommand{\EMVPlot}[4]{%
	\addplot[
	thick,
	#3,
	domain=0:1,
	samples=100
	]
	{#1*x+#2};
	\addlegendentry{#4}
}
