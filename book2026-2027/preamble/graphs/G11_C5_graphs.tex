% ------------------------------------------------------------
% Global distribution graph dimensions
% ------------------------------------------------------------
\newlength{\DistributionGraphWidth}
\newlength{\DistributionGraphHeight}

\setlength{\DistributionGraphWidth}{0.8\textwidth}
\setlength{\DistributionGraphHeight}{6cm}


% ------------------------------------------------------------
% Uniform distribution
% X ~ U(a,b)
% Includes vertical lines at mu-2sigma, mu-sigma, mu,
% mu+sigma, and mu+2sigma
% ------------------------------------------------------------
\newcommand{\UniformGraph}[2]{%
	\pgfmathsetmacro{\Mean}{(#1+#2)/2}%
	\pgfmathsetmacro{\StdDev}{(#2-#1)/sqrt(12)}%
	\pgfmathsetmacro{\MuMinusTwoSigma}{\Mean-2*\StdDev}%
	\pgfmathsetmacro{\MuMinusSigma}{\Mean-\StdDev}%
	\pgfmathsetmacro{\MuPlusSigma}{\Mean+\StdDev}%
	\pgfmathsetmacro{\MuPlusTwoSigma}{\Mean+2*\StdDev}%
	\pgfmathsetmacro{\Density}{1/(#2-#1)}%
	%
	\begin{center}
		\begin{tikzpicture}
			\begin{axis}[
				width=\DistributionGraphWidth,
				height=\DistributionGraphHeight,
				axis lines=left,
				xlabel={$x$},
				ylabel={$f(x)$},
				xmin=#1-0.5,
				xmax=#2+0.5,
				ymin=0,
				ymax={\Density*1.35},
				xtick={
					{\MuMinusTwoSigma},
					{\MuMinusSigma},
					{\Mean},
					{\MuPlusSigma},
					{\MuPlusTwoSigma}
				},
				xticklabels={
					{\shortstack{$\mu-2\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuMinusTwoSigma}$}},
					{\shortstack{$\mu-\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuMinusSigma}$}},
					{\shortstack{$\mu$\\
							$\pgfmathprintnumber[fixed,precision=2]{\Mean}$}},
					{\shortstack{$\mu+\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuPlusSigma}$}},
					{\shortstack{$\mu+2\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuPlusTwoSigma}$}}
				},
				ytick={0,\Density},
				yticklabels={
					{$0$},
					{\shortstack{$\frac{1}{b-a}$\\
							$\pgfmathprintnumber[fixed,precision=2]{\Density}$}}
				},
				tick label style={font=\small},
				grid=both,
				clip=false
				]
				
				% Uniform probability density function
				\addplot[
				very thick,
				blue
				] coordinates {
					(#1,0)
					(#1,\Density)
					(#2,\Density)
					(#2,0)
				};
				
				% Mean
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					(\Mean,0)
					(\Mean,\Density)
				};
				
				% One standard deviation below the mean
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					(\MuMinusSigma,0)
					(\MuMinusSigma,\Density)
				};
				
				% One standard deviation above the mean
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					(\MuPlusSigma,0)
					(\MuPlusSigma,\Density)
				};
				
				% Two standard deviations below the mean
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					(\MuMinusTwoSigma,0)
					(\MuMinusTwoSigma,\Density)
				};
				
				% Two standard deviations above the mean
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					(\MuPlusTwoSigma,0)
					(\MuPlusTwoSigma,\Density)
				};
				
			\end{axis}
		\end{tikzpicture}
	\end{center}%
}

% ------------------------------------------------------------
% Triangular distribution
% X ~ Tri(a,b,c)
% a = minimum, b = mode, c = maximum
% ------------------------------------------------------------
\newcommand{\TriangularGraph}[3]{%
	% Mean
	\pgfmathsetmacro{\Mean}{(#1+#2+#3)/3}%
	%
	% Standard deviation
	\pgfmathsetmacro{\StdDev}{sqrt((#1^2+#2^2+#3^2-#1*#2-#1*#3-#2*#3)/18)}%
	%
	% Mean +/- standard deviations
	\pgfmathsetmacro{\MuMinusTwoSigma}{\Mean-2*\StdDev}%
	\pgfmathsetmacro{\MuMinusSigma}{\Mean-\StdDev}%
	\pgfmathsetmacro{\MuPlusSigma}{\Mean+\StdDev}%
	\pgfmathsetmacro{\MuPlusTwoSigma}{\Mean+2*\StdDev}%
	%
	% Maximum density
	\pgfmathsetmacro{\Density}{2/(#3-#1)}%
	%
	\begin{center}
		\begin{tikzpicture}
			\begin{axis}[
				width=\DistributionGraphWidth,
				height=\DistributionGraphHeight,
				axis lines=left,
				xlabel={$x$},
				ylabel={$f(x)$},
				xmin={#1-0.5},
				xmax={#3+0.5},
				ymin=0,
				ymax={\Density*1.25},
				xtick={
					\MuMinusTwoSigma,
					\MuMinusSigma,
					\Mean,
					\MuPlusSigma,
					\MuPlusTwoSigma
				},
				xticklabels={
					{\shortstack{$\mu-2\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuMinusTwoSigma}$}},
					{\shortstack{$\mu-\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuMinusSigma}$}},
					{\shortstack{$\mu$\\
							$\pgfmathprintnumber[fixed,precision=2]{\Mean}$}},
					{\shortstack{$\mu+\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuPlusSigma}$}},
					{\shortstack{$\mu+2\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuPlusTwoSigma}$}}
				},
				ytick={
					0,
					\Density
				},
				yticklabels={
					{$0$},
					{$\frac{2}{c-a}$}
				},
				tick label style={font=\small},
				grid=both,
				clip=false
				]
				
				% Triangular density
				\addplot[
				very thick,
				blue,
				domain=#1:#2,
				samples=100
				]
				{2*(x-#1)/((#3-#1)*(#2-#1))};
				
				\addplot[
				very thick,
				blue,
				domain=#2:#3,
				samples=100
				]
				{2*(#3-x)/((#3-#1)*(#3-#2))};
				
				% Vertical dashed lines
				\addplot[
				dashed,
				gray,
				domain=0:\Density
				]
				coordinates {
					(\MuMinusTwoSigma,0)
					(\MuMinusTwoSigma,\Density)
				};
				
				\addplot[
				dashed,
				gray,
				domain=0:\Density
				]
				coordinates {
					(\MuMinusSigma,0)
					(\MuMinusSigma,\Density)
				};
				
				\addplot[
				dashed,
				gray,
				domain=0:\Density
				]
				coordinates {
					(\Mean,0)
					(\Mean,\Density)
				};
				
				\addplot[
				dashed,
				gray,
				domain=0:\Density
				]
				coordinates {
					(\MuPlusSigma,0)
					(\MuPlusSigma,\Density)
				};
				
				\addplot[
				dashed,
				gray,
				domain=0:\Density
				]
				coordinates {
					(\MuPlusTwoSigma,0)
					(\MuPlusTwoSigma,\Density)
				};
				
			\end{axis}
		\end{tikzpicture}
	\end{center}%
}

% ------------------------------------------------------------
% Exponential distribution
% X ~ Exp(lambda)
% Includes vertical lines at mu-sigma, mu, mu+sigma,
% and mu+2sigma
% ------------------------------------------------------------
\newcommand{\ExponentialGraph}[1]{%
	\pgfmathsetmacro{\Mean}{1/#1}%
	\pgfmathsetmacro{\StdDev}{1/#1}%
	\pgfmathsetmacro{\MuMinusSigma}{\Mean-\StdDev}%
	\pgfmathsetmacro{\MuPlusSigma}{\Mean+\StdDev}%
	\pgfmathsetmacro{\MuPlusTwoSigma}{\Mean+2*\StdDev}%
	%
	\begin{center}
		\begin{tikzpicture}
			\begin{axis}[
				width=\DistributionGraphWidth,
				height=\DistributionGraphHeight,
				axis lines=left,
				xlabel={$x$},
				ylabel={$f(x)$},
				xmin=0,
				xmax={5/#1},
				ymin=0,
				ymax={#1*1.15},
				xtick={
					{\MuMinusSigma},
					{\Mean},
					{\MuPlusSigma},
					{\MuPlusTwoSigma}
				},
				xticklabels={
					{\shortstack{$\mu-\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuMinusSigma}$}},
					{\shortstack{$\mu$\\
							$\pgfmathprintnumber[fixed,precision=2]{\Mean}$}},
					{\shortstack{$\mu+\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuPlusSigma}$}},
					{\shortstack{$\mu+2\sigma$\\
							$\pgfmathprintnumber[fixed,precision=2]{\MuPlusTwoSigma}$}}
				},
				ytick={0,#1},
				yticklabels={
					{$0$},
					{\shortstack{$\lambda$\\
							$\pgfmathprintnumber[fixed,precision=2]{#1}$}}
				},
				tick label style={font=\small},
				grid=both,
				clip=false
				]
				
				% Exponential probability density function
				\addplot[
				very thick,
				blue,
				domain=0:{5/#1},
				samples=100
				]
				{#1*exp(-#1*x)};
				
				% mu - sigma = 0
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					(0,0)
					(0,#1)
				};
				
				% mu = 1/lambda
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					({1/#1},0)
					({1/#1},{#1*exp(-1)})
				};
				
				% mu + sigma = 2/lambda
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					({2/#1},0)
					({2/#1},{#1*exp(-2)})
				};
				
				% mu + 2sigma = 3/lambda
				\addplot[
				dashed,
				black,
				thick
				] coordinates {
					({3/#1},0)
					({3/#1},{#1*exp(-3)})
				};
				
			\end{axis}
		\end{tikzpicture}
	\end{center}%
}

				width=\DistributionGraphWidth,
				height=\DistributionGraphHeight,
