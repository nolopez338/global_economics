	\sectionScored
	{Example Uniform Investment Plan}
	{C1-\sml{3} \hspace{2mm} C2-\sml{2}}
	{sec:example-uniform-investment-plan-c1}
	{
	A variable $V$ follows a uniform model on $[8,16]$. Use the graph and density model to answer the tasks.
	
	\textbf{Questions/Tasks.}
	\begin{enumerate}
		\CriterionTask
		{subsec:example-uniform-investment-plan-c1-a}{1}
		{Write the density function and support. \score{3}}
		
		\CriterionTask
		{subsec:example-uniform-investment-plan-c1-b}{2}
		{Compute $P(10\le V\le 14)$ and justify with area. \score{2}}
	\end{enumerate}
	
	\subsection*{C1 - Write the density function and support. \score{3}}
	\phantomsection
	\label{subsec:example-uniform-investment-plan-c1-a}
	
	\[
	f_V(v)=
	\begin{cases}
		\dfrac{1}{16-8}=\dfrac18, & 8\le v\le 16,\\
		0, & \text{otherwise.}
	\end{cases}
	\]
	
	\[
	8\le v\le 16
	\qquad \longrightarrow \qquad
	f_V(v)=\frac18
	\qquad \longrightarrow \qquad
	\int_8^{16} f_V(v)\,dv=1
	\]
	
	\vspace{-5mm}

	\subsection*{C2 - Compute \boldmath{$P(10\le V\le 14)$} and justify with area. \score{2}}
	\phantomsection
	\label{subsec:example-uniform-investment-plan-c1-b}
	
	\begin{center}
		\setlength{\tabcolsep}{3pt}
		\renewcommand{\arraystretch}{1.08}
		\begin{tabular}{@{}p{0.38\textwidth}p{0.58\textwidth}@{}}
			\centering
			\InvestmentThresholdGraph{8}{16}{1/8}{10}{14}{0.18}
			&
			\[
			\begin{aligned}
				P(10\le V\le 14)

	\sectionScored
	{Example Linear Density Model}
	{C2-\sml{5}}
	{sec:example-linear-density-model-c2}
	{
	Suppose $X$ has density $f(x)=kx$ for $0\le x\le 2$.
	
	\textbf{Questions/Tasks.}
	\begin{enumerate}
		\CriterionTask
		{subsec:example-linear-density-model-c2-a}{2}
		{Find the value of $k$. \score{3}}
		
		\CriterionTask
		{subsec:example-linear-density-model-c2-b}{2}
		{Compute $P(0.5\le X\le 1.5)$. \score{2}}
	\end{enumerate}
	
	\subsection*{C2 - Find the value of \boldmath{$k$} \score{3}}
	\phantomsection
	\label{subsec:example-linear-density-model-c2-a}
	
	\[
	\begin{aligned}
		1
		&=\int_0^2 kx\,dx\\
		&=k\left[\frac{x^2}{2}\right]_0^2\\
		&=2k
	\end{aligned}
	\quad\Rightarrow\quad
	k=\frac12.
	\]
	
	\COneGraph{0}{2}{0.5*x}{0.5}{1.5}{0,0.5,1,1.5,2}{1.2}
	
	\vspace{-5mm}
	\secInicio[sec:worked-solution-examples][sec:example-linear-density-model-c2]
	
	\subsection*{C2 - Compute \boldmath{$P(0.5\le X\le 1.5)$} \score{2}}
	\phantomsection
	\label{subsec:example-linear-density-model-c2-b}
	
	\[
	\begin{aligned}
		P(0.5\le X\le 1.5)
		&=\int_{0.5}^{1.5}\frac12 x\,dx\\
		&=\left[\frac{x^2}{4}\right]_{0.5}^{1.5}\\
		&=\frac{1.5^2-0.5^2}{4}\\
		&=0.5
	\end{aligned}
	\]
