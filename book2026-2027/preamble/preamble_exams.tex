% ----------------------------------------------------------
% Exam table and list spacing (overrides general defaults)
% ----------------------------------------------------------
\setlength{\parindent}{0pt}
\setlength{\tabcolsep}{6pt}
\renewcommand{\arraystretch}{1.15}

% Compact bullet and numbered lists.
\setlist[itemize]{topsep=2pt,itemsep=1pt,parsep=0pt,partopsep=0pt}
\setlist[enumerate]{topsep=2pt,itemsep=0.8pt,parsep=0pt,partopsep=0pt}

% ----------------------------------------------------------
% Adjustable objective/criteria columns and centered header cells
% ----------------------------------------------------------
% Shared width used by the Y/Z pair; change via \setYColumnWidth.
\newlength{\YColumnWidth}

% \setYColumnWidth{fraction}: allocate a fraction of \textwidth to Y.
% Z uses the remainder; column formulas subtract cell padding and rule widths.
\newcommand{\setYColumnWidth}[1]{%
	\setlength{\YColumnWidth}{#1\textwidth}%
}

% Default allocation: 30% objective, 70% criteria, before padding/rule deductions.
\setYColumnWidth{0.30}

% Y and Z: vertically centered, ragged-right columns for \begin{tabular}{|Y|Z|}.
\newcolumntype{Y}{%
	>{\raggedright\arraybackslash}%
	m{\dimexpr\YColumnWidth-2\tabcolsep-2\arrayrulewidth\relax}%
}

\newcolumntype{Z}{%
	>{\raggedright\arraybackslash}%
	m{\dimexpr\textwidth-\YColumnWidth-2\tabcolsep-2\arrayrulewidth\relax}%
}

% C{width}: vertically and horizontally centered column, e.g. C{2.8cm}.
\newcolumntype{C}[1]{%
	>{\centering\arraybackslash}m{#1}%
}


% ----------------------------------------------------------
% Heading and cell helpers
% ----------------------------------------------------------
% \SubsectionBox{text}: bold gray heading spanning the current \linewidth.
% Works within columns; adds spacing but no section numbering or TOC entry.
\newcommand{\SubsectionBox}[1]{%
	\vspace{0.25cm}
	\noindent\colorbox{gray!30}{%
		\parbox{\linewidth}{\textbf{#1}}%
	}\par\vspace{0.0cm}%
}

% \CellCenter{content}: centered multi-line table cell at the current \linewidth.
% Separate lines with \par; used by \ExamHeader.
\newcommand{\CellCenter}[1]{%
	\parbox{\linewidth}{\centering #1}%
}



% ----------------------------------------------------------
% Assessment header, student details, and learning objective
% ----------------------------------------------------------
% \ExamHeader{grade}{evidence number}{title}{term}: full-width three-cell header.
% Example: \ExamHeader{11th}{1}{Financial modeling}{1st}.
% Requires logo.png on the graphics search path; teacher/year are fixed below.
\newcommand{\ExamHeader}[4]{%
	\noindent
	\begin{tabularx}{\textwidth}{|C{2.8cm}|C{\dimexpr\textwidth-6cm-4\tabcolsep-4\arrayrulewidth\relax}|C{2.8cm}|}
		\hline
		\centering
		\vspace{3mm}
		\includegraphics[width=2.5cm]{logo.png}
		&
		\CellCenter{%
			\vspace{-5mm}
			\textbf{GLOBAL ECONOMICS}\par
			\textbf{Grade: #1 - Learning Evidence #2}\par
			\textbf{#3}\par
			\textbf{Teacher: Nicolás López Cuéllar}
		}
		&
		\CellCenter{%
			\textbf{#4 TERM}\par
			\textbf{2026--2027}%
		}
		\\
		\hline
	\end{tabularx}%
}


% \StudentInfo: full-width name, grade, and date blanks; no arguments.
% Place below \ExamHeader, before entering a multicolumn layout.
\newcommand{\StudentInfo}{%
	\noindent
	\textbf{Name:} \rule{9.5cm}{0.4pt}\hfill
	\textbf{Grade:} \rule{1.5cm}{0.4pt}\hfill
	\textbf{Date:} \rule{3cm}{0.4pt}%
}

% \LearningObjective[objective-width fraction]{objective text}{criteria text}.
% Full-width two-column table; optional fraction defaults to 0.30.
% Use \par between criteria; place before entering a multicolumn layout.
% Sets the shared Y/Z allocation for subsequent tables in the current scope.
\newcommand{\LearningObjective}[3][0.30]{%
	\setYColumnWidth{#1}%
	\noindent
	\begin{tabular}{|Y|Z|}
		\hline
		{\small
			\textbf{Learning objective:} #2
		}
		&
		{\footnotesize
			\textbf{Assessment criteria:}\par
			#3
		}
		\\
		\hline
	\end{tabular}%
}
