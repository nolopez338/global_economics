% Column types
\newlength{\YColumnWidth}

% Set Y as a fraction of \textwidth.
% Z automatically uses the remaining fraction.
\newcommand{\setYColumnWidth}[1]{%
	\setlength{\YColumnWidth}{#1\textwidth}%
}

% Default: Y = 30%, Z = 70%
\setYColumnWidth{0.30}

% Column types
\newcolumntype{Y}{%
	>{\raggedright\arraybackslash}%
	m{\dimexpr\YColumnWidth-2\tabcolsep-2\arrayrulewidth\relax}%
}

\newcolumntype{Z}{%
	>{\raggedright\arraybackslash}%
	m{\dimexpr\textwidth-\YColumnWidth-2\tabcolsep-2\arrayrulewidth\relax}%
}

\newcolumntype{C}[1]{%
	>{\centering\arraybackslash}m{#1}%
}
