		\heading{1. What does solving mean?}
		
		\textbf{E1:} A taxi charges a fixed fee of \$4 plus \$2 for each kilometre travelled.
		Suppose the total fare is \$18. \textbf{How many kilometres did the taxi travel?}
		
		Let $k =$ number of kilometres travelled. We can write $4+2k=18$.
		To \textbf{solve} the equation means to find the value of $k$ that makes the
		statement true: \vspace{-2mm}
		\[
		4+2(7)=18 \qquad\Rightarrow\qquad k=7
		\]
		The taxi travelled \textbf{7 km}.
		
		\smallskip
		
		\textbf{E2:} Two phone plans have different costs. Plan A charges \$10 plus \$3 per month,
		while Plan B charges \$22 plus \$2 per month. \textbf{After how many months will
			both plans have the same total cost?}
		
		Let $m =$ number of months. We can write $10+3m=22+2m$. Solving gives \vspace{-2mm}
		\[
		3m-2m=22-10 \qquad\Rightarrow\qquad m=12
		\]
		
		\vspace{-2mm}
		After \textbf{12 months}, both plans have the same total cost.
		
		\heading{2. Basic solving process}
		Equations such as $ax+b=c$ and $ax+b=cx+d$ can be solved by:
		\begin{enumerate}
			\item Simplifying each side (distribute and combine like terms);
			\item Collecting variable terms on one side;
			\item Collecting constants on the other side;
			\item Isolating the variable; and
			\item Checking by substitution in the \emph{original} equation.
		\end{enumerate}
		
		\textbf{Model: \quad $3x+5=20$}
		\vspace{-2mm}
		\begin{align*}
			3x+5-5&=20-5 &&\text{(subtract 5)}\\
			3x&=15\\
			\frac{3x}{3}&=\frac{15}{3} &&\text{(divide by 3)}\\
			x&=5
		\end{align*}
		Check: $3(5)+5=20$, so the left side equals the right side. Thus $x=5$.
		
		\textbf{Model: \quad $5x+4=2x+19$}
		\vspace{-2mm}
		\begin{align*}
			5x-2x+4&=2x-2x+19 &&\text{(subtract $2x$)}\\
			3x+4&=19\\
			3x+4-4&=19-4 &&\text{(subtract 4)}\\
			3x&=15\\
			\frac{3x}{3}&=\frac{15}{3} &&\text{(divide by 3)}\\
			x&=5
		\end{align*}
		Check: $5(5)+4=29$ and $2(5)+19=29$, so both sides are equal. Thus $x=5$.
		
		\heading{3. Worked examples}
		
		\textbf{A. Form:}
		\vspace{-7mm}
		\begin{align*}
			ax+b&=c \\ 
			-4x-7&=17 \\
			-4x-7+7&=17+7 &&\text{(add 7)}\\
			-4x&=24\\
			\frac{-4x}{-4}&=\frac{24}{-4} &&\text{(divide by $-4$)}\\
			x&=-6
		\end{align*}
		\vspace{-4mm}
		Check: $-4(-6)-7=17$.
		\vspace{5mm}
		
		\textbf{B. Form:}
		\vspace{-7mm}
		\begin{align*}
			ax+b&=cx+d \\
			-5x+4&=2x-17 \\
			-5x-2x+4&=2x-2x-17 &&\text{(subtract $2x$)}\\
			-7x+4&=-17\\
			-7x+4-4&=-17-4 &&\text{(subtract 4)}\\
			-7x&=-21\\
			\frac{-7x}{-7}&=\frac{-21}{-7} &&\text{(divide by $-7$)}\\
			x&=3
		\end{align*}
		\vspace{-4mm}
		Check: $-5(3)+4=-11=2(3)-17$.
		\vspace{5mm}
		
		\textbf{C. Form:}
		\vspace{-7mm}
		\begin{align*}
			a(x+b)&=c \\
			-3(x-2)&=18 \\
			-3x+6&=18 &&\text{(distribute $-3$)}\\
			-3x+6-6&=18-6 &&\text{(subtract 6)}\\
			-3x&=12\\
			\frac{-3x}{-3}&=\frac{12}{-3} &&\text{(divide by $-3$)}\\
			x&=-4
		\end{align*}
		\vspace{-4mm}
		Check: $-3(-4-2)=18$.
		\vspace{5mm}
		
		\textbf{D. Form:}
		\vspace{-7mm}
		\begin{align*}
			\frac{x}{a}+b&=c \\
			-\frac{x}{4}+3&=8 \\
			-\frac{x}{4}+3-3&=8-3 &&\text{(subtract 3)}\\
			-\frac{x}{4}&=5\\
			(-4)\left(-\frac{x}{4}\right)&=(-4)(5) &&\text{(multiply by $-4$)}\\
			x&=-20
		\end{align*}
		\vspace{-4mm}
		Check: $-\dfrac{-20}{4}+3=8$.

		\heading{4. Equations in context}
		Follow the chain
		\[
		\text{context}\longrightarrow\text{equation}\longrightarrow
		\text{solution}\longrightarrow\text{interpretation}.
		\]
		\textbf{Bike rental.} A bike rental service charges a \$6 unlocking fee and
		\$3 per hour. The total cost is \$24. Let $h$ be the number of hours.\vspace{-2mm}
		\[
		6+3h=24\;\Longrightarrow\;3h=18\;\Longrightarrow\;h=6
		\]
		
		\vspace{-2mm}
		The bike was rented for \textbf{6 hours}.
		
		\textbf{Online order.} Four identical notebooks plus a \$3 delivery charge cost
		\$27. Let $p$ be the price of one notebook.\vspace{-2mm}
		\[
		4p+3=27\;\Longrightarrow\;4p=24\;\Longrightarrow\;p=6
		\]
		
		\vspace{-2mm}
		Each notebook costs \textbf{\$6}.
		
		\textbf{Saving.} Zuleika already has \$25 and saves \$8 each week. When will she
		have \$81? Let $w$ be the number of weeks.\vspace{-2mm}
		\[
		25+8w=81\;\Longrightarrow\;8w=56\;\Longrightarrow\;w=7
		\]
		
		\vspace{-2mm}
		Zuleika needs \textbf{7 weeks}.

		\heading{8. Answer key}
		
		\textbf{Practice problems}
		
		\begin{enumerate}[label=\textbf{\arabic*.}]
			
			\item Let $h$ be the number of rental hours.
			\[
			6+4h<14+2h
			\]
			\[
			2h<8
			\quad\Longrightarrow\quad
			h<4.
			\]
			Therefore, Company A is cheaper for
			\[
			\boxed{0\leq h<4}.
			\]
			
			\item Let $m$ be the number of premium movies.
			\[
			3m<12+m
			\]
			\[
			2m<12
			\quad\Longrightarrow\quad
			m<6.
			\]
			Since $m$ is a whole number, Service A is cheaper for
			\[
			\boxed{m=0,1,2,3,4,5}.
			\]
			
			\item Let $w$ be the number of weeks.
			\[
			20+8w>44+5w
			\]
			\[
			3w>24
			\quad\Longrightarrow\quad
			w>8.
			\]
			Therefore, Plan A contains more money
			\[
			\boxed{\text{after more than 8 weeks}},
			\]
			or from week $9$ onward.
			
			\item Let $d$ be the trip distance in kilometres.
			\[
			5d<30
			\]
			\[
			d<6.
			\]
			Therefore, the taxi company is cheaper for
			\[
			\boxed{0\leq d<6\text{ km}}.
			\]
			
			\item Let $n$ be the number of notebooks.
			\[
			4n+10<6n
			\]
			\[
			10<2n
			\quad\Longrightarrow\quad
			n>5.
			\]
			Since $n$ is a whole number, Store A is cheaper for
			\[
			\boxed{n\geq 6}.
			\]
			
			\item Let $h$ be the number of parking hours.
			\[
			3h+5<3h+9
			\]
			\[
			5<9.
			\]
			This statement is always true. Therefore, Garage A is cheaper for
			\[
			\boxed{\text{every possible length of stay}}.
			\]
			
			\item Let $h$ be the rental duration in hours.
			\[
			4h+6<2+4+4h
			\]
			\[
			4h+6<4h+6
			\]
			\[
			6<6.
			\]
			This statement is never true. Both shops charge exactly the same amount
			for every rental duration. Therefore,
			\[
			\boxed{\text{Shop A is never cheaper than Shop B}}.
			\]
			
			
		\end{enumerate}

		\heading{Challenge problems -- Answer key}
		
		\begin{enumerate}[label=\textbf{\arabic*.}]
			
			\item
			\[
			3(2x-5)-2(3x-7)
			=
			5(x-4)-(5x-19).
			\]
			Simplifying each side,
			\[
			6x-15-6x+14
			=
			5x-20-5x+19,
			\]
			so
			\[
			-1=-1.
			\]
			This identity is true for every real number. Hence,
			\[
			\boxed{x\in\mathbb{R}}.
			\]
			The equation behaves differently from a typical linear equation because
			all terms containing $x$ cancel and the remaining statement is always true.
			
			\item Let
			\[
			A=\frac{2x-3}{5},
			\qquad
			B=\frac{3x+1}{7}.
			\]
			Notice that
			\[
			\frac{4x-9}{10}
			=
			\frac{2x-3}{5}-\frac{3}{10}
			=A-\frac{3}{10},
			\]
			and
			\[
			\frac{6x+5}{14}
			=
			\frac{3x+1}{7}+\frac{3}{14}
			=B+\frac{3}{14}.
			\]
			Therefore, the equation becomes
			\[
			A-B
			=
			A-\frac{3}{10}
			-\left(B+\frac{3}{14}\right).
			\]
			Thus,
			\[
			A-B
			=
			A-B-\frac{3}{10}-\frac{3}{14}
			=
			A-B-\frac{18}{35}.
			\]
			Subtracting $A-B$ from both sides gives
			\[
			0=-\frac{18}{35},
			\]
			which is impossible. Therefore,
			\[
			\boxed{\text{no real solution}}.
			\]
			

			\item Consider
			\[
			(a+1)\bigl((a-1)x+1\bigr)-1
			=
			(a+1)x+2.
			\]
			
			Expand the left-hand side:
			\[
			(a+1)(a-1)x+(a+1)-1
			=
			(a+1)x+2,
			\]
			so
			\[
			(a^2-1)x+a
			=
			(a+1)x+2.
			\]
			
			Move all terms to one side:
			\[
			(a^2-a-2)x+a-2=0.
			\]
			
			Factor:
			\[
			(a-2)(a+1)x+(a-2)=0,
			\]
			and therefore
			\[
			(a-2)\bigl((a+1)x+1\bigr)=0.
			\]
			
			We now consider the special values of $a$.
			
			\begin{itemize}
				\item If $a=2$, then
				\[
				0=0,
				\]
				so the equation is true for every real $x$. Therefore,
				\[
				\boxed{\text{infinitely many solutions}}.
				\]
				
				\item If $a=-1$, substituting into the original equation gives
				\[
				-1=2,
				\]
				which is impossible. Therefore,
				\[
				\boxed{\text{no solution}}.
				\]
				
				\item If $a\neq 2$ and $a\neq -1$, we may divide by $a-2$:
				\[
				(a+1)x+1=0.
				\]
				Hence
				\[
				x=-\frac{1}{a+1}.
				\]
				Therefore, there is exactly one real solution:
				\[
				\boxed{x=-\frac{1}{a+1}}.
				\]
			\end{itemize}
			
			Thus,
			\[
			\boxed{
				\begin{cases}
					\text{infinitely many solutions}, & a=2,\\[1mm]
					\text{no solution}, & a=-1,\\[1mm]
					x=-\dfrac{1}{a+1}, & a\neq -1,2.
				\end{cases}
			}
			\]
		
			
		\end{enumerate}
