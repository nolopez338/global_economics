	\textbf{Good survey questions:}
	
	\begin{multicols}{2}
		\begin{itemize}[leftmargin=0.6cm]
			\item Ask about real experiences;
			\item Are neutral and clear;
			\item Do not suggest an answer;
			\item Help identify needs or difficulties;
			\item Avoid asking users to choose a solution.
		\end{itemize}
	\end{multicols}
	
	\textbf{Example:} ``How often do you experience this situation?''
	
	\textbf{Avoid:} ``Is this situation difficult or frustrating for you?''
