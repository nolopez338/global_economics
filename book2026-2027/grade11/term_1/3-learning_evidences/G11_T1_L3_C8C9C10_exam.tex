\SubsectionBox{Conventions}

\textbf{Models.} C8 (1--10): $X=m+aT$.
C9 (11--16): $X_T=m+aT$, $X_N=m+aZ$, where
$T\sim t_{n-1}$, $Z\sim N(0,1)$, $n$ is sample size,
$m$ is location, and $a>0$ is scale.

\textbf{Standardization.} Subtract $m$ and divide by $a$:
\[
T=\frac{X_T-m}{a},\qquad Z=\frac{X_N-m}{a}.
\]
For inverse problems, transform back using
$x=m+at$ or $x=m+az$.

\textbf{C9 comparisons.} For every problem, calculate
$r_T$, $r_N$, and
\[
E_A=|r_T-r_N|,\qquad
E_{Pt}=\frac{100E_A}{|r_T|},\qquad
E_{Pn}=\frac{100E_A}{|r_N|}.
\]


\SubsectionBox{1. Courier arrival deviations \score{C8-\sml{1}}}
\textbf{C8 -- Direct left tail.} A sample of $n=10$ courier trips has arrival-time deviation from schedule, in minutes, with mean $m=1$ and approximate standard deviation $a=3$. Use $X=1+3T$, $T\sim t_{n-1}$.
Find $P(X<-4.499)$ and interpret the threshold. \score{1}

\SubsectionBox{2. Freight cost deviations \score{C8-\sml{1}}}
\textbf{C8 -- Direct right tail.} A sample of $n=13$ consignments has cost deviation from budget, in dollars, with mean $m=-4$ and approximate standard deviation $a=8$. Use $X=-4+8T$, $T\sim t_{n-1}$.
Find $P(X>-10.984)$ using a complement, and explain why it exceeds $0.5$. \score{1}

\SubsectionBox{3. Warehouse energy deviations \score{C8-\sml{1}}}

Find $P(7.19<X<12.476)$. \score{1}

\SubsectionBox{4. Crop shipment deviations \score{C8-\sml{1}}}

\SubsectionBox{5. Retail revenue alerts \score{C8-\sml{1}}}
