\section*{Evaluated criterion}
\begin{itemize}
 \item C3: Applies circular-arc probability density functions to calculate interval probabilities using the geometry of circles and circular segments.
\end{itemize}

\section{Introduction}\label{sec:c3-introduction}
A continuous probability density function (pdf) assigns probability through
\emph{area}: the probability that $X$ lies in an interval is the area between
the graph and the horizontal axis above that interval.  The entire area on the
support is $1$, and areas are never negative.  Individual points have zero
area, so including or excluding an endpoint does not change the probability.

In this activity the graph is a scaled upper semicircle.  If the underlying
circle has centre $(m,0)$ and radius $r$, its upper arc is
\[
 y=\sqrt{r^2-(x-m)^2},\qquad m-r\le x\le m+r.
\]
The raw semicircle has area $\pi r^2/2$, not $1$.  Multiplication by
$2/(\pi r^2)$ builds normalization into the pdf:
\[
 f_X(x)=
 \begin{cases}
 \dfrac{2}{\pi r^2}\sqrt{r^2-(x-m)^2},&m-r\le x\le m+r,\\[4pt]
 0,&\text{otherwise.}
 \end{cases}
 \tag{1}
\]
Thus a raw geometric area $A$ under the unscaled arc corresponds to
$P=\dfrac{2}{\pi r^2}A$, or equivalently $P=A/(\pi r^2/2)$.  When a problem
states (1), normalization is already built in; do not normalize a second time.

 \node[below] at (2,0) {$m+r$};
 \node at (0,1) {interval area};
\end{tikzpicture}
\end{center}

Only the following circle geometry is needed.
\begin{itemize}
 \item Circle area: $\pi r^2$; semicircle area: $\pi r^2/2$.
 \item A sector of central angle $\theta$ radians has area
 $A_{\rm sector}=\tfrac12r^2\theta$.
 \item The triangle between two radii has area
 $A_{\triangle}=\tfrac12r^2\sin\theta$.
 \item A circular segment is sector minus triangle:
 $A_{\rm segment}=\tfrac12r^2(\theta-\sin\theta)$.
 \item The vertical axis $x=m$ is an axis of symmetry.  Mirror regions have
 equal area; all omitted regions can be found using complements from area $1$.
\end{itemize}

For a right-hand cutoff $x=m+a$ ($0\le a\le r$), mark the point on the circle.
The radius to that point and the positive horizontal radius enclose
\[
 \theta=\arccos\!\left(\frac{a}{r}\right),\qquad
 h=\sqrt{r^2-a^2}.
\]
The raw area from $m+a$ to $m+r$ is the circular segment
\[
 S_r(a)=\frac12r^2\theta-\frac12ah
 =\frac12r^2\left(\theta-\sin\theta\right).                 \tag{2}
\]

\textbf{General method.} (1) Write the support and locate the requested
interval. (2) Sketch its vertical boundaries and name the raw region.
(3) Use symmetry or a complement first if it shortens the work. (4) Find each
needed central angle from a right triangle. (5) calculate sector area and
triangle area, then subtract to obtain each segment. (6) Combine the raw
regions. (7) Divide by the raw semicircle area, or multiply by the factor in
(1), and interpret the probability in context.

\subsection{Map intervals to regions}\label{sec:c3-geometry-map}
The unscaled arc $y=\sqrt{16-x^2}$ has support $[-4,4]$ and centre $0$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-geometry-map-a}{3.1}{Identify the geometric regions for $[-4,0]$, $[2,4]$, and $[-2,2]$.}
 \CriterionTask{subsec:c3-geometry-map-b}{3.2}{Explain how symmetry and a complement relate their areas.}
\end{enumerate}

\subsubsection*{C3.1 - Identify the geometric regions.}\phantomsection\label{subsec:c3-geometry-map-a}
The radius is $4$.  The interval $[-4,0]$ is the left quarter of the full
circle (one half of the upper semicircle).  The interval $[2,4]$ is a right
circular segment cut off by the vertical chord $x=2$.  The interval $[-2,2]$
is the central part of the upper semicircle between two vertical chords.  Its
raw area is the semicircle with the two congruent outer segments removed.
These identifications translate horizontal intervals into geometric regions;
no probability factor has yet been applied.
\secInicio[sec:c3-geometry][sec:c3-geometry-map]

\subsubsection*{C3.2 - Relate the areas.}\phantomsection\label{subsec:c3-geometry-map-b}
Reflection in $x=0$ sends $[2,4]$ to $[-4,-2]$, so the two tail segments have
equal raw area.  Therefore
\[
 A[-2,2]=\frac12\pi(4)^2-2A[2,4]=8\pi-2A[2,4].
\]
After normalization, the identical relationship is
$P(-2\le X\le2)=1-2P(2\le X\le4)$.  Also symmetry gives
$P(-4\le X\le0)=1/2$.
\secInicio[sec:c3-geometry][sec:c3-geometry-map]

\subsection{Build circular segments}\label{sec:c3-geometry-segments}
In a circle of radius $6$, a vertical cutoff is $3$ units to the right of the
centre.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-geometry-segments-a}{3.1}{Find the central angle, sector area, and triangle area for the right tail.}
 \CriterionTask{subsec:c3-geometry-segments-b}{3.2}{Find the segment area and the corresponding fraction of the upper semicircle.}
\end{enumerate}

\subsubsection*{C3.1 - Find the angle, sector, and triangle.}\phantomsection\label{subsec:c3-geometry-segments-a}
Here $a=3$, $r=6$, and the height at the cutoff is
$h=\sqrt{36-9}=3\sqrt3$.  Hence
\[
 \cos\theta=\frac{3}{6}=\frac12,quad \theta=\frac\pi3,
\]
\[
 A_{\rm sector}=\frac12(6^2)\frac\pi3=6\pi,qquad
 A_\triangle=\frac12(3)(3\sqrt3)=\frac{9\sqrt3}{2}.
\]
The sector and triangle share the two radii; their difference is exactly the
region under the arc to the right of the cutoff.
\secInicio[sec:c3-geometry][sec:c3-geometry-segments]

\subsubsection*{C3.2 - Find the segment and fraction.}\phantomsection\label{subsec:c3-geometry-segments-b}
Subtracting gives
\[
 A_{\rm segment}=6\pi-\frac{9\sqrt3}{2}.
\]
The upper semicircle area is $\pi(6^2)/2=18\pi$, so its fraction is
\[
 \frac{A_{\rm segment}}{18\pi}
 =\frac13-\frac{\sqrt3}{4\pi}\approx0.1955.
\]
If the semicircle is normalized as a pdf, that fraction is the probability of
the right-tail interval.
\secInicio[sec:c3-geometry][sec:c3-geometry-segments]

\section{Symmetric Interval Probabilities}\label{sec:c3-symmetry}
\SectionProblemsTableOfContents{

\subsection{Halves and mirrored intervals}\label{sec:c3-symmetry-halves}
Let $X$ have the already-normalized pdf
$f(x)=\dfrac{2}{25\pi}\sqrt{25-x^2}$ on $[-5,5]$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-symmetry-halves-a}{3.1}{Find $P(X\le0)$ and justify it using area.}
 \CriterionTask{subsec:c3-symmetry-halves-b}{3.2}{Compare $P(-5\le X\le-4)$ with $P(4\le X\le5)$.}
\end{enumerate}

\subsubsection*{C3.1 - Find the left-half probability.}\phantomsection\label{subsec:c3-symmetry-halves-a}
The raw semicircle area is $25\pi/2$.  The region from $-5$ to $0$ is one
half of it, with raw area $25\pi/4$.  Conversion to probability gives
\[
 P(X\le0)=\frac{2}{25\pi}\left(\frac{25\pi}{4}\right)=\frac12.
\]
Thus half of the modeled outcomes lie on either side of the centre.
\secInicio[sec:c3-symmetry][sec:c3-symmetry-halves]

\subsubsection*{C3.2 - Compare mirrored tail probabilities.}\phantomsection\label{subsec:c3-symmetry-halves-b}
Reflection across $x=0$ makes the raw regions congruent.  Both are multiplied
by the same normalization factor, so
\[
 P(-5\le X\le-4)=P(4\le X\le5).
\]
This conclusion follows from geometric symmetry without calculating either
segment separately.
\secInicio[sec:c3-symmetry][sec:c3-symmetry-halves]

\subsection{A centred interval by complement}\label{sec:c3-symmetry-centre}
Let $Y$ have pdf $f(y)=\dfrac{1}{8\pi}\sqrt{16-y^2}$ on $[-4,4]$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-symmetry-centre-a}{3.1}{Find the probability of one outer tail, $P(2\le Y\le4)$.}
 \CriterionTask{subsec:c3-symmetry-centre-b}{3.2}{Use symmetry and a complement to find $P(-2\le Y\le2)$.}
\end{enumerate}

\subsubsection*{C3.1 - Find one outer tail.}\phantomsection\label{subsec:c3-symmetry-centre-a}
With $r=4$ and $a=2$, the cutoff height is $2\sqrt3$ and
$\theta=\arccos(1/2)=\pi/3$.  Therefore
\[
 A_{\rm sector}=\tfrac12(16)(\pi/3)=\frac{8\pi}{3},
 \quad A_\triangle=\tfrac12(2)(2\sqrt3)=2\sqrt3,
\]
\[
 P(2\le Y\le4)=\frac1{8\pi}\left(\frac{8\pi}{3}-2\sqrt3\right)
 =\frac13-\frac{\sqrt3}{4\pi}\approx0.1955.
\]
The factor $1/(8\pi)=2/(\pi r^2)$ is already in the pdf.
\secInicio[sec:c3-symmetry][sec:c3-symmetry-centre]

\subsubsection*{C3.2 - Find the centred probability.}\phantomsection\label{subsec:c3-symmetry-centre-b}
The left and right outer segments are mirror images.  Their combined
probability is twice the result above, so
\[
 P(-2\le Y\le2)=1-2\left(\frac13-\frac{\sqrt3}{4\pi}\right)
 =\frac13+\frac{\sqrt3}{2\pi}\approx0.6090.
\]
Thus about $60.90\%$ of outcomes lie within $2$ units of the axis of symmetry.
\secInicio[sec:c3-symmetry][sec:c3-symmetry-centre]

\section{General Interval Probabilities}\label{sec:c3-general}

\subsection{Between two positive cutoffs}\label{sec:c3-general-positive}
Let $X$ have pdf $f(x)=\dfrac{2}{9\pi}\sqrt{9-x^2}$ on $[-3,3]$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-general-positive-a}{3.1}{Calculate the raw segments to the right of $x=1$ and $x=\frac32$.}
 \CriterionTask{subsec:c3-general-positive-b}{3.2}{Find $P(1\le X\le\frac32)$ and explain the subtraction.}
\end{enumerate}

\subsubsection*{C3.1 - Calculate the two tail segments.}\phantomsection\label{subsec:c3-general-positive-a}
For $a=1$, $h=\sqrt8=2\sqrt2$ and
$\theta_1=\arccos(1/3)$.  For $a=3/2$, $h=3\sqrt3/2$ and
$\theta_2=\arccos(1/2)=\pi/3$.  Sector minus triangle gives
\[
 S_3(1)=\frac92\arccos\!\left(\frac13\right)-\sqrt2,
\]
\[
 S_3\!\left(\frac32\right)=\frac92\left(\frac\pi3\right)
 -\frac12\left(\frac32\right)\left(\frac{3\sqrt3}{2}\right)
 =\frac{3\pi}{2}-\frac{9\sqrt3}{8}.
\]
\secInicio[sec:c3-general][sec:c3-general-positive]

\subsubsection*{C3.2 - Find the interval probability.}\phantomsection\label{subsec:c3-general-positive-b}
The tail beyond $1$ contains the tail beyond $3/2$ plus the desired strip.
Consequently the raw strip is $S_3(1)-S_3(3/2)$.  Applying the normalized
factor gives
\[
 P\left(1\le X\le\frac32\right)
 =\frac{2}{9\pi}\left[
 \frac92\arccos\!\left(\frac13\right)-\sqrt2
 -\frac{3\pi}{2}+\frac{9\sqrt3}{8}\right]
 \approx0.0963.
\]
The positive result is the area of the narrow region between the two vertical
cutoffs, expressed as a fraction of the whole semicircle.
\secInicio[sec:c3-general][sec:c3-general-positive]

\subsection{An interval crossing the axis}\label{sec:c3-general-cross}
For the same radius-$3$ normalized pdf, find the interval from $-1$ to $2$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-general-cross-a}{3.1}{Find the two raw areas excluded from $[-3,3]$.}
 \CriterionTask{subsec:c3-general-cross-b}{3.2}{Calculate $P(-1\le X\le2)$ and check that it lies between $0$ and $1$.}
\end{enumerate}

\subsubsection*{C3.1 - Find the excluded raw areas.}\phantomsection\label{subsec:c3-general-cross-a}
By reflection, the left tail $[-3,-1]$ has raw area $S_3(1)$:
\[
 A_L=\frac92\arccos\!\left(\frac13\right)-\sqrt2.
\]
For the right tail $[2,3]$, $h=\sqrt5$ and
$\theta_R=\arccos(2/3)$, so
\[
 A_R=\frac92\arccos\!\left(\frac23\right)-\frac12(2)(\sqrt5)
 =\frac92\arccos\!\left(\frac23\right)-\sqrt5.
\]
Both formulas explicitly subtract a radius triangle from its sector.
\secInicio[sec:c3-general][sec:c3-general-cross]

\subsubsection*{C3.2 - Calculate and check the probability.}\phantomsection\label{subsec:c3-general-cross-b}
The whole raw area is $9\pi/2$.  Subtract the two outside segments and
normalize:
\[
 P(-1\le X\le2)=1-\frac{2}{9\pi}(A_L+A_R)
\]
\[
 =1-\frac{2}{9\pi}\left[
 \frac92\arccos\!\left(\frac13\right)-\sqrt2
 +\frac92\arccos\!\left(\frac23\right)-\sqrt5\right]
 \approx0.5987.
\]
This is between $0$ and $1$ and exceeds $1/2$, reasonable because the interval
contains the centre and most of the high part of the arc.
\secInicio[sec:c3-general][sec:c3-general-cross]

\subsection{Bogot\'a delivery time}\label{sec:c3-applied-delivery}
A bicycle courier cooperative models delivery time $T$ (minutes) in central
Bogot\'a by
\[
 f_T(t)=\frac{1}{32\pi}\sqrt{64-(t-30)^2},\qquad22\le t\le38,
\]
and $0$ otherwise.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-applied-delivery-a}{3.1}{Verify geometrically that this pdf is normalized and locate $26\le T\le34$.}
 \CriterionTask{subsec:c3-applied-delivery-b}{3.2}{Find and interpret $P(26\le T\le34)$.}
\end{enumerate}

\subsubsection*{C3.1 - Verify and locate.}\phantomsection\label{subsec:c3-applied-delivery-a}
The arc has centre $30$, radius $8$, and raw area
$\pi(8^2)/2=32\pi$.  Multiplication by $1/(32\pi)$ makes total area $1$, so
normalization is already built in.  The interval $[26,34]$ is centred on $30$
and leaves two congruent tails, each beginning $4$ units from the centre.
\secInicio[sec:c3-applied][sec:c3-applied-delivery]

\subsubsection*{C3.2 - Find and interpret the probability.}\phantomsection\label{subsec:c3-applied-delivery-b}
For one tail, $a=4$, $h=4\sqrt3$, and $\theta=\pi/3$.  Thus
\[
 A_{\rm tail}=\tfrac12(64)(\pi/3)-\tfrac12(4)(4\sqrt3)
 =\frac{32\pi}{3}-8\sqrt3.
\]
Using two tails,
\[
 P(26\le T\le34)=1-\frac{2}{32\pi}\left(\frac{32\pi}{3}-8\sqrt3\right)
 =\frac13+\frac{\sqrt3}{2\pi}\approx0.6090.
\]
About $60.90\%$ of deliveries are predicted to take from $26$ through $34$
minutes.
\secInicio[sec:c3-applied][sec:c3-applied-delivery]

\subsection{Weekend demand in Chapinero}\label{sec:c3-applied-demand}
A Chapinero bakery models Saturday demand $D$ (hundreds of loaves) with a
normalized circular arc centred at $5$ with radius $3$:
\[
 f_D(d)=\frac{2}{9\pi}\sqrt{9-(d-5)^2},\qquad2\le d\le8.
\]
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-applied-demand-a}{3.1}{Find the sector and triangle areas needed for demand above $6.5$ hundred loaves.}
 \CriterionTask{subsec:c3-applied-demand-b}{3.2}{Calculate and interpret $P(D>6.5)$.}
\end{enumerate}

\subsubsection*{C3.1 - Find the geometric areas.}\phantomsection\label{subsec:c3-applied-demand-a}
The cutoff is $a=6.5-5=1.5$ units right of the centre.  Its height is
$\sqrt{9-1.5^2}=3\sqrt3/2$ and
$\theta=\arccos(1.5/3)=\pi/3$.  Hence
\[
 A_{\rm sector}=\tfrac12(9)(\pi/3)=\frac{3\pi}{2},\qquad
 A_\triangle=\tfrac12(1.5)\left(\frac{3\sqrt3}{2}\right)
 =\frac{9\sqrt3}{8}.
\]
\secInicio[sec:c3-applied][sec:c3-applied-demand]

\subsubsection*{C3.2 - Calculate and interpret the tail probability.}\phantomsection\label{subsec:c3-applied-demand-b}
The required raw segment is sector minus triangle.  Therefore
\[
 P(D>6.5)=\frac{2}{9\pi}\left(\frac{3\pi}{2}-\frac{9\sqrt3}{8}\right)
 =\frac13-\frac{\sqrt3}{4\pi}\approx0.1955.
\]
The model predicts about a $19.55\%$ chance that demand exceeds $650$ loaves;
the bakery can use this risk when planning extra production.
\secInicio[sec:c3-applied][sec:c3-applied-demand]

\subsection{Entrepreneurial investment return}\label{sec:c3-applied-return}
A Bogot\'a entrepreneur models the one-year return $R$ (percentage points) on
a small-business expansion by
\[
 f_R(r)=\frac{1}{50\pi}\sqrt{100-(r-8)^2},\qquad-2\le r\le18,
\]
and $0$ otherwise.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-applied-return-a}{3.1}{Determine the circular regions needed for $P(3\le R\le13)$ and calculate one tail segment.}
 \CriterionTask{subsec:c3-applied-return-b}{3.2}{Find the probability and interpret it for the investment decision.}
\end{enumerate}

\subsubsection*{C3.1 - Determine and calculate the regions.}\phantomsection\label{subsec:c3-applied-return-a}
The centre is $8$, radius is $10$, and $[3,13]$ is centred on $8$ with
half-width $a=5$.  It leaves two congruent segments.  For the right one,
$h=\sqrt{100-25}=5\sqrt3$ and $\theta=\arccos(1/2)=\pi/3$.  Therefore
\[
 A_{\rm tail}=\tfrac12(100)(\pi/3)-\tfrac12(5)(5\sqrt3)
 =\frac{50\pi}{3}-\frac{25\sqrt3}{2}.
\]
The raw semicircle area is $50\pi$, matching the reciprocal normalization
factor shown in the pdf.
\secInicio[sec:c3-applied][sec:c3-applied-return]

\subsubsection*{C3.2 - Find and interpret the probability.}\phantomsection\label{subsec:c3-applied-return-b}
Subtract both outside segments and convert raw area to probability:
\[
 P(3\le R\le13)
 =1-\frac{2}{50\pi}\left(\frac{50\pi}{3}-\frac{25\sqrt3}{2}\right)
 =\frac13+\frac{\sqrt3}{2\pi}\approx0.6090.
\]
According to the model, there is about a $60.90\%$ chance that the annual
return is between $3\%$ and $13\%$.  This probability describes uncertainty;
it does not guarantee that the expansion will earn a positive return.
\secInicio[sec:c3-applied][sec:c3-applied-return]
