\section{Introduction}\label{sec:c3-introduction}
A \emph{circular distribution} is a probability distribution for a variable
measured on a circle, such as an angle, a compass direction, or a time of day.
It is not a density whose graph happens to be a circular arc.  The defining
feature is that the beginning and end of the scale are the same point: for
example, $0^\circ$ and $360^\circ$ represent the same direction.

If $\Theta$ is measured in radians, then
\[
 \Theta\equiv\Theta+2\pi.
\]
Consequently a circular probability density function (pdf) is periodic:
\[
 f(\theta)=f(\theta+2\pi),
 \qquad
 \int_0^{2\pi}f(\theta)\,d\theta=1.
 \tag{1}
\]
As for an ordinary continuous distribution, probability is area under the
density:
\[
 P(a\leq\Theta\leq b)=\int_a^b f(\theta)\,d\theta.
 \tag{2}
\]
If an interval crosses $0$, split it at the end of a chosen cycle.  For
example, on $[0,2\pi)$,
\[
 P\left(\frac{11\pi}{6}\leq\Theta\leq\frac{\pi}{6}\right)
 =\int_{11\pi/6}^{2\pi}f(\theta)\,d\theta
  +\int_0^{\pi/6}f(\theta)\,d\theta.
 \tag{3}
\]
This notation means the short arc travelled through $0$, not the empty
ordinary linear interval.

The simplest example is the \emph{circular uniform distribution}:
\[
 f(\theta)=\frac1{2\pi},\qquad 0\leq\theta<2\pi.
 \tag{4}
\]
Every direction is equally likely, so an arc of angular width $L$ has
probability $L/(2\pi)$.  Thus
\[
 P\left(0\leq\Theta\leq\frac\pi2\right)
 =\int_0^{\pi/2}\frac1{2\pi}\,d\theta=\frac14.
\]

Ordinary averages can be misleading for circular data: the arithmetic mean of
$1^\circ$ and $359^\circ$ is $180^\circ$, although both directions are close
to $0^\circ$.  Instead, represent $\theta_i$ by the unit vector
$(\cos\theta_i,\sin\theta_i)$ and calculate
\[
 \bar C=\frac1n\sum_{i=1}^n\cos\theta_i,
 \qquad
 \bar S=\frac1n\sum_{i=1}^n\sin\theta_i,
 \qquad
 \bar\theta=\operatorname{atan2}(\bar S,\bar C).
 \tag{5}
\]
If $\operatorname{atan2}$ returns a negative angle, add $2\pi$ (or
$360^\circ$) to report it on the chosen cycle.  The mean resultant length
$R=\sqrt{\bar C^2+\bar S^2}$ indicates concentration: values near $1$ mean
the observations point in similar directions, while values near $0$ mean
there is little or no single representative direction.

``Circular distribution'' names a category, not one particular model.  Its
members include the circular uniform distribution, the von Mises distribution
(often used for angles in a role analogous to the normal distribution), and
wrapped distributions such as the wrapped normal.  This activity concentrates
on the circular uniform model and circular means.

\subsection{A quarter-circle and periodicity}\label{sec:c3-uniform-basics}
Suppose $\Theta$ is uniformly distributed on $[0,2\pi)$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-uniform-basics-a}{3.1}{Verify that the density in (4) has total probability $1$.}
 \CriterionTask{subsec:c3-uniform-basics-b}{3.2}{Find $P(\pi/3\leq\Theta\leq5\pi/6)$ and explain why adding $2\pi$ to both endpoints does not change it.}
\end{enumerate}

\subsubsection*{C3.1 - Verify normalization.}\phantomsection\label{subsec:c3-uniform-basics-a}
The density is nonnegative and
\[
 \int_0^{2\pi}\frac1{2\pi}\,d\theta
 =\frac{2\pi}{2\pi}=1.
\]
It therefore satisfies both requirements for a pdf on one complete cycle.
\secInicio[sec:c3-uniform][sec:c3-uniform-basics]

\subsubsection*{C3.2 - Find and interpret the probability.}\phantomsection\label{subsec:c3-uniform-basics-b}
The angular width is $5\pi/6-\pi/3=\pi/2$.  Therefore
\[
 P\left(\frac\pi3\leq\Theta\leq\frac{5\pi}6\right)
 =\frac{\pi/2}{2\pi}=\frac14.
\]
The endpoints $\pi/3+2\pi$ and $5\pi/6+2\pi$ describe the same directions
one revolution later.  Periodicity, $f(\theta)=f(\theta+2\pi)$, makes the
probability unchanged.
\secInicio[sec:c3-uniform][sec:c3-uniform-basics]

\subsection{An interval that crosses zero}\label{sec:c3-uniform-wrap}
A wind direction $\Theta$ is uniformly distributed around the compass.  Find
the probability that it lies within $30^\circ$ of north, where north is
$0^\circ$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-uniform-wrap-a}{3.1}{Write the event as two intervals on $[0^\circ,360^\circ)$.}
 \CriterionTask{subsec:c3-uniform-wrap-b}{3.2}{Calculate the probability and express it as a percentage.}
\end{enumerate}

\subsubsection*{C3.1 - Split the wrapped interval.}\phantomsection\label{subsec:c3-uniform-wrap-a}
Directions within $30^\circ$ of north run from $330^\circ$ to $360^\circ$
and from $0^\circ$ to $30^\circ$.  Thus the event is
\[
 \{330^\circ\leq\Theta<360^\circ\}\ \cup\
 \{0^\circ\leq\Theta\leq30^\circ\}.
\]
The two pieces are disjoint on the chosen representation of the cycle.
\secInicio[sec:c3-uniform][sec:c3-uniform-wrap]

\subsubsection*{C3.2 - Calculate the probability.}\phantomsection\label{subsec:c3-uniform-wrap-b}
The two arcs have total width $30^\circ+30^\circ=60^\circ$.  Hence
\[
 P=\frac{60^\circ}{360^\circ}=\frac16\approx0.1667.
\]
There is therefore a probability of about $16.67\%$.  Treating
$330^\circ\leq\Theta\leq30^\circ$ as an ordinary linear interval would miss
the circular meaning of the event.
\secInicio[sec:c3-uniform][sec:c3-uniform-wrap]

\subsection{Directions near the cycle boundary}\label{sec:c3-means-boundary}
Two delivery riders leave a depot in directions $10^\circ$ and $350^\circ$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-means-boundary-a}{3.1}{Explain why the arithmetic mean is inappropriate.}
 \CriterionTask{subsec:c3-means-boundary-b}{3.2}{Use unit vectors to find the circular mean and resultant length.}
\end{enumerate}

\subsubsection*{C3.1 - Compare the directions.}\phantomsection\label{subsec:c3-means-boundary-a}
The arithmetic mean is $(10^\circ+350^\circ)/2=180^\circ$, which points
south.  Both actual directions are only $10^\circ$ from north, so this answer
is an artefact of cutting the circular scale at $0^\circ$.
\secInicio[sec:c3-means][sec:c3-means-boundary]

\subsubsection*{C3.2 - Calculate the circular mean.}\phantomsection\label{subsec:c3-means-boundary-b}
Using $350^\circ=-10^\circ$ on the circle,
\[
 \bar C=\frac{\cos10^\circ+\cos(-10^\circ)}2=\cos10^\circ,
 \qquad
 \bar S=\frac{\sin10^\circ+\sin(-10^\circ)}2=0.
\]
Therefore
\[
 \bar\theta=\operatorname{atan2}(0,\cos10^\circ)=0^\circ,
 \qquad R=\cos10^\circ\approx0.9848.
\]
The circular mean correctly points north, and $R$ close to $1$ shows strong
agreement between the two directions.
\secInicio[sec:c3-means][sec:c3-means-boundary]

\subsection{A mean with weak concentration}\label{sec:c3-means-concentration}
Customer arrivals are grouped by position on a 24-hour clock.  Three observed
times correspond to angles $0^\circ$, $120^\circ$, and $240^\circ$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-means-concentration-a}{3.1}{Calculate $\bar C$ and $\bar S$.}
 \CriterionTask{subsec:c3-means-concentration-b}{3.2}{Decide whether a single average direction is meaningful.}
\end{enumerate}

\subsubsection*{C3.1 - Average the vectors.}\phantomsection\label{subsec:c3-means-concentration-a}
The three unit vectors are equally spaced, and
\[
 \bar C=\frac{1-\frac12-\frac12}{3}=0,
 \qquad
 \bar S=\frac{0+\frac{\sqrt3}{2}-\frac{\sqrt3}{2}}{3}=0.
\]
Thus the average vector is the zero vector and $R=0$.
\secInicio[sec:c3-means][sec:c3-means-concentration]

\subsubsection*{C3.2 - Interpret the result.}\phantomsection\label{subsec:c3-means-concentration-b}
Because $\operatorname{atan2}(0,0)$ has no defined direction, there is no
unique circular mean for these data.  This is informative rather than an
error: the observations are perfectly balanced around the circle, so a single
representative time would be misleading.
\secInicio[sec:c3-means][sec:c3-means-concentration]

\subsection{Online orders by time of day}\label{sec:c3-applied-time}
Under a circular uniform model for time of day, let $H$ be the time on a
24-hour clock.  A shop manager studies orders from 22:30 through 01:30.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-applied-time-a}{3.1}{Represent the event without treating midnight as an endpoint of time.}
 \CriterionTask{subsec:c3-applied-time-b}{3.2}{Find and interpret its probability.}
\end{enumerate}

\subsubsection*{C3.1 - Represent the event.}\phantomsection\label{subsec:c3-applied-time-a}
On $[0,24)$ hours the event is
\[
 \{22.5\leq H<24\}\ \cup\ \{0\leq H\leq1.5\}.
\]
Equivalently, converting time $h$ to angle by $\theta=2\pi h/24$, it is the
wrapped arc from $15\pi/8$ through $0$ to $\pi/8$.
\secInicio[sec:c3-applied][sec:c3-applied-time]

\subsubsection*{C3.2 - Find and interpret the probability.}\phantomsection\label{subsec:c3-applied-time-b}
The event lasts $1.5+1.5=3$ hours.  Uniformity gives
\[
 P(22{:}30\text{ through }01{:}30)=\frac3{24}=\frac18=0.125.
\]
The model assigns $12.5\%$ of orders to that time window.  This result depends
on the uniform-model assumption; real order times may be concentrated at
particular hours.
\secInicio[sec:c3-applied][sec:c3-applied-time]

\subsection{Choosing an appropriate model}\label{sec:c3-applied-model}
An analyst considers three variables: daily revenue in pesos, wind direction,
and the time of day at which peak electricity demand occurs.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c3-applied-model-a}{3.1}{Identify which variables are naturally circular and justify the choice.}
 \CriterionTask{subsec:c3-applied-model-b}{3.2}{Name possible circular models when directions are uniform and when they cluster around one direction.}
\end{enumerate}

\subsubsection*{C3.1 - Identify circular variables.}\phantomsection\label{subsec:c3-applied-model-a}
Wind direction and peak-demand time are circular: $0^\circ=360^\circ$, and
midnight is both the end of one day and the beginning of the next.  Revenue is
linear; its largest and smallest values are not identified with each other.
The shape of a graph alone does not determine whether a variable is circular.
\secInicio[sec:c3-applied][sec:c3-applied-model]

\subsubsection*{C3.2 - Choose model families.}\phantomsection\label{subsec:c3-applied-model-b}
The circular uniform distribution is suitable if every direction is equally
likely.  If observations cluster around a preferred direction, a von Mises
model may be suitable; a wrapped normal is another possible model.  Data and
context are needed to choose among them: ``circular distribution'' does not
refer to one universal density.
\secInicio[sec:c3-applied][sec:c3-applied-model]
