\section*{Introduction}

The exponential distribution is commonly used to model the time $T$ until the
next occurrence of an event. In general, we write
\[
\boxed{T\sim\operatorname{Exponential}(\lambda)},\qquad \lambda>0,
\]
where $\lambda$ is the rate at which events occur.

For an exponential random variable, the possible waiting times satisfy
\[
\boxed{T\geq0}.
\]

The general case can be summarized by the following relationships:

\[
\begin{array}{c|c}
	\textbf{Concept} & \textbf{General expression}\\[4pt]
	\hline
	\text{Random variable} & T=\text{waiting time until the next event}\\[4pt]
	\text{Possible values} & T\geq0\\[4pt]
	\text{Rate} & \lambda>0
\end{array}
\]

The three basic probability cases are shown below. The equation is presented
alongside each graph, and the shaded region represents the corresponding
probability.

exponential model. In the following problems, we will identify the random
variable, determine the rate parameter when necessary, and use the appropriate
probability expression.

\subsection{Mobile-data support requests}\label{sec:c4-identify-callcenter}

A mobile phone company provides technical support for customers who
experience problems with their mobile data. Throughout the afternoon, customers call the company's support line to report problems or ask for assistance. The company wants to understand how frequently these calls arrive.

To study this, an analyst randomly chooses a moment during the afternoon and starts a timer. The analyst then records how much time passes before the next mobile-data support call arrives. For example, if the analyst starts the timer at 2:15 p.m. and the next call arrives at 2:18 p.m., the recorded time is
3 minutes.


\textbf{Question.}
\begin{enumerate}
	\CriterionTask{subsec:c4-identify-callcenter-a}{4.1}{Identify the random variable and state the possible values it can take.}
\end{enumerate}

\columnbreak

\subsubsection*{C4 - Identify the random variable and its possible values}\phantomsection\label{subsec:c4-identify-callcenter-a}
Let
\[
\boxed{
	\begin{aligned}
		T={}&\text{the time from a randomly selected observation time}\\
		&\text{until the next mobile-data support call arrives.}
\end{aligned}}
\]
The random variable $T$ measures elapsed time, so it cannot take negative
values. Therefore,
$$
\boxed{T\geq0}.
$$

\end{multicols}

\begin{multicols}{2}
	
	\subsection{Customer-support calls}\label{sec:c4-probability-support}
	
	A mobile phone company in Bogotá receives technical-support calls throughout
	the afternoon. The time, in minutes, from a randomly selected observation
	time until the next support call is modeled by an exponential distribution
	with rate 2 calls per minute.
	
	\textbf{Question.}
	\begin{enumerate}
		\CriterionTask{subsec:c4-probability-support-b}{4}{Find the probability that the next support call takes more than 5 minutes to arrive.}
	\end{enumerate}
	
	\subsubsection*{Define the random variable}\phantomsection\label{subsec:c4-probability-support-a}
	
	\[
	\boxed{T=\text{waiting time until the next support call, in minutes}}
	\]
	
	\subsubsection*{C4.3 - Find a probability}\phantomsection\label{subsec:c4-probability-support-b}
	
	The company wants to find the probability that the next support call takes
	more than 5 minutes to arrive, i.e.
	\[
	P(T>5)
	\]
	
	\[
	\begin{array}{c|c}
		\textbf{General} & P(T>t)=e^{-\lambda t} \\[6pt]
		\textbf{Context} & P(T>5),\quad \lambda=2
	\end{array}
	\]
	Therefore,
	\[
	P(T>5)=e^{-(2)(5)}
	=e^{-10} \]
	\[
	\boxed{P(T>5)\approx0.0000454}
	\]
	
\end{multicols}

\begin{multicols}{2}
	
	\subsection{Customer-support calls}\label{sec:c4-find-rate-support}
	
	A mobile phone company in Bogotá receives technical-support calls throughout
	the afternoon. The time $T$, in minutes, between a randomly selected
	observation time and the next support call is modeled by an exponential
	distribution.
	
	Based on previous observations, the company estimates that there is a 90\%
	probability that the next support call arrives within 10 minutes.
	
	\textbf{Question.}
	\begin{enumerate}
		\CriterionTask{subsec:c4-find-rate-support-a}{4.2}{Find the rate parameter $\lambda$ of the exponential distribution.}
	\end{enumerate}
	
	\subsubsection*{C4.2 - Find the rate parameter}\phantomsection\label{subsec:c4-find-rate-support-a}
	
	\[
	\begin{array}{c|c}
		\textbf{General} & P(T\leq t)=1-e^{-\lambda t} \\[6pt]
		\textbf{Context} & P(T\leq10)=0.90
	\end{array}
	\]
	Therefore,
	\[
	0.90=1-e^{-10\lambda}
	\quad \longrightarrow \quad e^{-10\lambda}=0.10
	\]
	\[
	\longrightarrow \quad \ln(e^{-10\lambda})=\ln(0.10)
	\]
	\[
	\longrightarrow \quad -10\lambda=\ln(0.10)
	\quad \longrightarrow \quad
	\lambda=-\frac{\ln(0.10)}{10}
	\]
	
	\[
	\boxed{\lambda\approx0.2303\text{ calls/minute}}
	\]
	
\end{multicols}

\noindent\rule{\textwidth}{0.4pt}

\begin{multicols}{2}
	
	\subsection{Food-delivery requests}\label{sec:c4-find-rate-delivery}
	
	A food-delivery platform in Bogotá receives delivery requests throughout
	the evening. The time $T$, in minutes, from a randomly selected observation
	time until the next delivery request is modeled by an exponential
	distribution.
	
	Based on previous observations, the platform estimates that there is a 75\%
	probability that the next delivery request arrives within 8 minutes.
	
	\textbf{Question.}
	\begin{enumerate}
		\CriterionTask{subsec:c4-find-rate-delivery-a}{4.2}{Find the rate parameter $\lambda$ of the exponential distribution.}
	\end{enumerate}
	
	\subsubsection*{C4.2 - Find the rate parameter}\phantomsection\label{subsec:c4-find-rate-delivery-a}
	
	\[
	\begin{array}{c|c}
		\textbf{General} & P(T\leq t)=1-e^{-\lambda t} \\[6pt]
		\textbf{Context} & P(T\leq8)=0.75
	\end{array}
	\]
	Therefore,
	\[
	0.75=1-e^{-8\lambda}
	\quad \longrightarrow \quad e^{-8\lambda}=0.25
	\]
	\[
	\longrightarrow \quad \ln(e^{-8\lambda})=\ln(0.25)
	\]
	\[
	\longrightarrow \quad -8\lambda=\ln(0.25)
	\quad \longrightarrow \quad
	\lambda=-\frac{\ln(0.25)}{8}
	\]
	
	\[
	\boxed{\lambda\approx0.1733\text{ requests/minute}}
	\]
	
\end{multicols}

	\[
	\begin{array}{c|c}

	\end{array}

	\[
