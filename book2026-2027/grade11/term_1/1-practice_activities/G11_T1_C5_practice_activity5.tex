		{\large\bfseries Practice: Mean and Variance of Continuous Distributions}
	\end{center}
	
	\vspace{-4mm}
	
	\noindent\textbf{Purpose.}
	Calculate and interpret the mean (expected value) and variance of
	continuous probability distributions using standard formulas.
	
	\vspace{1mm}
	
	\noindent\textbf{Instructions.}
	For each situation:
	
	\begin{enumerate}[
		leftmargin=*,
		itemsep=1pt,
		parsep=0pt,
		topsep=1pt
		]
		\item Identify the probability distribution and its parameters.
		\item Calculate the mean, or expected value, $E(X)$.
		\item Calculate the variance, $Var(X)$.
		\item Calculate the standard deviation, $\sigma_X = \sqrt{Var(X)}$.
	\end{enumerate}

		\item \textbf{Waiting for a bus after school.}

			\item Identify the distribution and its parameters.
			\item Calculate the mean waiting time.
			\item Calculate the variance of the waiting time.
			\item Calculate the standard deviation of the waiting time.
			\item Calculate the probability that the waiting time is within one
			standard deviation of the mean.
		\end{enumerate}
		
		
		\item \textbf{Delivery time for a small business.}
		
		A local business records the time required to deliver an order within
		the city. Delivery times range from 20 to 50 minutes. The probability
		density increases linearly from 20 to 50 minutes, so longer delivery
		times are more likely than shorter ones.
		
		\begin{enumerate}[label=\alph*), itemsep=1pt, topsep=1pt]
			\item Identify the distribution and its parameters.
			\item Calculate the mean delivery time.
			\item Calculate the variance of the delivery time.

			\item Calculate the probability that the delivery time is within one
			standard deviation of the mean.
		\end{enumerate}

		\begin{enumerate}[label=\alph*), itemsep=1pt, topsep=1pt]

			\item Calculate the variance of the number of sandwiches sold.

		\begin{enumerate}[label=\alph*), itemsep=1pt, topsep=1pt]
			\item Identify the distribution and its parameters.
			\item Calculate the mean time spent in the library.
			\item Calculate the variance of the time spent in the library.

	\section*{Solutions}
	
	\section{Uniform distribution}
	
	\SolutionStage{1}{Distribution and parameters}
	
	The waiting time $T$ is equally likely to take any value between 2 and
	12 minutes. Therefore,
	\[
	T\sim U(a,b)
	\qquad \longrightarrow \qquad
	T\sim U(2,12)
	\qquad \longrightarrow \qquad
	a=2,\qquad b=12.
	\]
	
	\SolvedGraph{Uniform density}{
		\UniformGraph{2}{12}
	}
	
	\SolutionStage{2}{Mean}
	
	For a uniform distribution,
	\[
	E(T)=\frac{a+b}{2}
	\qquad \longrightarrow \qquad
	E(T)=\frac{2+12}{2}
	=7
	\qquad \longrightarrow \qquad
	\boxed{E(T)=7\text{ minutes}}
	\]
	
	\SolutionStage{3}{Variance}
	
	For a uniform distribution,
	\[
	Var(T)=\frac{(b-a)^2}{12}
	\qquad \longrightarrow \qquad
	Var(T)
	=\frac{(12-2)^2}{12}
	=\frac{100}{12}
	=\frac{25}{3}
	\approx8.33
	\qquad \longrightarrow \qquad
	\boxed{Var(T)\approx8.33\text{ minutes}^2}
	\]
	
	\SolutionStage{4}{Standard Deviation}
	
	The standard deviation is
	\[
	\sigma_T=\sqrt{\frac{25}{3}}
	\approx2.89
	\qquad \longrightarrow \qquad
	\boxed{\sigma_T\approx2.89\text{ minutes}}
	\]
	
	\SolutionStage{5}{Probability within one standard deviation}
	
	We want the probability that the waiting time is within one standard
	deviation of the mean:
	\[
	P(\mu-\sigma_T<T<\mu+\sigma_T)
	\]
	
	Using $\mu=7$ and $\sigma_T\approx2.89$,
	\[
	P(7-2.89<T<7+2.89)
	=
	P(4.11<T<9.89)
	\]
	
	For a uniform distribution,
	\[
	P(c<T<d)=\frac{d-c}{b-a}
	\]
	
	Therefore,
	\[
	P(4.11<T<9.89)
	=
	\frac{9.89-4.11}{12-2}
	=
	\frac{5.78}{10}
	\approx0.578
	\]
	
	\[
	\boxed{P(\mu-\sigma_T<T<\mu+\sigma_T)\approx0.578}
	\]
	
	Thus, approximately $57.8\%$ of waiting times are within one standard
	deviation of the mean.
	
	\section{Linear increasing distribution}
	
	\SolutionStage{1}{Distribution and parameters}
	
	The delivery time $D$ ranges from 20 to 50 minutes, and its probability
	density increases linearly. Therefore,
	\[
	D\sim\operatorname{Lin}_{\mathrm{inc}}(a,b)
	\qquad \longrightarrow \qquad
	D\sim\operatorname{Lin}_{\mathrm{inc}}(20,50)
	\qquad \longrightarrow \qquad
	a=20\qquad b=50
	\]
	
	\SolvedGraph{Linear increasing density}{
		\LinearIncreasingGraph{20}{50}
	}
	
	\SolutionStage{2}{Mean}
	
	For a linear increasing distribution,
	\[
	E(D)=\frac{a+2b}{3}.
		\qquad \longrightarrow \qquad
	E(D)
	=\frac{20+2(50)}{3}
	=\frac{120}{3}
	=40
		\qquad \longrightarrow \qquad
	\boxed{E(D)=40\text{ minutes}}
	\]
	
	\SolutionStage{3}{Variance}
	
	For a linear increasing distribution,
	\[
	Var(D)=\frac{(b-a)^2}{18}
		\qquad \longrightarrow \qquad
	Var(D)
	=\frac{(50-20)^2}{18}
	=\frac{900}{18}
	=50
		\qquad \longrightarrow \qquad
	\boxed{Var(D)=50\text{ minutes}^2}
	\]
	
	\SolutionStage{4}{Interpretation}
	
	The standard deviation is
	\[
	\sigma_D=\sqrt{50}
	\approx7.07
		\qquad \longrightarrow \qquad
	\boxed{\sigma_D\approx7.07\text{ minutes}}
	\]
	
	\SolutionStage{5}{Probability within one standard deviation}
	
	We want the probability that the delivery time is within one standard
	deviation of the mean:
	\[
	P(\mu-\sigma_D<D<\mu+\sigma_D)
	\]
	
	Using $\mu=40$ and $\sigma_D\approx7.07$,
	\[
	P(40-7.07<D<40+7.07)
	=
	P(32.93<D<47.07)
	\]
	
	For a linear increasing distribution, the probability density is
	\[
	f(d)=\frac{2(d-a)}{(b-a)^2}.
	\]
	
	Therefore,
	\[
	P(32.93<D<47.07)
	=
	\int_{32.93}^{47.07}
	\frac{2(d-20)}{(50-20)^2}\,dd
	\]
	
	\[
	=
	\left[
	\frac{(d-20)^2}{900}
	\right]_{32.93}^{47.07}
	\approx0.619
	\]
	
	\[
	\boxed{P(\mu-\sigma_D<D<\mu+\sigma_D)\approx0.619}
	\]
	
	Thus, approximately $61.9\%$ of delivery times are within one standard
	deviation of the mean.

	\section{Linear decreasing distribution}
	
	\SolutionStage{1}{Distribution and parameters}
	
	The library time $L$ ranges from 20 to 80 minutes, and its probability
	density decreases linearly. Therefore,
	\[
	L\sim\operatorname{Lin}_{\mathrm{dec}}(a,b)
	\qquad \longrightarrow \qquad
	L\sim\operatorname{Lin}_{\mathrm{dec}}(20,80)
		\qquad \longrightarrow \qquad
	a=20,\qquad b=80
	\]
	\SolvedGraph{Linear increasing density}{
		\LinearDecreasingGraph{20}{80}
	}
	
	\SolutionStage{2}{Mean}
	
	For a linear decreasing distribution,
	\[
	E(L)=\frac{2a+b}{3}
		\qquad \longrightarrow \qquad
	E(L)
	=\frac{2(20)+80}{3}
	=\frac{120}{3}
	=40
		\qquad \longrightarrow \qquad
	\boxed{E(L)=40\text{ minutes}}
	\]
	
	\SolutionStage{3}{Variance}
	
	For a linear decreasing distribution,
	\[
	Var(L)=\frac{(b-a)^2}{18}
		\qquad \longrightarrow \qquad
	Var(L)
	=\frac{(80-20)^2}{18}
	=\frac{3600}{18}
	=200
		\qquad \longrightarrow \qquad
	\boxed{Var(L)=200\text{ minutes}^2}
	\]
	
	\SolutionStage{4}{Interpretation}
	The standard deviation is
	\[
	\sigma_L=\sqrt{200}
	\approx14.14
		\qquad \longrightarrow \qquad
	\boxed{\sigma_L\approx14.14\text{ minutes}}.
	\]
	
	\SolutionStage{5}{Probability within one standard deviation}
	
	We want the probability that the library time is within one standard
	deviation of the mean:
	\[
	P(\mu-\sigma_L<L<\mu+\sigma_L)
	\]
	
	Using $\mu=40$ and $\sigma_L\approx14.14$,
	\[
	P(40-14.14<L<40+14.14)
	=
	P(25.86<L<54.14)
	\]
	
	For a linear decreasing distribution, the probability density is
	\[
	f(l)=\frac{2(b-l)}{(b-a)^2}.
	\]
	
	Therefore,
	\[
	P(25.86<L<54.14)
	=
	\int_{25.86}^{54.14}
	\frac{2(80-l)}{(80-20)^2}\,dl
	\]
	
	\[
	=
	\left[
	\frac{2bl-l^2}{3600}
	\right]_{25.86}^{54.14}
	\approx0.635
	\]
	
	\[
	\boxed{P(\mu-\sigma_L<L<\mu+\sigma_L)\approx0.635}
	\]
	
	Thus, approximately $63.5\%$ of library times are within one standard
	deviation of the mean.
	
	
	\section{Triangular distribution}
	
	\SolutionStage{1}{Distribution and parameters}
	
	The number of sandwiches sold, $S$, ranges from 20 to 80, with the
	highest density at 50 sandwiches. Therefore,
	\[
	S\sim\operatorname{Tri}(a,b,c)
	\qquad \longrightarrow \qquad
	S\sim\operatorname{Tri}(20,80,50)
		\qquad \longrightarrow \qquad
	a=20\qquad b=80\qquad c=50
	\]
	
	\SolvedGraph{Triangular density}{
		\TriangularGraph{20}{50}{80}
	}
	
	\SolutionStage{2}{Mean}
	
	For a triangular distribution,
	\[
	E(S)=\frac{a+b+c}{3}
		\qquad \longrightarrow \qquad
	E(S)
	=\frac{20+80+50}{3}
	=\frac{150}{3}
	=50
		\qquad \longrightarrow \qquad
	\boxed{E(S)=50\text{ sandwiches}}
	\]
	
	\SolutionStage{3}{Variance}
	
	For a triangular distribution,
	\[
	Var(S)
	=
	\frac{a^2+b^2+c^2-ab-ac-bc}{18}.
	\]
	\[
	 \longrightarrow \qquad
	\begin{aligned}
		Var(S)
		&=
		\frac{
			20^2+80^2+50^2
			-(20)(80)
			-(20)(50)
			-(80)(50)
		}{18}
		\\
		&=
		\frac{400+6400+2500-1600-1000-4000}{18}=
		\frac{2700}{18}=150
	\end{aligned}
		\qquad \longrightarrow \qquad
	\boxed{Var(S)=150\text{ sandwiches}^2}
	\]
	
	\SolutionStage{4}{Interpretation}
	
	The standard deviation is
	\[
	\sigma_S=\sqrt{150}
	\approx12.25
			\qquad \longrightarrow \qquad
	\boxed{\sigma_S\approx12.25\text{ sandwiches}}.
	\]
	
	\SolutionStage{5}{Probability within one standard deviation}
	
	We want the probability that the number of sandwiches sold is within one
	standard deviation of the mean:
	\[
	P(\mu-\sigma_S<S<\mu+\sigma_S)
	\]
	
	Using $\mu=50$ and $\sigma_S\approx12.25$,
	\[
	P(50-12.25<S<50+12.25)
	=
	P(37.75<S<62.25)
	\]
	
	Since the mode is $c=50$, the probability density is
	\[
	f(s)=
	\begin{cases}
		\dfrac{2(s-a)}{(b-a)(c-a)}, & a\leq s\leq c,\\[2mm]
		\dfrac{2(b-s)}{(b-a)(b-c)}, & c<s\leq b.
	\end{cases}
	\]
	
	Therefore,
	\[
	P(37.75<S<62.25)
	=
	\int_{37.75}^{50}
	\frac{2(s-20)}{(80-20)(50-20)}\,ds
	+
	\int_{50}^{62.25}
	\frac{2(80-s)}{(80-20)(80-50)}\,ds
	\]
	
	\[
	\approx0.611
	\]
	
	\[
	\boxed{P(\mu-\sigma_S<S<\mu+\sigma_S)\approx0.611}
	\]
	
	Thus, approximately $61.1\%$ of sandwich sales are within one standard
	deviation of the mean.

	\section{Exponential distribution}
	
	\SolutionStage{1}{Parameter}
	
	The average time between customers is 4 minutes. For an exponential
	distribution,
	\[
	E(X)=\frac{1}{\lambda}
		\qquad \longrightarrow \qquad
	4=\frac{1}{\lambda}
	\qquad\longrightarrow\qquad
	\lambda=\frac14
		\qquad \longrightarrow \qquad
	\boxed{\lambda=\frac14\text{ min}^{-1}}
	\]
	\SolvedGraph{Exponential density}{
		\ExponentialGraph{0.25}
	}
	
	\SolutionStage{2}{Mean}
	
	For an exponential distribution,
	\[
	E(X)=\frac{1}{\lambda}
			\qquad \longrightarrow \qquad
	E(X)
	=\frac{1}{1/4}
	=4
		\qquad \longrightarrow \qquad
	\boxed{E(X)=4\text{ minutes}}
	\]
	
	\SolutionStage{3}{Variance}
	
	For an exponential distribution,
	\[
	Var(X)=\frac{1}{\lambda^2}
			\qquad \longrightarrow \qquad
	Var(X)
	=\frac{1}{(1/4)^2}
	=16
		\qquad \longrightarrow \qquad
	\boxed{Var(X)=16\text{ minutes}^2}
	\]
	
	\SolutionStage{4}{Interpretation}

	The standard deviation is
	\[
	\sigma_X=\sqrt{16}=4
		\qquad \longrightarrow \qquad
	\boxed{\sigma_X=4\text{ minutes}}
	\]
	
	\SolutionStage{5}{Probability within one standard deviation}
	
	We want the probability that the time between customers is within one
	standard deviation of the mean:
	\[
	P(\mu-\sigma_X<X<\mu+\sigma_X)
	\]
	
	Using $\mu=4$ and $\sigma_X=4$,
	\[
	P(4-4<X<4+4)
	=
	P(0<X<8)
	\]
	
	For an exponential distribution, the cumulative distribution function is
	\[
	P(X<x)=1-e^{-\lambda x}.
	\]
	
	Therefore,
	\[
	P(0<X<8)
	=
	1-e^{-(1/4)(8)}
	=
	1-e^{-2}
	\approx0.8647
	\]
	
	\[
	\boxed{P(\mu-\sigma_X<X<\mu+\sigma_X)\approx0.865}
	\]
	
	Thus, approximately $86.5\%$ of times between customers are within one
	standard deviation of the mean.
	
	\section{Normal distribution}
	
	\SolutionStage{1}{Parameters}
	
	The travel time $Y$ follows a normal distribution with mean 45 minutes
	and standard deviation 8 minutes. Therefore,
	\[
	Y\sim N(45,8^2)
		\qquad \longrightarrow \qquad
	\mu=45\qquad \sigma=8
	\]
	
	\SolvedGraph{Normal density}{
		\NormalGraph{45}{8}
	}
	
	\SolutionStage{2}{Mean}
	
	For a normal distribution,
	\[
	E(Y)=\mu
		\qquad \longrightarrow \qquad
	\boxed{E(Y)=45\text{ minutes}}
	\]
	
	\SolutionStage{3}{Variance}
	
	For a normal distribution,
	\[
	Var(Y)=\sigma^2
		\qquad \longrightarrow \qquad
	Var(Y)=8^2=64
		\qquad \longrightarrow \qquad
	\boxed{Var(Y)=64\text{ minutes}^2}.
	\]
	
	\SolutionStage{4}{Interpretation}

	The standard deviation is
	\[
	\sigma_Y=\sqrt{64}=8
		\qquad \longrightarrow \qquad
	\boxed{\sigma_Y=8\text{ minutes}}.
	\]
	
	\SolutionStage{5}{Probability within one standard deviation}
	
	We want the probability that the travel time is within one standard
	deviation of the mean:
	\[
	P(\mu-\sigma_Y<Y<\mu+\sigma_Y)
	\]
	
	Using $\mu=45$ and $\sigma_Y=8$,
	\[
	P(45-8<Y<45+8)
	=
	P(37<Y<53)
	\]
	
	Standardizing the two endpoints,
	\[
	z_1=\frac{37-45}{8}=-1,
	\qquad
	z_2=\frac{53-45}{8}=1
	\]
	
	Therefore,
	\[
	P(37<Y<53)
	=
	P(-1<Z<1)
	\approx0.6827
	\]
	
	\[
	\boxed{P(\mu-\sigma_Y<Y<\mu+\sigma_Y)\approx0.683}
	\]
	
	Thus, approximately $68.3\%$ of travel times are within one standard
	deviation of the mean.
