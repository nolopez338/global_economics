\section*{Evaluated criterion}
\begin{enumerate}[label=\textbf{C4}]
 \item Applies the exponential distribution to constant-rate waiting-time situations and assesses its suitability.
\end{enumerate}

\section{Introduction}\label{sec:c4-introduction}
The exponential distribution models a \emph{non-negative waiting time} $T$ until the next event when events occur independently at a stable average rate. Its parameter $\lambda>0$ is the mean number of events per unit of time, so the unit used for $T$ must match the time unit in $\lambda$. We write $T\sim\operatorname{Exp}(\lambda)$, with support $t\geq0$, and
\[
 f_T(t)=\begin{cases}\lambda e^{-\lambda t},&t\geq0,\\0,&t<0,\end{cases}\qquad
 F_T(t)=P(T\leq t)=1-e^{-\lambda t},\qquad
 P(T>t)=e^{-\lambda t}.
\]
For a continuous variable, endpoint equality has probability zero. Thus $P(T<t)=P(T\leq t)$, and
\[
 P(a<T<b)=F_T(b)-F_T(a)=e^{-\lambda a}-e^{-\lambda b}.
\]
The expected waiting time is $E(T)=1/\lambda$. A larger event rate therefore means a shorter typical wait: for example, $6$ events per hour gives a mean wait of $1/6$ hour, or $10$ minutes. Rates must be converted before substitution; $12$ per hour is $12/60=0.2$ per minute.

The exponential distribution is \emph{memoryless}:
\[
 P(T>s+t\mid T>s)=P(T>t).
\]
After waiting $s$ time units without an event, the distribution of the \emph{additional} wait is the same as at the start. This is reasonable only if elapsed time does not change the event mechanism.

An exponential model is plausible when the variable is time until the next event, events are approximately independent, simultaneous events are negligible, and the average rate is reasonably constant over the period studied. It is generally inappropriate for fixed schedules, appointment systems, queues whose service speed depends on their length, rush-hour demand, aging equipment with an increasing failure risk, or any process with a changing rate.

\paragraph{General procedure.} (1) Define the waiting-time variable and its units. (2) identify and convert the rate so that $\lambda$ and time use matching units. (3) write $T\sim\operatorname{Exp}(\lambda)$ and select the CDF, survival, interval, or memoryless formula. (4) substitute, calculate, and state the probability or mean with units and context. (5) assess the model by checking waiting-time meaning, independence, rate stability, and relevant operational patterns; qualify the conclusion when assumptions are only approximate.

\section{Rate and Waiting-Time Interpretation}\label{sec:c4-rate}
\SectionProblemsTableOfContents{
 \SectionProblemLink{sec:c4-rate-cafe}{Cafeteria mobile orders}\\
 \SectionProblemLink{sec:c4-rate-fintech}{Fintech support enquiries}}

\subsection{Cafeteria mobile orders}\label{sec:c4-rate-cafe}
A Bogotá school cafeteria receives an average of 18 mobile orders per hour during a quiet mid-morning period. Let $T$ be the time from opening the order screen until the next order appears.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-rate-cafe-a}{4.1}{Explain why $T$ is a waiting time and interpret the given rate.}
 \CriterionTask{subsec:c4-rate-cafe-b}{4.2}{Express the model using minutes and find the expected wait.}
 \CriterionTask{subsec:c4-rate-cafe-c}{4.3}{State the assumptions needed for this model to be reasonable.}
\end{enumerate}
\subsubsection*{C4.1 - Identify the variable and rate}\phantomsection\label{subsec:c4-rate-cafe-a}
$T\geq0$ measures elapsed time until a clearly defined next event, an order. The rate 18 per hour means that over many comparable quiet hours the business averages 18 orders per hour; it does not mean exactly one order every $60/18$ minutes.
\secInicio[sec:c4-rate][sec:c4-rate-cafe]
\subsubsection*{C4.2 - Convert the rate and find the mean}\phantomsection\label{subsec:c4-rate-cafe-b}
\[
 \lambda=\frac{18}{60}=0.3\ \text{orders per minute},\qquad
 T\sim\operatorname{Exp}(0.3),\qquad E(T)=\frac1{0.3}=3.33\ \text{minutes}.
\]
The screen is expected to wait about 3 minutes 20 seconds for the next order.
\secInicio[sec:c4-rate][sec:c4-rate-cafe]
\subsubsection*{C4.3 - Assess the modeling assumptions}\phantomsection\label{subsec:c4-rate-cafe-c}
Orders should arrive approximately independently and at a stable rate during the selected period. A lunch rush, a timed promotion, or customers ordering together would undermine those assumptions. The model is therefore defensible for the stated quiet period only if order records support a roughly constant rate.
\secInicio[sec:c4-rate][sec:c4-rate-cafe]

\subsection{Fintech support enquiries}\label{sec:c4-rate-fintech}
A Colombian fintech receives 96 account-verification enquiries in a typical 8-hour weekday shift. An analyst initially proposes an exponential model for the time $W$ until the next enquiry.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-rate-fintech-a}{4.1}{Find the hourly and per-minute rates and write the model.}
 \CriterionTask{subsec:c4-rate-fintech-b}{4.2}{Calculate and interpret $E(W)$.}
 \CriterionTask{subsec:c4-rate-fintech-c}{4.3}{Explain why a whole-shift average may be misleading.}
\end{enumerate}
\subsubsection*{C4.1 - Build the model}\phantomsection\label{subsec:c4-rate-fintech-a}
\[
 \lambda=\frac{96}{8}=12\ \text{per hour}=\frac{12}{60}=0.2\ \text{per minute},
 \qquad W\sim\operatorname{Exp}(0.2)\quad(W\text{ in minutes}).
\]
Both forms describe the same rate; the second matches a wait measured in minutes.
\secInicio[sec:c4-rate][sec:c4-rate-fintech]
\subsubsection*{C4.2 - Interpret reciprocal rate}\phantomsection\label{subsec:c4-rate-fintech-b}
\[
 E(W)=\frac1\lambda=\frac1{0.2}=5\ \text{minutes}.
\]
Across comparable periods, the mean time from one observation point to the next enquiry is 5 minutes; individual waits vary.
\secInicio[sec:c4-rate][sec:c4-rate-fintech]
\subsubsection*{C4.3 - Examine rate stability}\phantomsection\label{subsec:c4-rate-fintech-c}
The calculation spreads 96 enquiries uniformly across the shift. If demand surges after salary payments or at lunchtime, one constant $\lambda$ is unsuitable. The analyst should inspect rates by time block and model only intervals whose rates are approximately stable.
\secInicio[sec:c4-rate][sec:c4-rate-fintech]

\section{Waiting-Time Probabilities}\label{sec:c4-probabilities}
\SectionProblemsTableOfContents{
 \SectionProblemLink{sec:c4-prob-delivery}{Dispatch requests}\\
 \SectionProblemLink{sec:c4-prob-atm}{ATM fault alerts}}

\subsection{Dispatch requests}\label{sec:c4-prob-delivery}
A Chapinero bicycle courier receives independent dispatch requests at a stable rate of 4 per hour during a specified off-peak hour. Let $T$ be the wait in minutes for the next request.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-prob-delivery-a}{4.1}{Find the probability of waiting less than 10 minutes.}
 \CriterionTask{subsec:c4-prob-delivery-b}{4.2}{Find the probability of waiting more than 20 minutes.}
 \CriterionTask{subsec:c4-prob-delivery-c}{4.3}{Find and interpret the probability of waiting between 10 and 20 minutes.}
\end{enumerate}
\subsubsection*{C4.1 - Wait less than 10 minutes}\phantomsection\label{subsec:c4-prob-delivery-a}
The per-minute rate is $\lambda=4/60=1/15$. Therefore
\[
 P(T<10)=F(10)=1-e^{-(1/15)(10)}=1-e^{-2/3}\approx0.4866.
\]
There is about a $48.7\%$ chance that the next dispatch arrives within 10 minutes.
\secInicio[sec:c4-probabilities][sec:c4-prob-delivery]
\subsubsection*{C4.2 - Wait more than 20 minutes}\phantomsection\label{subsec:c4-prob-delivery-b}
Using the survival function,
\[
 P(T>20)=e^{-(1/15)(20)}=e^{-4/3}\approx0.2636.
\]
Thus roughly $26.4\%$ of comparable off-peak waits exceed 20 minutes under the model.
\secInicio[sec:c4-probabilities][sec:c4-prob-delivery]
\subsubsection*{C4.3 - Wait between 10 and 20 minutes}\phantomsection\label{subsec:c4-prob-delivery-c}
\[
 P(10<T<20)=F(20)-F(10)=e^{-2/3}-e^{-4/3}
 \approx0.5134-0.2636=0.2498.
\]
The courier has about a $25.0\%$ chance of receiving the next request after 10 but before 20 minutes. This conclusion relies on the stated off-peak rate being stable.
\secInicio[sec:c4-probabilities][sec:c4-prob-delivery]

\subsection{ATM fault alerts}\label{sec:c4-prob-atm}
A bank operations desk models the waiting time $H$ until the next ATM fault alert as exponential with rate $0.5$ per hour during overnight monitoring.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-prob-atm-a}{4.1}{Calculate $P(H\leq1)$ using a complement.}
 \CriterionTask{subsec:c4-prob-atm-b}{4.2}{Calculate $P(1<H<3)$.}
 \CriterionTask{subsec:c4-prob-atm-c}{4.3}{Interpret the interval probability and identify one operational use.}
\end{enumerate}
\subsubsection*{C4.1 - Alert within one hour}\phantomsection\label{subsec:c4-prob-atm-a}
\[
 P(H\leq1)=1-P(H>1)=1-e^{-0.5(1)}\approx1-0.6065=0.3935.
\]
The modeled chance of at least one next alert within an hour is about $39.3\%$.
\secInicio[sec:c4-probabilities][sec:c4-prob-atm]
\subsubsection*{C4.2 - Alert between one and three hours}\phantomsection\label{subsec:c4-prob-atm-b}
\[
 P(1<H<3)=e^{-0.5(1)}-e^{-0.5(3)}
 \approx0.6065-0.2231=0.3834.
\]
\secInicio[sec:c4-probabilities][sec:c4-prob-atm]
\subsubsection*{C4.3 - Interpret and use the result}\phantomsection\label{subsec:c4-prob-atm-c}
About $38.3\%$ of overnight monitoring periods have their next alert between one and three hours from now, according to the model. A manager could combine this probability with response capacity when scheduling on-call coverage, but should first verify that maintenance cycles do not make the fault rate change overnight.
\secInicio[sec:c4-probabilities][sec:c4-prob-atm]

\section{Conditional Waiting Times and Memorylessness}\label{sec:c4-memory}
\SectionProblemsTableOfContents{
 \SectionProblemLink{sec:c4-memory-orders}{Online marketplace orders}\\
 \SectionProblemLink{sec:c4-memory-calls}{Tourism enquiries}}

\subsection{Online marketplace orders}\label{sec:c4-memory-orders}
A Bogotá craft seller's off-peak order wait $T$ is modeled by $\operatorname{Exp}(0.1)$, with time in minutes. No order has arrived in the first 8 minutes.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-memory-orders-a}{4.1}{Use conditional probability to find the chance of waiting more than 5 additional minutes.}
 \CriterionTask{subsec:c4-memory-orders-b}{4.2}{Obtain the same result using memorylessness.}
 \CriterionTask{subsec:c4-memory-orders-c}{4.3}{Explain the result in context and state its limitation.}
\end{enumerate}
\subsubsection*{C4.1 - Use conditional probability}\phantomsection\label{subsec:c4-memory-orders-a}
More than 5 additional minutes means $T>13$, given $T>8$. Since $\{T>13\}\subset\{T>8\}$,
\[
 P(T>13\mid T>8)=\frac{P(T>13)}{P(T>8)}
 =\frac{e^{-0.1(13)}}{e^{-0.1(8)}}=e^{-0.5}\approx0.6065.
\]
\secInicio[sec:c4-memory][sec:c4-memory-orders]
\subsubsection*{C4.2 - Use memorylessness}\phantomsection\label{subsec:c4-memory-orders-b}
Directly,
\[
 P(T>8+5\mid T>8)=P(T>5)=e^{-0.1(5)}=e^{-0.5}\approx0.6065,
\]
which agrees with the ratio calculation.
\secInicio[sec:c4-memory][sec:c4-memory-orders]
\subsubsection*{C4.3 - Interpret memorylessness}\phantomsection\label{subsec:c4-memory-orders-c}
Given no order in 8 minutes, the chance of waiting more than 5 \emph{further} minutes is still about $60.7\%$; the elapsed wait does not make an order ``due.'' This interpretation is credible only while demand stays constant. Approaching a promotion launch would change the rate and invalidate memorylessness.
\secInicio[sec:c4-memory][sec:c4-memory-orders]

\subsection{Tourism enquiries}\label{sec:c4-memory-calls}
A hotel receives independent website enquiries at a stable rate of 3 per hour in the late evening. There has been no enquiry for 20 minutes.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-memory-calls-a}{4.1}{Find the probability that the total wait exceeds 50 minutes.}
 \CriterionTask{subsec:c4-memory-calls-b}{4.2}{Find the probability that an enquiry arrives in the next 15 minutes.}
 \CriterionTask{subsec:c4-memory-calls-c}{4.3}{Compare the two event descriptions and explain what elapsed time changes.}
\end{enumerate}
\subsubsection*{C4.1 - Total wait exceeds 50 minutes}\phantomsection\label{subsec:c4-memory-calls-a}
The additional wait required is $50-20=30$ minutes $=0.5$ hour. Hence
\[
 P(T>50\text{ min}\mid T>20\text{ min})=P(T>0.5\text{ h})=e^{-3(0.5)}\approx0.2231.
\]
\secInicio[sec:c4-memory][sec:c4-memory-calls]
\subsubsection*{C4.2 - Enquiry within the next 15 minutes}\phantomsection\label{subsec:c4-memory-calls-b}
Fifteen minutes is $0.25$ hour, so by memorylessness and a complement,
\[
 P(\text{arrival in next }15\text{ min}\mid T>20\text{ min})
 =1-e^{-3(0.25)}\approx0.5276.
\]
There is about a $52.8\%$ modeled chance of an enquiry in the next quarter-hour.
\secInicio[sec:c4-memory][sec:c4-memory-calls]
\subsubsection*{C4.3 - Explain the role of elapsed time}\phantomsection\label{subsec:c4-memory-calls-c}
The first event asks for more than 30 additional minutes; the second asks for no more than 15 additional minutes, so their probabilities need not complement each other. The 20 elapsed minutes changes the total clock time but, under the exponential model, not the distribution of the remaining wait.
\secInicio[sec:c4-memory][sec:c4-memory-calls]

\section{Assessing Model Suitability}\label{sec:c4-suitability}
\SectionProblemsTableOfContents{
 \SectionProblemLink{sec:c4-suitable-payments}{Payment confirmations}\\
 \SectionProblemLink{sec:c4-suitable-operations}{Schedules, queues, and failures}}

\subsection{Payment confirmations}\label{sec:c4-suitable-payments}
A small e-commerce firm observes payment confirmations during a stable overnight period. Confirmations are individual customer actions, and hourly counts fluctuate around the same mean with no visible trend.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-suitable-payments-a}{4.1}{Assess whether an exponential model for the next-confirmation wait is reasonable.}
 \CriterionTask{subsec:c4-suitable-payments-b}{4.2}{Explain what additional evidence should be checked.}
\end{enumerate}
\subsubsection*{C4.1 - Make a justified assessment}\phantomsection\label{subsec:c4-suitable-payments-a}
The variable is a non-negative time until an event, and the description supports an approximately stable rate. Independent customer actions also support independent arrivals. An exponential model is therefore reasonable as an approximation for this restricted overnight period, not automatically for the full day.
\secInicio[sec:c4-suitability][sec:c4-suitable-payments]
\subsubsection*{C4.2 - Seek diagnostic evidence}\phantomsection\label{subsec:c4-suitable-payments-b}
The firm should compare rates across shorter time blocks, look for bursts caused by campaigns or batch processing, and check whether a recent confirmation changes the chance of another. Strong time trends, clustering, or fixed batch releases would contradict constant-rate independent arrivals.
\secInicio[sec:c4-suitability][sec:c4-suitable-payments]

\subsection{Schedules, queues, and failures}\label{sec:c4-suitable-operations}
A manager proposes the same exponential model for (i) time until a TransMilenio service advertised every 10 minutes, (ii) a customer's wait in a bank queue, and (iii) time until failure of new electronic scanners operating under unchanged conditions.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-suitable-operations-a}{4.1}{Assess cases (i) and (ii).}
 \CriterionTask{subsec:c4-suitable-operations-b}{4.2}{Assess case (iii) and explain how aging changes the conclusion.}
 \CriterionTask{subsec:c4-suitable-operations-c}{4.3}{Summarize the assumption violated in each unsuitable case.}
\end{enumerate}
\subsubsection*{C4.1 - Scheduled service and a queue}\phantomsection\label{subsec:c4-suitable-operations-a}
Case (i) is unsuitable: a timetable concentrates arrivals near scheduled times rather than producing constant-rate random arrivals. Case (ii) is usually unsuitable: the wait depends on queue length, number of tellers, and service durations, all information that changes the remaining wait.
\secInicio[sec:c4-suitability][sec:c4-suitable-operations]
\subsubsection*{C4.2 - Equipment failures}\phantomsection\label{subsec:c4-suitable-operations-b}
Case (iii) can be suitable if each scanner has an approximately constant failure hazard and failures are independent. If wear raises the hazard as a scanner ages, the remaining lifetime depends on age, contradicting memorylessness; an exponential lifetime model would then be inappropriate.
\secInicio[sec:c4-suitability][sec:c4-suitable-operations]
\subsubsection*{C4.3 - Identify the contradictions}\phantomsection\label{subsec:c4-suitable-operations-c}
The schedule violates random constant-rate occurrence; the queue makes the wait state-dependent; aging makes failure risk time-dependent. Thus merely observing a positive waiting time is not enough: independence and a stable event rate or hazard must also be defensible.
\secInicio[sec:c4-suitability][sec:c4-suitable-operations]

\subsection{Logistics staffing decision}\label{sec:c4-applied-logistics}
A Bogotá flower exporter receives independent urgent pickup requests at a stable off-peak rate of 2.5 per hour. Keeping a dispatcher continuously at the desk is costly, but a request unanswered for more than 30 minutes may lose a sale. Let $T$ be the time to the next request.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-applied-logistics-a}{4.1}{Write the model and find the expected wait.}
 \CriterionTask{subsec:c4-applied-logistics-b}{4.2}{Find the probability that the next request waits more than 30 minutes to appear.}
 \CriterionTask{subsec:c4-applied-logistics-c}{4.3}{Interpret the result for staffing and assess whether it may be used during Valentine's-week demand.}
\end{enumerate}
\subsubsection*{C4.1 - Model the request wait}\phantomsection\label{subsec:c4-applied-logistics-a}
\[
 T\sim\operatorname{Exp}(2.5)\quad(T\text{ in hours}),\qquad
 E(T)=\frac1{2.5}=0.4\text{ h}=24\text{ minutes}.
\]
The mean off-peak time to the next urgent request is 24 minutes.
\secInicio[sec:c4-applied][sec:c4-applied-logistics]
\subsubsection*{C4.2 - Calculate the business-risk probability}\phantomsection\label{subsec:c4-applied-logistics-b}
Thirty minutes is $0.5$ hour. Therefore
\[
 P(T>0.5)=e^{-2.5(0.5)}=e^{-1.25}\approx0.2865.
\]
About $28.7\%$ of modeled off-peak waits exceed 30 minutes, information the exporter can weigh against staffing cost and the value of a lost sale.
\secInicio[sec:c4-applied][sec:c4-applied-logistics]
\subsubsection*{C4.3 - Use and qualify the result}\phantomsection\label{subsec:c4-applied-logistics-c}
The probability is substantial, but it alone does not prescribe staffing: the manager also needs loss amounts and response rules. It should not be transferred unchanged to Valentine's week, when promotions and deadlines create a higher, changing rate and possibly clustered requests. A time-varying or segmented model is preferable then.
\secInicio[sec:c4-applied][sec:c4-applied-logistics]

\subsection{Financial-operations alert decision}\label{sec:c4-applied-finance}
During a stable four-hour overnight window, a Colombian digital wallet receives suspicious-transaction alerts at an average rate of 1.2 per hour. An analyst has already waited 45 minutes without an alert. A staffing decision depends on the chance of an alert in the next hour.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c4-applied-finance-a}{4.1}{Calculate that conditional probability.}
 \CriterionTask{subsec:c4-applied-finance-b}{4.2}{Find the probability that the total wait is between 1 and 2 hours, given the first 45 minutes were alert-free.}
 \CriterionTask{subsec:c4-applied-finance-c}{4.3}{Interpret both results and assess the model under a fraud wave.}
\end{enumerate}
\subsubsection*{C4.1 - Alert in the next hour}\phantomsection\label{subsec:c4-applied-finance-a}
By memorylessness, the elapsed 0.75 hour does not alter the next-hour probability:
\[
 P(T\leq1.75\mid T>0.75)=1-P(T>1\text{ additional hour})
 =1-e^{-1.2(1)}\approx0.6988.
\]
There is about a $69.9\%$ modeled chance of at least one next alert within the coming hour.
\secInicio[sec:c4-applied][sec:c4-applied-finance]
\subsubsection*{C4.2 - Conditional interval probability}\phantomsection\label{subsec:c4-applied-finance-b}
Given survival to 0.75 hour, total times from 1 to 2 hours correspond to additional waits from 0.25 to 1.25 hours. Thus
\[
 P(1<T<2\mid T>0.75)=e^{-1.2(0.25)}-e^{-1.2(1.25)}
 =e^{-0.3}-e^{-1.5}\approx0.7408-0.2231=0.5177.
\]
\secInicio[sec:c4-applied][sec:c4-applied-finance]
\subsubsection*{C4.3 - Advise with assumptions}\phantomsection\label{subsec:c4-applied-finance-c}
The next-hour probability helps judge whether on-call coverage is likely to be needed; the interval result says about $51.8\%$ of modeled cases place the next alert between 1 and 2 hours after monitoring began, conditional on the observed 45 alert-free minutes. During a coordinated fraud wave, alerts may cluster and the rate may rise, violating independence and stability. The wallet should then update the model rather than rely on memorylessness.
\secInicio[sec:c4-applied][sec:c4-applied-finance]
