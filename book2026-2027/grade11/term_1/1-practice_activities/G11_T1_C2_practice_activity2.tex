For every problem, use the same seven-step routine: (1) identify the \textbf{support};
(2) name the density's shape; (3) use total area $=1$ to find any unknown
height or parameter; (4) mark the requested interval; (5) split its region
into familiar figures; (6) calculate and add or subtract their areas; and
(7) state the probability in context.  A density is a height, so it may exceed
$1$; it is the \emph{area} that must total $1$.  Endpoints do not affect a
continuous probability, for example $P(a<X<b)=P(a\le X\le b)$.

\subsection{Uniform distribution}\label{sec:c2-intro-uniform}
A uniform density has one constant height on support $[L,U]$:
\[
 f_X(x)=\begin{cases}h,&L\le x\le U,\\0,&\text{otherwise.}\end{cases}
 \qquad (U-L)h=1,\quad h=\frac1{U-L}.
\]
Thus every probability is a rectangle: $P(a\le X\le b)=(b-a)h$ for
$L\le a<b\le U$.

\textbf{Worked example.} A delivery time $T$ is equally plausible from $2$ to
$8$ minutes.  The support is $[2,8]$ and the whole rectangle has width $6$ and
height $h$, so $6h=1$ and $h=1/6$.  For $3\le T\le6$, the relevant rectangle
has width $6-3=3$ and height $1/6$:
\[
 P(3\le T\le6)=A_{\rm rectangle}=3\left(\frac16\right)=\frac12.
\]
There is a $50\%$ chance that delivery takes from $3$ to $6$ minutes.
\CtwoProbGraph{2}{8}{1/6}{3}{6}{2,3,6,8}{0.22}
\secInicio[sec:c2-introduction][sec:c2-intro-uniform]

\subsection{Linear distribution}\label{sec:c2-intro-linear}
A single linear density has $f_X(x)=mx+d$ throughout one support $[L,U]$
(and $0$ outside), with nonnegative endpoint heights
$h_L=f(L)$ and $h_U=f(U)$.  Its whole area is the trapezoid
\[
 1=\frac12(h_L+h_U)(U-L).
\]
An interval $[a,b]$ is also a trapezoid of width $b-a$ and parallel heights
$f(a),f(b)$.  A triangle is only the special case in which one endpoint height
is zero.

\textbf{Worked example.} The density of a shop's daily price adjustment $X$
(in dollars) is $f(x)=kx+0.1$ on $[0,4]$.  The support width is $4$; its endpoint
heights are $0.1$ and $4k+0.1$.  Normalisation gives
\[
 1=\tfrac12[0.1+(4k+0.1)]4=8k+0.4,
 \qquad k=0.075.
\]
For $1\le X\le3$, $f(1)=0.175$ and $f(3)=0.325$.  The required figure is a
trapezoid of width $2$ with those parallel heights:
\[
 P(1\le X\le3)=\tfrac12(0.175+0.325)(2)=0.5.
\]
Both support endpoints have positive density, so this is genuinely linear,
not triangular.
\COneGraph{0}{4}{0.075*x+0.1}{1}{3}{0,1,3,4}{0.45}
\secInicio[sec:c2-introduction][sec:c2-intro-linear]

\subsection{Triangular distribution}\label{sec:c2-intro-triangular}
A triangular density has support $[L,U]$, height $0$ at both ends, and mode
$M$ ($L<M<U$) with height $H$.  It rises linearly on $[L,M]$ and falls linearly
on $[M,U]$.  Since the whole figure has base $U-L$,
\[
 1=\tfrac12(U-L)H,\qquad H=\frac2{U-L}.
\]
Heights inside either sloping side are found by proportional similar triangles.

\textbf{Worked example.} A project's cost overrun $C$ (thousand dollars) has
minimum $0$, most likely value $2$, and maximum $6$.  Its support is $[0,6]$,
mode is $2$, and $\tfrac12(6)H=1$, so $H=1/3$.  To find
$P(1\le C\le4)$, subtract two tail triangles.  At $C=1$ the height is $1/6$;
at $C=4$ it is also $1/6$.  The left tail has base $1$, height $1/6$; the
right tail has base $2$, height $1/6$:
\[
 P(1\le C\le4)=1-\tfrac12(1)(\tfrac16)-\tfrac12(2)(\tfrac16)
 =1-\tfrac1{12}-\tfrac16=\tfrac34.
\]
The region crosses the mode, but subtraction avoids splitting it into two
trapezoids.
\COneGraph{0}{6}{(x<=2)*(x/6)+(x>2)*((6-x)/12)}{1}{4}{0,1,2,4,6}{0.4}
\secInicio[sec:c2-introduction][sec:c2-intro-triangular]

\subsection{Piecewise uniform distribution}\label{sec:c2-intro-piecewise-uniform}
Here the density is constant \emph{within} each interval but may jump at a
breakpoint.  Normalisation is the sum of rectangles:
$1=\sum(\text{segment width})(\text{segment height})$.

\textbf{Worked example.} A warehouse waiting time $W$ has
\[
 f_W(w)=\begin{cases}k,&0\le w<2,\\2k,&2\le w\le5,\\0,&\text{otherwise.}\end{cases}
\]
The support is $[0,5]$.  A width-$2$, height-$k$ rectangle plus a width-$3$,
height-$2k$ rectangle must have area $1$: $2k+6k=1$, so $k=1/8$.
The event $1\le W\le4$ spans the jump and uses two rectangles:
\[
 P(1\le W\le4)=\underbrace{(2-1)(1/8)}_{A_1}
 +\underbrace{(4-2)(1/4)}_{A_2}=\frac18+\frac12=\frac58.
\]
\CtwoProbGraph{0}{5}{(x<2)*(1/8)+(x>=2)*(1/4)}{1}{4}{0,1,2,4,5}{0.3}
\secInicio[sec:c2-introduction][sec:c2-intro-piecewise-uniform]

\subsection{Piecewise linear distribution}\label{sec:c2-intro-piecewise-linear}
A piecewise-linear density joins straight segments at stated breakpoints.
Read the height at each endpoint and breakpoint, then treat each piece as a
triangle, rectangle, or trapezoid.  Continuity means adjacent pieces meet at
the same height.

\textbf{Worked example.} Demand $D$ has support $[0,6]$.  Its density joins
$(0,0)$, $(2,H)$, $(4,H)$, and $(6,0)$.  The whole shape is a left triangle
(base $2$), rectangle (width $2$), and right triangle (base $2$):
\[
 1=\tfrac12(2)H+(2)H+\tfrac12(2)H=4H,
 \qquad H=\tfrac14.
\]
For $1\le D\le5$, subtract the two congruent tail triangles.  By linearity,
the heights at $1$ and $5$ are $H/2=1/8$:
\[
 P(1\le D\le5)=1-\tfrac12(1)(\tfrac18)-\tfrac12(1)(\tfrac18)=\frac78.
\]
This reusable approach identifies all three geometric pieces and uses
subtraction from total area $1$.
\COneGraph{0}{6}{(x<=2)*(x/8)+(x>2)*(x<=4)*(1/4)+(x>4)*((6-x)/8)}{1}{5}{0,1,2,4,5,6}{0.3}
\secInicio[sec:c2-introduction][sec:c2-intro-piecewise-linear]

\subsection{Decreasing delay}\label{sec:c2-linear-decreasing}
A dispatch delay $D$ has density $f(d)=k(6-d)$ on $[0,6]$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c2-linear-decreasing-a}{2}{Find $k$ from the whole triangular area. \score{3}}
 \CriterionTask{subsec:c2-linear-decreasing-b}{2}{Find $P(D\le2)$ geometrically. \score{3}}
 \CriterionTask{subsec:c2-linear-decreasing-c}{2}{Use a complement to find $P(D>2)$ and interpret it. \score{2}}

\subsection{Trapezoidal returns}\label{sec:c2-linear-trapezoid}
The density of a return index $R$ is one line on $[0,5]$, with
$f(0)=0.1$ and $f(5)=h>0$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c2-linear-trapezoid-a}{2}{Determine $h$ by normalisation, then write the linear density. \score{4}}
 \CriterionTask{subsec:c2-linear-trapezoid-b}{2}{Calculate $P(1\le R\le4)$ as a trapezoid. \score{3}}
 \CriterionTask{subsec:c2-linear-trapezoid-c}{2}{Calculate $P(R<1\text{ or }R>4)$ by a complement. \score{2}}
\end{enumerate}
\textbf{Answers.} $1=\tfrac12(0.1+h)5$ gives $h=0.3$; the slope is
$0.2/5=0.04$, so $f(r)=0.1+0.04r$.  Heights at $1,4$ are $0.14,0.26$;
the width-$3$ trapezoid has area $\tfrac12(0.14+0.26)3=0.6$.
The two tails total $1-0.6=0.4$.  Both endpoint heights are positive, so the
support region is not a triangle.
\secInicio[sec:c2-linear-practice][sec:c2-linear-trapezoid]

\subsection{Symmetric sales}\label{sec:c2-triangular-symmetric}
Sales $S$ have triangular support $[0,8]$ and mode $4$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c2-triangular-symmetric-a}{2}{Find the modal height. \score{2}}
 \CriterionTask{subsec:c2-triangular-symmetric-b}{2}{Find $P(2\le S\le6)$ by subtracting tail triangles. \score{3}}
 \CriterionTask{subsec:c2-triangular-symmetric-c}{2}{Compare $P(S<2)$ and $P(S>6)$. \score{2}}
\end{enumerate}
\textbf{Answers.} $H=2/8=1/4$.  Heights at $2,6$ are $1/8$.
Each tail area is $\tfrac12(2)(1/8)=1/8$, so the middle probability is
$1-1/8-1/8=3/4$.  Symmetry makes the two tail probabilities equal.
\secInicio[sec:c2-triangular-practice][sec:c2-triangular-symmetric]

\subsection{Asymmetric project time}\label{sec:c2-triangular-asymmetric}
Completion time $T$ is triangular on $[1,7]$ with mode $3$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c2-triangular-asymmetric-a}{2}{Find the modal height and the heights at $2$ and $5$. \score{3}}
 \CriterionTask{subsec:c2-triangular-asymmetric-b}{2}{Find $P(2\le T\le5)$ using two trapezoids split at the mode. \score{4}}
 \CriterionTask{subsec:c2-triangular-asymmetric-c}{2}{Interpret the probability in context. \score{1}}
\end{enumerate}
\textbf{Answers.} $H=2/6=1/3$.  Linear proportions give
$f(2)=1/6$ and $f(5)=1/6$.  Therefore
\[
 P(2\le T\le5)=\underbrace{\tfrac12(1)(1/6+1/3)}_{A_1=1/4}
 +\underbrace{\tfrac12(2)(1/3+1/6)}_{A_2=1/2}=\frac34.
\]
There is a $75\%$ chance of completion between two and five time units.
\secInicio[sec:c2-triangular-practice][sec:c2-triangular-asymmetric]

\subsection{Inventory tail}\label{sec:c2-triangular-tail}
Inventory demand $Q$ has triangular support $[0,10]$, mode $2$, and unknown
height $H$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c2-triangular-tail-a}{2}{Determine $H$. \score{2}}
 \CriterionTask{subsec:c2-triangular-tail-b}{2}{Find $P(Q\ge6)$ from the right tail triangle. \score{3}}
 \CriterionTask{subsec:c2-triangular-tail-c}{2}{Find $P(1\le Q\le6)$ by subtracting both outside tails. \score{4}}
\end{enumerate}
\textbf{Answers.} $H=2/10=1/5$.  On the descending side
$f(6)=H(10-6)/(10-2)=1/10$, so $P(Q\ge6)=\tfrac12(4)(1/10)=1/5$.
Also $f(1)=1/10$, making the left excluded triangle area
$\tfrac12(1)(1/10)=1/20$.  Thus $P(1\le Q\le6)=1-1/20-1/5=3/4$.
\secInicio[sec:c2-triangular-practice][sec:c2-triangular-tail]

\subsection{Order processing}\label{sec:c2-piecewise-uniform-orders}

\subsection{Demand peak}\label{sec:c2-piecewise-linear-demand}
A continuous density joins $(0,0)$, $(2,H)$, $(5,H/2)$, and $(6,0)$.
It models a demand index $D$ on $[0,6]$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c2-piecewise-linear-demand-a}{2}{Find $H$ by decomposing the support into one triangle and two trapezoids. \score{4}}
 \CriterionTask{subsec:c2-piecewise-linear-demand-b}{2}{Find $P(1\le D\le5)$ by subtracting the two tail triangles. \score{4}}
 \CriterionTask{subsec:c2-piecewise-linear-demand-c}{2}{Interpret the result. \score{1}}
\end{enumerate}
\textbf{Answers.} The areas are $H$, $\tfrac12(H+H/2)3=9H/4$,
and $\tfrac12(1)(H/2)=H/4$.  Thus $7H/2=1$ and $H=2/7$.
At $x=1$, the height is $H/2=1/7$; the left excluded area is $1/14$.
The right excluded area from $5$ to $6$ is also $1/14$.  Therefore
$P(1\le D\le5)=1-1/14-1/14=6/7$, so demand lies in that planning range about
$85.7\%$ of the time.

\subsection{Delivery profile}\label{sec:c2-piecewise-linear-delivery}
A delivery time density is continuous through the points
$(0,0)$, $(2,H)$, $(4,H)$, $(6,H/2)$, and $(8,0)$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c2-piecewise-linear-delivery-a}{2}{Find $H$ using the triangle, rectangle, and trapezoids across the support. \score{4}}
 \CriterionTask{subsec:c2-piecewise-linear-delivery-b}{2}{Find $P(1\le T\le7)$ using a complement. \score{4}}
 \CriterionTask{subsec:c2-piecewise-linear-delivery-c}{2}{Find $P(3\le T\le6)$ by a rectangle plus a trapezoid. \score{3}}
\end{enumerate}
\textbf{Answers.} Successive areas are $H$, $2H$, $3H/2$, and $H/2$;
hence $5H=1$ and $H=1/5$.  At $1$, the height is $H/2=1/10$, so the left
tail area is $1/20$.  At $7$, the descending height is $H/4=1/20$, so the
right tail area is $1/40$.  Thus $P(1\le T\le7)=1-1/20-1/40=37/40$.
From $3$ to $4$ is a width-$1$, height-$1/5$ rectangle; from $4$ to $6$ is a
width-$2$ trapezoid with heights $1/5,1/10$:
\[
 P(3\le T\le6)=\underbrace{1(1/5)}_{A_1}
 +\underbrace{\tfrac12(2)(1/5+1/10)}_{A_2}=\frac12.
\]
\secInicio[sec:c2-piecewise-linear-practice][sec:c2-piecewise-linear-delivery]

\subsection{Liquidity window}\label{sec:c2-piecewise-linear-liquidity}
A liquidity index $X$ has a continuous density joining
$(0,0)$, $(1,H)$, $(3,H)$, $(5,0)$, and $(7,H)$ on support $[0,7]$.
\textbf{Questions/Tasks.}
\begin{enumerate}
 \CriterionTask{subsec:c2-piecewise-linear-liquidity-a}{2}{Find $H$ from all four geometric pieces. \score{4}}
 \CriterionTask{subsec:c2-piecewise-linear-liquidity-b}{2}{Find $P(0.5\le X\le4)$, splitting at $1$ and $3$. \score{5}}
 \CriterionTask{subsec:c2-piecewise-linear-liquidity-c}{2}{Find $P(X>4)$ by a complement and identify the pieces avoided. \score{4}}
\end{enumerate}
\textbf{Answers.} The support comprises a base-$1$ triangle, width-$2$
rectangle, base-$2$ triangle, and base-$2$ triangle.  Hence
$\tfrac12H+2H+H+H=\tfrac92H=1$, so $H=2/9$.  At $0.5$, the height is
$H/2=1/9$; at $4$, it is also $H/2=1/9$.  Therefore
\[
 P(0.5\le X\le4)=
 \underbrace{\tfrac12(0.5)(1/9+2/9)}_{A_1=1/12}
 +\underbrace{2(2/9)}_{A_2=4/9}
 +\underbrace{\tfrac12(1)(2/9+1/9)}_{A_3=1/6}
 =\frac{25}{36}.
\]
The complement is $P(X>4)=1-P(X\le4)$.  From $0$ to $4$, the pieces are the
whole first triangle ($1/9$), rectangle ($4/9$), and the $3$--$4$ trapezoid
($1/6$), totalling $13/18$; thus $P(X>4)=5/18$.  This avoids separately adding
the $4$--$5$ small triangle and the $5$--$7$ rising triangle.
