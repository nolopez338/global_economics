	\begin{itemize}
		

		\item \textbf{Customer Spending.}
		A company tracks the amount spent by customers during a typical
		shopping visit. Based on its historical data, the spending amount is
		approximately centered at \$75, with a standard deviation of \$18.
		\textbf{Task:} Find the mean, variance, and standard deviation of the amount spent by a randomly selected customer,
		measured in dollars.
		
		\item \textbf{Construction Repair Time.}
		A construction company estimates the time required to complete a small
		repair. Based on previous projects, the completion time ranges from
		6 to 18 hours. Most repairs are completed in about 10 hours, while
		shorter and longer completion times are less common. Therefore, the
		completion time has a triangular probability density with a mode at
		10 hours.
		\textbf{Task:} Find the mean, variance, and standard deviation of the completion time of a randomly selected repair,
		measured in hours.
		
		\item \textbf{Package Delivery Time.}
		A delivery company transports packages within a city. Based on previous
		deliveries, the delivery time for a randomly selected package ranges
		from 15 to 75 minutes. Most packages are delivered relatively quickly,
		so shorter delivery times are more likely and the probability density
		decreases linearly as the delivery time increases. At the longest
		delivery times, the probability can be considered negligible.
		\textbf{Task:} Find the mean, variance, and standard deviation of the delivery time for a randomly selected package,
		measured in minutes.
		
		\item \textbf{Online Orders.}
		A supermarket receives online orders throughout the day. Based on its
		recent activity, the supermarket receives an average of 12 orders per
		hour.
		\textbf{Task:} Find the mean, variance, and standard deviation of the time between consecutive orders,
		measured in minutes.
		
		\item \textbf{Coffee Shop Waiting Time.}
		A coffee shop records the time customers wait for their orders during
		the morning. Based on previous observations, the waiting time ranges
		from 3 to 15 minutes, with all values in this range considered equally
		likely.
		\textbf{Task:} Find the mean, variance, and standard deviation of the waiting time for a randomly selected customer,
		measured in minutes.
		
		\item \textbf{Furniture Production Time.}
		A manufacturer produces customized furniture. Based on previous orders,
		the time required to complete a randomly selected piece ranges from
		4 to 12 hours. More complex pieces are more common, so longer
		production times are more likely and the probability density increases
		linearly across this range. At the shortest production times, the
		probability can be considered negligible.
		\textbf{Task:} Find the mean, variance, and standard deviation of the production time for a randomly selected piece,
		measured in hours.

		
	\end{itemize}

	\subsection*{C4.1 - Finding the mean, variance, and standard deviation}\phantomsection
	\label{subsec:c4-solution-customer-spending}
	
	The amount spent by a randomly selected customer is approximately
	normally distributed, with mean \$75 and standard deviation \$18.
	\[
	\begin{array}{c|c}
		\textbf{General}
		&
		X\sim N(\mu,\sigma^2)
		\\[6pt]
		\textbf{Context}
		&
		\mu=75,\quad \sigma=18
	\end{array}
	\]
	
	Therefore,
	\[
	E(X)
	=
	\mu
	=
	75
	\]
	\[
	Var(X)
	=
	\sigma^2
	=
	18^2
	=
	324
	\]
	\[
	\sigma_X
	=
	\sigma
	=
	18
	\]
	\[
	\boxed{
		E(X)=75\qquad
		Var(X)=324\qquad
		\sigma_X=18
	}
	\]

	\subsection*{C4.2 - Finding the mean, variance, and standard deviation}\phantomsection
	\label{subsec:c4-solution-repair-time}
	
	The completion time ranges from 6 to 18 hours, with a mode at
	10 hours. Therefore, the triangular distribution has
	\[
	\begin{array}{c|c}
		\textbf{General}
		&
		X\sim\operatorname{Tri}(a,b,c)
		\\[6pt]
		\textbf{Context}
		&
		a=6,\quad b=18,\quad c=10
	\end{array}
	\]
	
	Therefore,
	\[
	E(T)
	=
	\frac{a+b+c}{3}
	=
	\frac{6+18+10}{3}
	=
	\frac{34}{3}
	\approx11.333
	\]
	\[
	Var(T)
	=
	\frac{a^2+b^2+c^2-ab-ac-bc}{18}
	\]
	\[
	\begin{aligned}
		&=
		\frac{
			6^2+18^2+10^2-(6)(18)-(6)(10)-(18)(10)
		}{18} \\[2pt]
		&=
		\frac{112}{18}
		=
		\frac{56}{9}
		\approx 6.222
	\end{aligned}
	\]
	\[
	\sigma_T
	=
	\sqrt{Var(T)}
	=
	\sqrt{\frac{56}{9}}
	\approx2.494
	\]
	\[
	\boxed{
		E(T)\approx11.333,\qquad
		Var(T)\approx6.222,\qquad
		\sigma_T\approx2.494
	}
	\]

	\subsection*{C4.3 - Finding the mean, variance, and standard deviation}\phantomsection
	\label{subsec:c4-solution-delivery-time}
	
	The delivery time ranges from 15 to 75 minutes, with shorter delivery
	times more likely and the probability density decreasing linearly as
	the delivery time increases.
	\[
	\begin{array}{c|c}
		\textbf{General}
		&
		X\sim\operatorname{Lin}_{\mathrm{dec}}(a,b)
		\\[6pt]
		\textbf{Context}
		&
		a=15,\quad b=75
	\end{array}
	\]
	
	Therefore,
	\[
	E(T)
	=
	\frac{2a+b}{3}
	=
	\frac{2(15)+75}{3}
	=
	35
	\]
	\[
	Var(T)
	=
	\frac{(b-a)^2}{18}
	=
	\frac{(75-15)^2}{18}
	=
	\frac{3600}{18}
	=
	200
	\]
	\[
	\sigma_T
	=
	\sqrt{Var(T)}
	=
	\sqrt{200}
	\approx14.142
	\]
	\[
	\boxed{
		E(T)=35,\qquad
		Var(T)=200,\qquad
		\sigma_T\approx14.142
	}
	\]

	\subsection*{C4.4 - Finding the mean, variance, and standard deviation}\phantomsection
	\label{subsec:c4-solution-online-orders}
	
	The supermarket receives an average of 12 orders per hour. Therefore,
	the rate in minutes is
	\[
	\lambda
	=
	\frac{12}{60}
	=
	0.20\text{ orders/minute}.
	\]
	
	Let $X$ represent the time between consecutive orders.
	\[
	\begin{array}{c|c}
		\textbf{General}
		&
		X\sim\operatorname{Exp}(\lambda)
		\\[6pt]
		\textbf{Context}
		&
		\lambda=0.20
	\end{array}
	\]
	
	Therefore,
	\[
	E(X)
	=
	\frac{1}{\lambda}
	=
	\frac{1}{0.20}
	=
	5
	\]
	\[
	Var(X)
	=
	\frac{1}{\lambda^2}
	=
	\frac{1}{(0.20)^2}
	=
	25
	\]
	\[
	\sigma_X
	=
	\sqrt{Var(X)}
	=
	\sqrt{25}
	=
	5
	\]	
	\[
	\boxed{
		E(X)=5,\qquad
		Var(X)=25,\qquad
		\sigma_X=5
	}
	\]

	\subsection*{C4.5 - Finding the mean, variance, and standard deviation}\phantomsection
	\label{subsec:c4-solution-waiting-time}

	range considered equally likely. Therefore, the waiting time follows
	a uniform distribution with
	\[
	\begin{array}{c|c}
		\textbf{General}
		&
		X\sim U(a,b)
		\\[6pt]
		\textbf{Context}
		&
		a=3,\quad b=15
	\end{array}
	\]
	
	Therefore,
	\[
	E(X)
	=
	\frac{a+b}{2}
	=
	\frac{3+15}{2}
	=
	9
	\]
	\[
	Var(X)
	=
	\frac{(b-a)^2}{12}
	=
	\frac{(15-3)^2}{12}
	=
	\frac{144}{12}
	=
	12
	\]
	\[
	\sigma_X
	=
	\sqrt{Var(X)}
	=
	\sqrt{12}
	\approx3.464
	\]	
	\[
	\boxed{
		E(X)=9,\qquad
		Var(X)=12,\qquad
		\sigma_X\approx3.464
	}

	\subsection*{C4.6 - Finding the mean, variance, and standard deviation}\phantomsection
	\label{subsec:c4-solution-production-time}
	
	The production time ranges from 4 to 12 hours, with longer production
	times more likely and the probability density increasing linearly
	across this range.
	\[
	\begin{array}{c|c}
		\textbf{General}
		&
		X\sim\operatorname{Lin}_{\mathrm{inc}}(a,b)
		\\[6pt]
		\textbf{Context}
		&
		a=4,\quad b=12
	\end{array}
	\]
	
	Therefore,
	\[
	E(T)
	=
	\frac{a+2b}{3}
	=
	\frac{4+2(12)}{3}
	=
	\frac{28}{3}
	\approx9.333
	\]
	\[
	Var(T)
	=
	\frac{(b-a)^2}{18}
	=
	\frac{(12-4)^2}{18}
	=
	\frac{64}{18}
	=
	\frac{32}{9}
	\approx3.556
	\]
	\[
	\sigma_T
	=
	\sqrt{Var(T)}
	=
	\sqrt{\frac{32}{9}}
	\approx1.886
	\]
	\[
	\boxed{
		E(T)\approx9.333,\qquad
		Var(T)\approx3.556,\qquad
		\sigma_T\approx1.886
	}
	\]
