	\section*{Evaluated criteria}
	\begin{enumerate}[label=\textbf{C\arabic*}]
		\item Characterises continuous random variables by distinguishing them from discrete variables and interpreting their probability density function, cumulative distribution function, and support in economic contexts.
		\item Applies uniform and triangular continuous distributions with constant or piecewise-linear probability density functions to calculate interval probabilities using geometric areas.
		\item Applies circular-arc probability density functions to calculate interval probabilities using the geometry of circles and circular segments.
		\item Applies the exponential distribution to constant-rate waiting-time situations and assesses its suitability.
	\end{enumerate}
	

	\sectionScored
	{Identifying Probability Distributions from Contexts}
	{C1-\sml{5}}
	{sec:identifying-probability-distributions}
	{
		A variety of Colombian organizations use probability models to represent
		uncertain quantities and events. In each situation below, a random
		variable is defined from the information provided. The appropriate
		probability distribution must be identified from the characteristics
		of the situation.
		
		\begin{enumerate}
			\CriterionTask
			{subsec:identifying-probability-distributions-c1-characterization}{1}
			{For each context, characterize the random variable and identify the appropriate probability distribution. \score{5}}
		\end{enumerate}
		
		\textbf{Context 1.} A payment platform monitors one scheduled
		electronic transfer. By the end of processing, the transfer is either
		completed successfully or not completed. The platform uses this information
		to estimate verification and support needs.
		
		\textbf{Context 2.} An agricultural technology company monitors the
		daily temperature variation inside a refrigerated storage unit. The
		variation can take any value from \(-2^\circ\mathrm{C}\) to
		\(2^\circ\mathrm{C}\), with all values in this interval considered
		equally likely. The company uses this model to set an operating margin.
		
		\textbf{Context 3.} A streaming platform sends the same offer to
		exactly 80 randomly selected eligible users. Each user either subscribes
		or does not. The platform records the number of users who subscribe to
		estimate the server capacity needed for the promotion.
		
		\textbf{Context 4.} A Bogotá parking company records the number of
		vehicle breakdowns during a fixed one-week period. Comparable records
		show that breakdowns occur independently and without a predictable
		timetable. The company uses these records to help schedule maintenance
		staff.
		
		\textbf{Context 5.} An online marketplace studies the delivery time
		for a particular type of order. Delivery times range from 20 to
		80 minutes, with deliveries around 45 minutes occurring most frequently.
		They become progressively less common toward either extreme. The
		company uses this model to estimate delivery times.
		
		\textbf{Context 6.} A quality-control laboratory tests electronic
		components until the first defective component is found. Components
		are tested independently, and testing stops after the first defect.
		The laboratory records the total number of components tested, including
		the defective one, to estimate inspection costs.
		
		\textbf{Context 7.} A cloud-computing provider studies storage usage
		among small business customers. Usage ranges from 0 to 50 gigabytes
		and becomes steadily more common toward the upper limit, without
		reaching a peak before 50 GB. The provider uses this model to plan
		capacity.
		
		\textbf{Context 8.} A retailer sends a promotional message to exactly
		120 customers selected using the same procedure. Each customer either
		makes a purchase or does not. The retailer records the number of
		purchases made in response to the message to estimate the inventory
		needed for the campaign.
		
		\textbf{Context 9.} A coffee exporter measures the moisture content
		of coffee bags under usual storage conditions. Measurements cluster
		around 11\%, with approximately symmetric behaviour around
		this value. Values become less common farther from 11\%. The exporter
		uses this model to establish quality-control thresholds.
		
		\textbf{Context 10.} A food-delivery company records the number of
		customer cancellations during a fixed three-hour dinner period in one
		delivery zone. Comparable evenings show cancellations occurring
		independently without a recurring timetable. The company uses these
		records to estimate customer-service needs.
		
		
		\phantomsection
		\label{subsec:identifying-probability-distributions-c1-solution}
		\subsection*{C1 - Distribution identification: Solutions \score{5}}
		
		\begin{enumerate}
			
			\item \textbf{Context 1 — Discrete, Bernoulli.}
			\(X_1\) records whether the transfer is completed or not.
			
			\item \textbf{Context 2 — Continuous, Uniform.}
			\(X_2\) takes any value in \([-2,2]\), all equally likely.
			
			\item \textbf{Context 3 — Discrete, Binomial.}
			\(X_3\) counts subscribers among 80 users, each with two outcomes.
			
			\item \textbf{Context 4 — Discrete, Poisson.}
			\(X_4\) counts independent breakdowns during a fixed period.
			
			\item \textbf{Context 5 — Continuous, Triangular.}
			\(X_5\) has a minimum of 20, maximum of 80, and mode of 45 minutes.
			
			\item \textbf{Context 6 — Discrete, Geometric.}
			\(X_6\) counts tests until the first defective component.
			
			\item \textbf{Context 7 — Continuous, Linear.}
			\(X_7\) has a density that increases steadily toward 50 GB.
			
			\item \textbf{Context 8 — Discrete, Binomial.}
			\(X_8\) counts purchases among 120 customers, each with two outcomes.
			
			\item \textbf{Context 9 — Continuous, Normal.}
			\(X_9\) is approximately symmetric and concentrated around 11\%.
			
			\item \textbf{Context 10 — Discrete, Poisson.}
			\(X_{10}\) counts independent cancellations during a fixed period.
			
		\end{enumerate}

		\secInicio[sec:identifying-probability-distributions][subsec:identifying-probability-distributions-c1-characterization]
		
	}
	
	\sectionScored
	{Delivery Time for a Local Grocery Order}
	{C2-\sml{3}}
	{sec:linear-delivery-time-distribution}
	{
		A Colombian grocery-delivery company studies the additional time
		required to deliver orders during a busy evening period. For a particular
		delivery zone, the added delivery time \(X\) can range from 0 to
		10 minutes. A delay of any length within this interval is possible, and
		shorter delays become steadily more common. Historical data indicate
		that no additional delay is 4 times as common as a 10-minute delay.
		
		\textbf{Questions/Tasks.}
		\begin{enumerate}
			\CriterionTask
			{subsec:linear-delivery-time-probabilities}{2}
			{Determine the probability that the added delivery time is between 3 and 7 minutes. \score{3}}
		\end{enumerate}
		
		\phantomsection
		\label{subsec:linear-delivery-time-probabilities}
		\subsection*{C2 - Linear probability distribution \score{3}}
		
		\SolutionStage{1}{Support and Shape}
		The random variable \(X\) has support \([0,10]\). The density decreases
		linearly, and its height at \(0\) is four times its height at \(10\).
		Let the endpoint heights be \(4h\) and \(h\).
		
		\SolvedGraph{Shape from the context}{%
			\CompactDensityCoordinatesGraph{-1}{11}{0.30}{0,10}{0,10}
			{0,0.20,0.05}{x}{$0$,$4h$,$h$}
			{(-1,0) (0,0) (0,0.20) (10,0.05) (10,0) (11,0)}%
		}
		
		\SolutionStage{2}{Total Area = 1}
		The total area under a probability density function is \(1\).
		For a trapezoid with parallel sides of heights \(h_1\) and \(h_2\) and base \(b\),
		\[
		A=\frac{h_1+h_2}{2} (b)
		\]
		Therefore, in this case,
		\[
		1=\frac{4h+h}{2}(10)
		\qquad \rightarrow \qquad
		1=5h(5)
		\qquad \rightarrow \qquad
		1=25h
		\qquad \rightarrow \qquad
		h=\frac1{25}.
		\]
		
		Therefore,
		\[
		f(0)= 4h = \frac4{25},
		\qquad
		f(10)= h = \frac1{25}.
		\]
		
		\SolvedGraph{Completed density graph}{%
			\CompactDensityCoordinatesGraph{-1}{11}{0.18}{0,3,7,10}{0,3,7,10}
			{0,0.16,0.04}{x}{$0$,$\frac4{25}$,$\frac1{25}$}
			{(-1,0) (0,0) (0,0.16) (10,0.04) (10,0) (11,0)}%
		}
		
		\SolutionStage{3}{Complete pdf}
		The density is the straight line through
		\[
		\begin{array}{c@{\qquad}c}
			(x_1,y_1) & (x_2,y_2) \\[2mm]
			\left(0,\dfrac{4}{25}\right)
			&
			\left(10,\dfrac{1}{25}\right)
		\end{array}
		\]
		For a general straight line,
		\[
		f(x)=mx+b
		\qquad
		\text{where}
		\qquad
		m=\frac{y_2-y_1}{x_2-x_1}
		\qquad\text{and}\qquad
		b=y_1-mx_1
		\]
		Substituting the given points,
		\[
		m=
		\frac{\frac1{25}-\frac4{25}}{10-0}
		=
		-\frac3{250}
		\qquad
		b=\frac4{25}-\left(-\frac3{250}\right)(0)
		=
		\frac4{25}
		\]
		Therefore,
		\[
		f(x) =mx+b =
		\left( -\frac3{250} \right) x+\frac4{25}
		=
		-\frac{3x}{250}+\frac{40}{250}
		=
		\begin{cases}
			\dfrac{40-3x}{250},&0\le x\le10\\[1mm]
			0,&\text{otherwise}
		\end{cases}
		\]
		
		\SolutionStage{4}{Probability Between 3 and 7 Minutes}
		The required probability is the area under the density from \(3\) to \(7\):
		\[
		P(3\le X\le7)
		=\frac{h_1+h_2}{2} (b) =
		\frac{
			f(3)+f(7)
			}{2}(7-3)		
		\]
		Since
		\[
		f(3)=\frac{31}{250},
		\qquad
		f(7)=\frac{19}{250}
		\]
		\[
		P(3\le X\le7)
		=
		\frac12
		\left(
		\frac{31}{250}+\frac{19}{250}
		\right)(4)
		=
		\frac25.
		\]
		Therefore,
		\[
		\boxed{P(3\le X\le7)=0.40}.
		\]
		
		\SolvedGraph{Band graph: $P(3\le X\le7)$}{%
			\COneGraph[x]{0}{10}{(40-3*x)/250}{3}{7}{0,3,7,10}{0.18}
		}
		
		\vspace{-5mm}
		\secInicio[sec:linear-delivery-time-distribution][subsec:linear-delivery-time-probabilities]
		
	}

	\sectionScored
	{Daily Commute Time for a Bogotá Apprentice}
	{C2-\sml{4}}
	{sec:triangular-commute-time-distribution}
	{
		An apprentice records the time spent travelling to a training programme each day. Commute times range from 1 to 9 hours. Times become steadily more common from 1 hour up to a most common time of 4 hours, then
		steadily less common through 9 hours; there are no commute times right at either limit.
		
		\textbf{Questions/Tasks.}
		\begin{enumerate}
			\CriterionTask
			{subsec:triangular-commute-time-probabilities}{2}
			{Determine the probability that the commute takes between 3 and 7 hours. \score{4}}
		\end{enumerate}
		
		\phantomsection
		\label{subsec:triangular-commute-time-probabilities}
		\subsection*{C2 - Triangular probability distribution \score{4}}
		
		\SolutionStage{1}{Support and Shape}
		The random variable \(T\) has support \([1,9]\). Its density rises
		linearly to a peak at \(t=4\), then falls linearly to zero at \(t=9\).
		Let the peak height be \(H\).
		
		\SolvedGraph{Shape from the context}{%
			\CompactDensityCoordinatesGraph{0}{10}{0.30}{1,4,9}{1,4,9}
			{0,0.25}{t}{$0$,$H$}
			{(0,0) (1,0) (4,0.25) (9,0) (10,0)}%
		}
		
		\SolutionStage{2}{Total Area = 1}
		The total area under a probability density function is \(1\).
		For a triangle with base \(b\) and height \(H\),
		\[
		A=\frac12 bH.
		\]
		Therefore, in this case,
		\[
		1
		=
		\frac12(9-1)H
		\qquad \rightarrow \qquad
		1
		=
		\frac12(8)H
		\qquad \rightarrow \qquad
		1=4H
		\qquad \rightarrow \qquad
		H=\frac14.
		\]
		
		Therefore,
		\[
		f(4)=\frac14.
		\]
		
		\SolvedGraph{Completed density graph}{%
			\CompactDensityCoordinatesGraph{0}{10}{0.30}{1,3,4,6,7,9}{1,3,4,6,7,9}
			{0,0.25}{t}{$0$,$\frac14$}
			{(0,0) (1,0) (4,0.25) (9,0) (10,0)}%
		}
		
		\SolutionStage{3}{Complete pdf}
		The density consists of two straight-line segments. For a general straight line,
		\[
		f(t)=mt+b,
		\qquad
		\text{where}
		\qquad
		m=\frac{y_2-y_1}{x_2-x_1}
		\qquad\text{and}\qquad
		b=y_1-mx_1.
		\]
		
		For the rising segment, the line passes through
		\[
		\begin{array}{c@{\qquad}c}
			(x_1,y_1) & (x_2,y_2) \\[2mm]
			\left(1,0\right)
			&
			\left(4,\dfrac14\right)
		\end{array}
		\]
		Thus
		\[
		m_1
		=
		\frac{\frac14-0}{4-1}
		=
		\frac1{12}
		\qquad
		b_1
		=
		0-\frac1{12}(1)
		=
		-\frac1{12}.
		\]
		Therefore
		\[
		f(t)
		=
		\frac1{12}t-\frac1{12}
		=
		\frac{t-1}{12}.
		\]
		
		For the falling segment, the line passes through
		\[
		\begin{array}{c@{\qquad}c}
			(x_1,y_1) & (x_2,y_2) \\[2mm]
			\left(4,\dfrac14\right)
			&
			(9,0)
		\end{array}
		\]
		Thus
		\[
		m_2
		=
		\frac{0-\frac14}{9-4}
		=
		-\frac1{20}
		\qquad
		b_2
		=
		0-\left(-\frac1{20}\right)(9)
		=
		\frac9{20}.
		\]
		Therefore
		\[
		f(t)
		=
		-\frac1{20}t+\frac9{20}
		=
		\frac{9-t}{20}.
		\]
		
		Therefore,
		\[
		f(t)=
		\begin{cases}
			\dfrac{t-1}{12},&1\le t\le4,\\[1mm]
			\dfrac{9-t}{20},&4<t\le9,\\[1mm]
			0,&\text{otherwise.}
		\end{cases}
		\]
		
		\SolutionStage{4}{Probability Between 3 and 7 Hours}
		The required probability is most easily found by subtracting the two
		tails from the total area:
		\[
		P(3\le T\le7)
		=
		1-P(T<3)-P(T>7).
		\]
		
		The left tail is a triangle with base \((3-1)\) and height \(f(3)\).
		Using the density,
		\[
		f(3)
		=
		\frac{3-1}{12}
		=
		\frac16.
		\]
		
		The right tail is a triangle with base \((9-7)\) and height \(f(7)\).
		Using the density,
		\[
		f(7)
		=
		\frac{9-7}{20}
		=
		\frac1{10}.
		\]
		
		Therefore,
		\[
		\begin{aligned}
			P(3\le T\le7)
			&=
			1
			-
			\frac12(3-1)\left(\frac16\right)
			-
			\frac12(9-7)\left(\frac1{10}\right)
			\\
			& =
			1
			-
			\frac12(2)\left(\frac16\right)
			-
			\frac12(2)\left(\frac1{10}\right) =
			1-\frac16-\frac1{10} = \frac{11}{15}.
		\end{aligned}
		\]
		
		Therefore,
		\[
		\boxed{P(3\le T\le7)=\frac{11}{15}\approx73.3\%}.
		\]
		
		\SolvedGraph{Band graph: $P(3\le T\le7)$}{%
			\COneGraph[t]{1}{9}
			{(x<=4)*((x-1)/12)+(x>4)*((9-x)/20)}
			{3}{7}{1,3,4,6,7,9}{0.30}
		}
		
		\vspace{-5mm}
		\secInicio[sec:triangular-commute-time-distribution]
		[subsec:triangular-commute-time-probabilities]
		
	}
	
	
	\sectionScored
	{Timing of an Unexpected Market Movement}
	{C3-\sml{2}}
	{sec:uniform-market-event-time}
	{
		A financial monitoring system records the time of day at which an
		unexpected movement in an international currency market occurs. The
		movement is equally likely to occur at any time during a 24-hour period,
		so no part of the day is more likely than another.
		
		\textbf{Questions/Tasks.}
		\begin{enumerate}
			\CriterionTask
			{subsec:uniform-market-event-time-probability}{3}
			{Determine the probability that the movement occurs between 9 p.m. and 5 a.m. \score{2}}
		\end{enumerate}
		
		\phantomsection
		\label{subsec:uniform-market-event-time-probability}
		\subsection*{C3 - Uniform probability distribution \score{2}}
		
		\SolutionStage{1}{Uniform Distribution}
		Let \(T\) represent the time of the movement, measured in hours after
		midnight. Since every time during the 24-hour period is equally likely,
		\[
		T\sim U(0,24) \qquad \longrightarrow \qquad 
		 f(t)=\frac1{24 - 0}=\frac1{24}
		\]
		
		\SolutionStage{2}{Probability Between 9 p.m. and 5 a.m.}
		The interval crosses midnight, so it consists of
		\[
		21\le T<24
		\qquad\text{and}\qquad
		0\le T\le5.
		\]
		
		The probability is found by calculating the area of each rectangle
		individually and then adding them:
		\[
		A_1=(5-0)\left(\frac1{24}\right)
		=\frac5{24},
		\]
		\[
		A_2=(24-21)\left(\frac1{24}\right)
		=\frac3{24}.
		\]
		
		Therefore,
		\[
		P(\text{between 9 p.m. and 5 a.m.})
		=
		A_1+A_2
		=
		\frac5{24}+\frac3{24}
		= \frac8{24} = 
		\boxed{\frac13\approx33.33\%}.
		\]

			\begin{center}

	{Direction of an Unexpected Sensor Reading}

	\sectionScored
	{Comparing Investment Transaction Rates}
	{C4-\sml{2}}
	{sec:exponential-comparing-investment-rates}
	{
		Two investment platforms process buy and sell transactions throughout
		the trading day. The transactions processed by each platform are
		modeled by independent Poisson processes.
		
		For Platform A, there is a \(75\%\) probability that the next
		transaction is processed within 8 minutes. For Platform B, there is
		a \(60\%\) probability that the next transaction is processed within
		5 minutes.
		
		Assume both platforms begin processing transactions at the same time.
		Using the transaction rates, determine after how many minutes the
		difference between the numbers of transactions processed by the two
		platforms reaches 1 transaction.
		
		\textbf{Questions/Tasks.}
		\begin{enumerate}
			\CriterionTask
			{subsec:c4-compare-rates-investment}{2}
			{Determine after how many minutes the expected numbers of transactions processed by the two platforms differ by 1 transaction. \score{2}}
		\end{enumerate}
		
	\subsection*{C4.10 - Comparing transaction rates}\phantomsection
	\label{subsec:c4-compare-rates-investment}
	
	For an exponential waiting time \(T\), if
	$$
	P(T<t)=p
	\qquad \longrightarrow \qquad
	p=1-e^{-\lambda t}
	\qquad \longrightarrow \qquad
	{
		\lambda=-\frac{\ln(1-p)}{t}
	}
	$$
	
	For Platform A, \(P(T<8)=0.75\), so
	$$
	\lambda_A
	=-\frac{\ln(1-0.75)}{8}
	\approx0.1733
	\>\frac{\text{transactions}}{\text{minute}}
	$$
	
	For Platform B, \(P(T<5)=0.60\), so
	$$
	\lambda_B
	=-\frac{\ln(1-0.60)}{5}
	\approx0.1833
	\>\frac{\text{transactions}}{\text{minute}}
	$$
	
	Therefore, Platform B has the higher transaction rate. The difference between the rates is
	$$
	\lambda_B-\lambda_A
	\approx0.1833-0.1733
	=0.0100
	\>\frac{\text{transactions}}{\text{minute}}
	$$
	
	After \(t\) minutes, this rate difference accumulates to
	$
	(\lambda_B-\lambda_A)t
	$
	transactions. We want this difference to reach 1 transaction:
	$$
	(\lambda_B-\lambda_A)t=1.
		\qquad \longrightarrow \qquad
	t
	=\frac{1}{\lambda_B-\lambda_A}
	=\frac{1}{0.0100}
	\approx \boxed{100\text{ minutes}}
	$$

	
	}
