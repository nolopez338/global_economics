	\sectionScored
	{Uniform Delivery Time}
	{C1-\sml{9}}
	{sec:uniform-delivery-time-c1}
	{
	A local delivery company records the time required for a particular short-distance route. Under normal traffic conditions, the delivery time ranges from $20$ to $50$ minutes, and every time within this interval is considered equally plausible. Let $T$ represent the delivery time in minutes.
	
	\textbf{Questions/Tasks.}
	\begin{enumerate}
		\CriterionTask
		{subsec:uniform-delivery-time-c1-a}{1}
		{Determine the density constant and write the pdf. \score{3}}
		\CriterionTask
		{subsec:uniform-delivery-time-c1-b}{1}
		{Find the probability that a delivery takes between $25$ and $40$ minutes. \score{3}}
		\CriterionTask
		{subsec:uniform-delivery-time-c1-c}{1}
		{Find the probability that a delivery takes more than $35$ minutes. \score{3}}
	\end{enumerate}
	
	\phantomsection
	\label{subsec:uniform-delivery-time-c1-a}
	\subsection*{C1 - Determine the density constant and write the pdf. \score{3}}
	
	Since all values from $20$ to $50$ minutes are equally plausible, the density is constant on this interval. Let the height of the density be $k$.
	
	The region under the density is a rectangle with width
	\[
	B=50-20=30
	\]
	and height
	\[
	h=k.
	\]
	
	Since the total area under a probability density function must be $1$,
	\[
	A_{\text{rectangle}}=Bh=1.
	\]
	
	Therefore,
	\[
	30k=1
	\qquad \longrightarrow \qquad
	k=\frac{1}{30}.
	\]
	
	Hence the pdf is
	\[
	f(t)=
	\begin{cases}
		\dfrac{1}{30}, & 20\le t\le 50,\\[6pt]
		0, & \text{otherwise.}
	\end{cases}
	\]
	
	\secInicio[sec:uniform-delivery-time-c1]

	\phantomsection
	\label{subsec:uniform-delivery-time-c1-b}
	\subsection*{C1 - Find the probability that a delivery takes between \boldmath{$25$} and \boldmath{$40$} minutes. \score{3}}
	
	We need to find
	\[
	P(25\le T\le40).
	\]
	
	The corresponding region under the density is a rectangle with width
	\[
	B=40-25=15
	\]
	and height
	\[
	h=\frac{1}{30}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		P(25\le T\le40)
		&=A_{\text{rectangle}}\\
		&=Bh\\
		&=(15)\left(\frac{1}{30}\right)\\
		&=\frac{1}{2}.
	\end{aligned}
	\]
	
	Thus,
	\[
	P(25\le T\le40)=\frac{1}{2}.
	\]
	
	\secInicio[sec:uniform-delivery-time-c1]

	\phantomsection
	\label{subsec:uniform-delivery-time-c1-c}
	\subsection*{C1 - Find the probability that a delivery takes more than \boldmath{$35$} minutes. \score{3}}
	
	We need to find
	\[
	P(T>35).
	\]
	
	The region from $35$ to $50$ is a rectangle with width
	\[
	B=50-35=15
	\]
	and height
	\[
	h=\frac{1}{30}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		P(T>35)
		&=A_{\text{rectangle}}\\
		&=(15)\left(\frac{1}{30}\right)\\
		&=\frac{1}{2}.
	\end{aligned}
	\]
	
	Thus,
	\[
	P(T>35)=\frac{1}{2}.
	\]

	\sectionScored
	{Triangular Production Time}
	{C1-\sml{9}}
	{sec:triangular-production-time-c1}
	{
	A workshop models the time required to complete a particular production task. The task cannot take less than $10$ minutes or more than $40$ minutes. A completion time of $20$ minutes is the most likely value. Times become gradually more likely from $10$ to $20$ minutes and gradually less likely from $20$ to $40$ minutes. Let $T$ represent the completion time in minutes.
	
	\textbf{Questions/Tasks.}
	\begin{enumerate}
		\CriterionTask
		{subsec:triangular-production-time-c1-a}{1}
		{Determine the height of the triangular density and write the pdf. \score{3}}
		\CriterionTask
		{subsec:triangular-production-time-c1-b}{1}
		{Find the probability that the task takes less than $20$ minutes. \score{3}}
		\CriterionTask
		{subsec:triangular-production-time-c1-c}{1}
		{Find the probability that the task takes between $20$ and $30$ minutes. \score{3}}
	\end{enumerate}
	
	\phantomsection
	\label{subsec:triangular-production-time-c1-a}
	\subsection*{C1 - Determine the height of the triangular density and write the pdf. \score{3}}
	
	The support extends from $10$ to $40$, and the peak occurs at $20$. Let the height of the peak be $h$.
	
	The complete region under the density is a triangle with base
	\[
	B=40-10=30.
	\]
	
	Since its area must equal $1$,
	\[
	\frac{1}{2}(30)h=1.
	\]
	
	Therefore,
	\[
	15h=1
	\qquad \longrightarrow \qquad
	h=\frac{1}{15}.
	\]
	
	\textbf{Left-hand piece.}
	
	From $(10,0)$ to $\left(20,\frac{1}{15}\right)$, the slope is
	\[
	m=\frac{\frac{1}{15}-0}{20-10}
	=\frac{1}{150}.
	\]
	
	Therefore,
	\[
	f(t)=\frac{t-10}{150}
	\qquad
	\text{for }10\le t\le20.
	\]
	
	\textbf{Right-hand piece.}
	
	From $\left(20,\frac{1}{15}\right)$ to $(40,0)$, the slope is
	\[
	m=
	\frac{0-\frac{1}{15}}{40-20}
	=-\frac{1}{300}.
	\]
	
	Therefore,
	\[
	f(t)=\frac{40-t}{300}
	\qquad
	\text{for }20\le t\le40.
	\]
	
	Hence,
	\[
	f(t)=
	\begin{cases}
		\dfrac{t-10}{150}, & 10\le t\le20,\\[6pt]
		\dfrac{40-t}{300}, & 20\le t\le40,\\[6pt]
		0, & \text{otherwise.}
	\end{cases}
	\]
	
	\secInicio[sec:triangular-production-time-c1]

	\phantomsection
	\label{subsec:triangular-production-time-c1-b}
	\subsection*{C1 - Find the probability that the task takes less than \boldmath{$20$} minutes. \score{3}}
	
	The region from $10$ to $20$ is a triangle with base
	\[
	B=20-10=10
	\]
	and height
	\[
	h=\frac{1}{15}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		P(T<20)
		&=A_{\triangle}\\
		&=\frac{1}{2}(10)\left(\frac{1}{15}\right)\\
		&=\frac{1}{3}.
	\end{aligned}
	\]
	
	Thus,
	\[
	P(T<20)=\frac{1}{3}.
	\]
	
	\secInicio[sec:triangular-production-time-c1]

	\phantomsection
	\label{subsec:triangular-production-time-c1-c}
	\subsection*{C1 - Find the probability that the task takes between \boldmath{$20$} and \boldmath{$30$} minutes. \score{3}}
	
	We need to find
	\[
	P(20\le T\le30).
	\]
	
	At $t=20$,
	\[
	f(20)=\frac{1}{15}.
	\]
	
	At $t=30$,
	\[
	f(30)=\frac{40-30}{300}
	=\frac{1}{30}.
	\]
	
	The region from $20$ to $30$ forms a trapezoid with width
	\[
	B=30-20=10
	\]
	and heights
	\[
	h_1=\frac{1}{15},
	\qquad
	h_2=\frac{1}{30}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		P(20\le T\le30)
		&=A_{\text{trapezoid}}\\
		&=\frac{1}{2}(10)
		\left(
		\frac{1}{15}+\frac{1}{30}
		\right)\\
		&=5\left(\frac{3}{30}\right)\\
		&=\frac{1}{2}.
	\end{aligned}
	\]
	
	Thus,
	\[
	P(20\le T\le30)=\frac{1}{2}.
	\]

	\sectionScored
	{Semicircular Density Model}
	{C2-\sml{9}}
	{sec:semicircular-density-model-c2}
	{
	A continuous random variable $X$ is modeled using the upper half of a circle. The density is symmetric around $x=0$ and has support from $-2$ to $2$. Its graph is a scaled semicircle.
	
	\textbf{Questions/Tasks.}
	\begin{enumerate}
		\CriterionTask
		{subsec:semicircular-density-model-c2-a}{2}
		{Determine the scale factor required for the semicircle to define a probability density function. \score{3}}
		\CriterionTask
		{subsec:semicircular-density-model-c2-b}{2}
		{Find $P(-2\le X\le0)$. \score{3}}
		\CriterionTask
		{subsec:semicircular-density-model-c2-c}{2}
		{Find $P(0\le X\le1)$ using circular-segment geometry. \score{3}}
	\end{enumerate}
	
	\phantomsection
	\label{subsec:semicircular-density-model-c2-a}
	\subsection*{C2 - Determine the scale factor required for the semicircle to define a probability density function. \score{3}}
	
	The unscaled curve is the upper semicircle
	\[
	y=\sqrt{4-x^2},
	\]
	which has radius
	\[
	r=2.
	\]
	
	The area of a complete circle is
	\[
	A_{\text{circle}}=\pi r^2=4\pi.
	\]
	
	Therefore, the area of the upper semicircle is
	\[
	A_{\text{semicircle}}
	=\frac{1}{2}(4\pi)
	=2\pi.
	\]
	
	Suppose the density is
	\[
	f(x)=k\sqrt{4-x^2}.
	\]
	
	The total area must equal $1$, so
	\[
	k(2\pi)=1.
	\]
	
	Therefore,
	\[
	k=\frac{1}{2\pi}.
	\]
	
	Hence,
	\[
	f(x)=
	\begin{cases}
		\dfrac{1}{2\pi}\sqrt{4-x^2}, & -2\le x\le2,\\[6pt]
		0, & \text{otherwise.}
	\end{cases}
	\]
	
	\secInicio[sec:semicircular-density-model-c2]

	\phantomsection
	\label{subsec:semicircular-density-model-c2-b}
	\subsection*{C2 - Find \boldmath{$P(-2\le X\le0)$}. \score{3}}
	
	The density is symmetric around $x=0$. Therefore, the region from $-2$ to $0$ is exactly half of the total density region.
	
	Since the total area is $1$,
	\[
	P(-2\le X\le0)=\frac{1}{2}.
	\]
	
	\secInicio[sec:semicircular-density-model-c2]

	\phantomsection
	\label{subsec:semicircular-density-model-c2-c}
	\subsection*{C2 - Find \boldmath{$P(0\le X\le1)$} using circular-segment geometry. \score{3}}
	
	Consider the unscaled circle
	\[
	x^2+y^2=4.
	\]
	
	At $x=1$,
	\[
	1^2+y^2=4
	\qquad \longrightarrow \qquad
	y=\sqrt{3}.
	\]
	
	The point
	\[
	(1,\sqrt{3})
	\]
	lies on the circle of radius $2$.
	
	The corresponding central angle from the positive $x$-axis is
	\[
	\theta=\frac{\pi}{3}.
	\]
	
	The region under the arc from $x=0$ to $x=1$ can be decomposed into a sector and a right triangle.
	
	The sector angle is
	\[
	\frac{\pi}{2}-\frac{\pi}{3}
	=\frac{\pi}{6}.
	\]
	
	Therefore, the sector area is
	\[
	A_{\text{sector}}
	=\frac{1}{2}r^2\theta
	=\frac{1}{2}(2^2)\left(\frac{\pi}{6}\right)
	=\frac{\pi}{3}.
	\]
	
	The right triangle has base $1$ and height $\sqrt{3}$, so
	\[
	A_{\triangle}
	=\frac{1}{2}(1)(\sqrt{3})
	=\frac{\sqrt{3}}{2}.
	\]
	
	Thus, the unscaled area under the arc is
	\[
	A=\frac{\pi}{3}+\frac{\sqrt{3}}{2}.
	\]
	
	Since the density is scaled by
	\[
	\frac{1}{2\pi},
	\]
	the probability is
	\[
	P(0\le X\le1)
	=
	\frac{1}{2\pi}
	\left(
	\frac{\pi}{3}+\frac{\sqrt{3}}{2}
	\right).
	\]
	
	Therefore,
	\[
	\boxed{
		P(0\le X\le1)
		=
		\frac{1}{6}+\frac{\sqrt{3}}{4\pi}
	}
	\]

	\secInicio[sec:semicircular-density-model-c2]
	}

	{Circular-Arc Measurement Model}
	{C2-\sml{9}}
	{sec:circular-arc-measurement-model-c2}
	{
	A continuous measurement $X$ is modeled by a symmetric circular-arc density supported from $-3$ to $3$. Before normalization, the graph follows the upper half of the circle
	\[
	x^2+y^2=9.
	\]
	
	\textbf{Questions/Tasks.}
	\begin{enumerate}
		\CriterionTask
		{subsec:circular-arc-measurement-model-c2-a}{2}
		{Determine the normalized pdf. \score{3}}
		\CriterionTask
		{subsec:circular-arc-measurement-model-c2-b}{2}
		{Find $P(X>0)$. \score{3}}
		\CriterionTask
		{subsec:circular-arc-measurement-model-c2-c}{2}
		{Find $P(\frac{3}{2}\le X\le3)$ using a circular segment. \score{3}}
	\end{enumerate}
	
	\phantomsection
	\label{subsec:circular-arc-measurement-model-c2-a}
	\subsection*{C2 - Determine the normalized pdf. \score{3}}
	
	The circle has radius
	\[
	r=3.
	\]
	
	The area of the upper semicircle is
	\[
	A_{\text{semicircle}}
	=\frac{1}{2}\pi r^2
	=\frac{9\pi}{2}.
	\]
	
	Let
	\[
	f(x)=k\sqrt{9-x^2}.
	\]
	
	Since the total area must equal $1$,
	\[
	k\left(\frac{9\pi}{2}\right)=1.
	\]
	
	Therefore,
	\[
	k=\frac{2}{9\pi}.
	\]
	
	Hence,
	\[
	f(x)=
	\begin{cases}
		\dfrac{2}{9\pi}\sqrt{9-x^2}, & -3\le x\le3,\\[6pt]
		0, & \text{otherwise.}
	\end{cases}
	\]
	
	\secInicio[sec:circular-arc-measurement-model-c2]

	\phantomsection
	\label{subsec:circular-arc-measurement-model-c2-b}
	\subsection*{C2 - Find \boldmath{$P(X>0)$}. \score{3}}
	
	The density is symmetric around $x=0$. Therefore,
	\[
	P(X>0)=\frac{1}{2}.
	\]
	
	\secInicio[sec:circular-arc-measurement-model-c2]

	\phantomsection
	\label{subsec:circular-arc-measurement-model-c2-c}
	\subsection*{C2 - Find \boldmath{$P(\frac{3}{2}\le X\le3)$} using a circular segment. \score{3}}
	
	At
	\[
	x=\frac{3}{2},
	\]
	the corresponding point on the circle satisfies
	\[
	\left(\frac{3}{2}\right)^2+y^2=9.
	\]
	
	Therefore,
	\[
	y^2
	=9-\frac{9}{4}
	=\frac{27}{4},
	\]
	so
	\[
	y=\frac{3\sqrt{3}}{2}.
	\]
	
	The point
	\[
	\left(
	\frac{3}{2},
	\frac{3\sqrt{3}}{2}
	\right)
	\]
	corresponds to the central angle
	\[
	\theta=\frac{\pi}{3}.
	\]
	
	The region under the arc from $\frac{3}{2}$ to $3$ is a circular segment.
	
	The sector area is
	\[
	A_{\text{sector}}
	=\frac{1}{2}r^2\theta
	=\frac{1}{2}(9)\left(\frac{\pi}{3}\right)
	=\frac{3\pi}{2}.
	\]
	
	The triangle formed by the two radii has area
	\[
	A_{\triangle}
	=
	\frac{1}{2}r^2\sin\theta
	=
	\frac{1}{2}(9)\sin\left(\frac{\pi}{3}\right).
	\]
	
	Therefore,
	\[
	A_{\triangle}
	=
	\frac{9}{2}\left(\frac{\sqrt{3}}{2}\right)
	=
	\frac{9\sqrt{3}}{4}.
	\]
	
	Thus, the circular-segment area is
	\[
	A_{\text{segment}}
	=
	\frac{3\pi}{2}
	-
	\frac{9\sqrt{3}}{4}.
	\]
	
	Applying the normalization factor,
	\[
	P\left(\frac{3}{2}\le X\le3\right)
	=
	\frac{2}{9\pi}
	\left(
	\frac{3\pi}{2}
	-
	\frac{9\sqrt{3}}{4}
	\right).
	\]
	
	Therefore,
	\[
	\boxed{
		P\left(\frac{3}{2}\le X\le3\right)
		=
		\frac{1}{3}
		-
		\frac{\sqrt{3}}{2\pi}
	}
	\]

	\sectionScored
	{Customer Arrival Waiting Time}
	{C3-\sml{9}}
	{sec:customer-arrival-waiting-time-c3}
	{
	During a quiet period at a service desk, customers arrive independently at an approximately constant average rate of $6$ customers per hour. Let $T$ represent the waiting time, in minutes, until the next customer arrives.
	
	\textbf{Questions/Tasks.}
	\begin{enumerate}
		\CriterionTask
		{subsec:customer-arrival-waiting-time-c3-a}{3}
		{Determine the exponential rate parameter in minutes and write the pdf. \score{3}}
		\CriterionTask
		{subsec:customer-arrival-waiting-time-c3-b}{3}
		{Find the probability that the next customer arrives within $5$ minutes. \score{3}}
		\CriterionTask
		{subsec:customer-arrival-waiting-time-c3-c}{3}
		{Explain whether an exponential model is suitable in this situation. \score{3}}
	\end{enumerate}
	
	\phantomsection
	\label{subsec:customer-arrival-waiting-time-c3-a}
	\subsection*{C3 - Determine the exponential rate parameter in minutes and write the pdf. \score{3}}
	
	The average arrival rate is
	\[
	6\text{ customers per hour}.
	\]
	
	Since one hour contains $60$ minutes,
	\[
	\lambda=\frac{6}{60}
	=\frac{1}{10}
	\]
	customer per minute.
	
	Therefore,
	\[
	\lambda=0.1.
	\]
	
	For an exponential waiting-time distribution,
	\[
	f(t)=\lambda e^{-\lambda t}.
	\]
	
	Hence,
	\[
	f(t)=
	\begin{cases}
		0.1e^{-0.1t}, & t\ge0,\\
		0, & t<0.
	\end{cases}
	\]
	
	\secInicio[sec:customer-arrival-waiting-time-c3]

	\phantomsection
	\label{subsec:customer-arrival-waiting-time-c3-b}
	\subsection*{C3 - Find the probability that the next customer arrives within \boldmath{$5$} minutes. \score{3}}
	
	For an exponential random variable,
	\[
	P(T\le t)=1-e^{-\lambda t}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		P(T\le5)
		&=1-e^{-(0.1)(5)}\\
		&=1-e^{-0.5}.
	\end{aligned}
	\]
	
	Thus,
	\[
	\boxed{
		P(T\le5)=1-e^{-0.5}
	}
	\]
	
	and approximately,
	\[
	P(T\le5)\approx0.3935.
	\]
	
	\secInicio[sec:customer-arrival-waiting-time-c3]

	\phantomsection
	\label{subsec:customer-arrival-waiting-time-c3-c}
	\subsection*{C3 - Explain whether an exponential model is suitable in this situation. \score{3}}
	
	An exponential model is suitable when arrivals can reasonably be treated as independent and when the average arrival rate remains approximately constant during the period being modeled.
	
	The problem states that customers arrive independently and that the average rate is approximately constant at $6$ customers per hour during the quiet period. These conditions support the use of an exponential waiting-time model.
	
	Therefore, the exponential distribution is a reasonable model for this situation, provided that the arrival rate remains approximately constant during the period considered.

	\sectionScored
	{Technical Support Waiting Time}
	{C3-\sml{9}}
	{sec:technical-support-waiting-time-c3}
	{
	A technical-support center receives calls throughout the day. Historical records show an overall average of $12$ calls per hour. However, the center is much busier during the morning and immediately after lunch than during the late afternoon. A manager proposes using a single exponential distribution with rate $12$ calls per hour to model the waiting time until the next call at any time during the working day.
	
	\textbf{Questions/Tasks.}
	\begin{enumerate}
		\CriterionTask
		{subsec:technical-support-waiting-time-c3-a}{3}
		{Using the proposed model, determine the rate parameter in minutes and the expected waiting time. \score{3}}
		\CriterionTask
		{subsec:technical-support-waiting-time-c3-b}{3}
		{Using the proposed model, find the probability that the waiting time exceeds $10$ minutes. \score{3}}
		\CriterionTask
		{subsec:technical-support-waiting-time-c3-c}{3}
		{Assess whether a single exponential distribution is suitable for modeling waiting times throughout the entire day. \score{3}}
	\end{enumerate}
	
	\phantomsection
	\label{subsec:technical-support-waiting-time-c3-a}
	\subsection*{C3 - Using the proposed model, determine the rate parameter in minutes and the expected waiting time. \score{3}}
	
	The proposed average rate is
	\[
	12\text{ calls per hour}.
	\]
	
	Converting this to calls per minute,
	\[
	\lambda=\frac{12}{60}
	=\frac{1}{5}.
	\]
	
	Therefore,
	\[
	\lambda=0.2
	\]
	calls per minute.
	
	For an exponential distribution,
	\[
	E(T)=\frac{1}{\lambda}.
	\]
	
	Therefore,
	\[
	E(T)=\frac{1}{0.2}=5.
	\]
	
	Thus, according to the proposed model, the expected waiting time is
	\[
	\boxed{5\text{ minutes}.}
	\]
	
	\secInicio[sec:technical-support-waiting-time-c3]

	\phantomsection
	\label{subsec:technical-support-waiting-time-c3-b}
	\subsection*{C3 - Using the proposed model, find the probability that the waiting time exceeds \boldmath{$10$} minutes. \score{3}}
	
	For an exponential random variable,
	\[
	P(T>t)=e^{-\lambda t}.
	\]
	
	Therefore,
	\[
	\begin{aligned}
		P(T>10)
		&=e^{-(0.2)(10)}\\
		&=e^{-2}.
	\end{aligned}
	\]
	
	Thus,
	\[
	\boxed{
		P(T>10)=e^{-2}
	}
	\]
	
	and approximately,
	\[
	P(T>10)\approx0.1353.
	\]
	
	\secInicio[sec:technical-support-waiting-time-c3]

	\phantomsection
	\label{subsec:technical-support-waiting-time-c3-c}
	\subsection*{C3 - Assess whether a single exponential distribution is suitable for modeling waiting times throughout the entire day. \score{3}}
	
	A single exponential waiting-time model assumes that the underlying event rate is constant during the period being modeled.
	
	In this situation, the call rate changes substantially throughout the day. The center is busier during the morning and immediately after lunch and quieter during the late afternoon. Therefore, the constant-rate assumption required by a single exponential model is not well supported.
	
	The overall average of $12$ calls per hour can describe the average number of calls across the day, but it does not imply that calls arrive at that same rate during every part of the day.
	
	Therefore, a single exponential distribution with rate $12$ calls per hour is not an appropriate model for waiting times throughout the entire working day. Separate models for periods with approximately constant call rates would be more appropriate.
