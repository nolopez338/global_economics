	\sectionScored
	{Waiting for a Study-Room Opening}
	{C4-\sml{5}}
	{sec:exponential-study-room-waiting-time}
	{
		At a university library in Bogotá, study rooms become available at an
		average rate of one room every 5 minutes. Let \(T\) represent the
		waiting time, in minutes, until the next study room becomes available.
		
		\textbf{Questions/Tasks.}
		\begin{enumerate}
			\CriterionTask
			{subsec:exponential-study-room-probability}{3}
			{Determine the probability that the next available study room appears
				between 2 and 7 minutes after the observation time. \score{4}}
		\end{enumerate}
		
		\phantomsection
		\label{subsec:exponential-study-room-probability}
		\subsection*{C4 - Exponential probability distribution \score{4}}
		
		\SolutionStage{1}{Rate Parameter}
		The average rate is one available room every \(5\) minutes.
		Therefore, the exponential rate parameter is
		\[
		\lambda=\frac{1}{5}=0.20.
		\]
		
		Thus, \(T\) has an exponential distribution with rate
		\[
		\lambda=0.20.
		\]
		
		\SolutionStage{2}{Exponential Probability Rule}
		For an exponential random variable, the probability that \(T\) lies
		between \(a\) and \(b\) is
		\[
		P(a<T<b)=e^{-\lambda a}-e^{-\lambda b}.
		\]
		
		In this problem,
		\[
		a=2,\qquad b=7,\qquad \lambda=0.20.
		\]
		
		\SolvedGraph{Probability region}{%
			\ExponentialProbabilityGraph{0.20}{2}{7}
		}
		
		\SolutionStage{3}{Substitution}
		Substituting \(a=2\), \(b=7\), and \(\lambda=0.20\),
		\[
		P(2<T<7)
		=e^{-(0.20)(2)}-e^{-(0.20)(7)}.
		\]
		
		Therefore,
		\[
		P(2<T<7)
		=e^{-0.4}-e^{-1.4}
		\]
		
		\SolutionStage{4}{Final Probability}
		Evaluating the expression gives
		\[
		P(2<T<7)
		\approx0.4259.
		\]
		
		Thus, the probability that the next study room becomes available
		between 2 and 7 minutes after the observation time is
		\[
		\boxed{P(2<T<7)\approx0.4259=42.59\%}
		\]
	}
