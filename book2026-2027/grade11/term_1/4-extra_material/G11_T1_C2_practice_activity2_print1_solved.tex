	\vspace{-4mm}
	
	\noindent\textbf{Line through two points.}  For $x_1\ne x_2$,
	\[
	m=\frac{y_2-y_1}{x_2-x_1},\qquad
	b=y_1-mx_1,\qquad
	y=mx+b.
	\]
	
	\begin{enumerate}[
		leftmargin=*,
		itemsep=2pt,
		parsep=0pt,
		topsep=2pt
		]
		
		\item \textbf{Waiting for TransMilenio after class.} On weekday evenings,
		a university student leaves class and waits at a TransMilenio station near
		the Universidad Nacional. The wait always lasts between 2 and 10
		minutes, and every waiting time in that span is equally plausible.
		Determine the complete probability density rule for the waiting time,
		including what happens outside the stated span. Then find the chance that
		the student waits longer than 7 minutes and the chance that the wait
		lasts from 3 through 7 minutes.
		
		\item \textbf{An evening commute across Bogot\'a.} A young apprentice
		travels by SITP from Kennedy to a training programme in Chapinero during
		the evening rush. Traffic can add anywhere from no delay to 8 minutes
		to the usual journey. A delay of any length in that span is possible, and
		longer delays become steadily more common. An 8-minute delay is 5
		times as common as no delay. Determine the complete probability density
		rule for the added travel time, including what happens outside the stated
		span. Then find the chance of more than 6 minutes of added time and the
		chance of between 2 and 6 minutes of added time.
		
		\item \textbf{Weekly study at a public library.} Members of a Bogot\'a
		youth scholarship programme record the time they study each week at the
		Biblioteca P\'ublica Virgilio Barco. Weekly totals range from 2 to 10
		hours. Totals become steadily more common from 2 hours up to a most
		common total of 5 hours, then steadily less common through 10 hours;
		there are no totals right at either limit. Determine the complete
		probability density rule for weekly study time, including what happens
		outside the stated span. Then find the chance of studying for more than
		7 hours and the chance of studying from 4 through 8 hours.
		
		\item \textbf{A Sunday on the Ciclov\'ia.} A group of young adults tracks
		how long its members spend cycling and meeting friends on Bogot\'a's Sunday
		Ciclov\'ia. A recorded visit can last from 0 to 5 hours. All
		durations up to 2 hours are equally common. All durations from 2 to
		5 hours are also equally common, but they occur 2 times as often as
		durations in the shorter band. Determine the complete probability density
		rule for the visit length, including what happens outside the stated span.
		Then find the chance that a visit lasts more than 4 hours and the chance
		that it lasts from 1 through 4 hours.
		
		\item \textbf{Time at a youth entrepreneurship fair.} Young adults attend
		a Bogot\'a entrepreneurship fair at Corferias for up to 6 hours.
		Immediate departures are not observed. Departures become
		steadily more common during the first 2 hours, remain equally common at
		their highest level from 2 through 4 hours, and then become steadily
		less common until 6 hours, when none are observed. Determine the
		complete probability density rule for the length of a visit, including
		what happens outside the stated span. Then find the chance of staying
		longer than 5 hours and the chance of staying from 1 through 5 hours.
		
	\end{enumerate}

	\section{Uniform distribution}
	
	\SolutionStage{1}{Support and Shape}
	The random variable $T$ has support $[2,10]$.  The context says that all
	waiting times in this interval are equally plausible, so the density is a
	rectangle of unknown height $h$ and is zero outside the support.
	\SolvedGraph{Shape from the context}{%
		\CompactDensityCoordinatesGraph{1}{11}{0.30}{2,10}{2,10}
		{0,0.20}{t}{$0$,$h$}{(1,0) (2,0) (2,0.20) (10,0.20) (10,0) (11,0)}%
	}
	
	\SolutionStage{2}{Total Area = 1}
	The rectangle must have area $1$:
	\[
	1=(10-2)h=8h,\qquad h=\frac18.
	\]
	\SolvedGraph{Completed density graph}{%
		\CompactDensityCoordinatesGraph{1}{11}{0.15}{2,3,7,10}{2,3,7,10}
		{0,0.125}{t}{$0$,$\frac18$}{(1,0) (2,0) (2,0.125) (10,0.125) (10,0) (11,0)}%
	}
	
	\SolutionStage{3}{Straight Line Rule}
	The only segment is horizontal.  Thus $m=0$, and its constant rule on the
	support is $f_T(t)=\frac18$; no nontrivial line calculation is needed.
	
	\SolutionStage{4}{Complete pdf}
	\[
	f_T(t)=\begin{cases}\dfrac18,&2\le t\le10,\\[1mm]0,&\text{otherwise.}\end{cases}
	\]
	
	\SolutionStage{5}{Tail Probability}
	The relevant right-tail region is a rectangle:
	\[
	P(T>7)=(10-7)\frac18=\frac38=37.5\%.
	\]
	\SolvedGraph{Tail graph: $P(T>7)$}{\CtwoProbGraph[t]{2}{10}{1/8}{7}{10}{2,3,7,10}{0.16}}
	
	\SolutionStage{6}{Interval Probability}
	This interval is obtained directly as a rectangle:
	\[
	P(3\le T\le7)=(7-3)\frac18=\frac12.
	\]
	\SolvedGraph{Band graph: $P(3\le T\le7)$}{\CtwoProbGraph[t]{2}{10}{1/8}{3}{7}{2,3,7,10}{0.16}}

	\section{Linear distribution}
	
	\SolutionStage{1}{Support and Shape}
	The random variable $X$ has support $[0,8]$.  The context supplies a steadily
	increasing straight line whose height at $8$ is five times its height at $0$.
	Write the endpoint heights as $h$ and $5h$; the density is zero outside the
	support.
	\SolvedGraph{Shape from the context}{%
		\CompactDensityCoordinatesGraph{-1}{9}{0.30}{0,8}{0,8}
		{0,0.05,0.25}{x}{$0$,$h$,$5h$}{(-1,0) (0,0) (0,0.05) (8,0.25) (8,0) (9,0)}%
	}
	
	\SolutionStage{2}{Total Area = 1}
	The trapezoid must have area $1$:
	\[
	1=\frac12(8)(h+5h)=24h,\qquad h=\frac1{24},\qquad5h=\frac5{24}.
	\]
	\SolvedGraph{Completed density graph}{%
		\CompactDensityCoordinatesGraph{-1}{9}{0.24}{0,2,6,8}{0,2,6,8}
		{0,0.0416667,0.2083333}{x}{$0$,$\frac1{24}$,$\frac5{24}$}
		{(-1,0) (0,0) (0,0.0416667) (8,0.2083333) (8,0) (9,0)}%
	}
	
	\SolutionStage{3}{Straight Line Rule}
	Use the points $(x_1,y_1)=(0,\frac1{24})$ and
	$(x_2,y_2)=(8,\frac5{24})$.  The supplied formulas give
	\[
	m=\frac{y_2-y_1}{x_2-x_1}
	=\frac{\frac5{24}-\frac1{24}}{8-0}=\frac1{48},\qquad
	b=y_1-mx_1=\frac1{24}-\frac1{48}(0)=\frac1{24}.
	\]
	Therefore $y=mx+b$ gives $f_X(x)=\frac{x}{48}+\frac1{24}=\frac{x+2}{48}$.
	
	\SolutionStage{4}{Complete pdf}
	\[
	f_X(x)=\begin{cases}\dfrac{x+2}{48},&0\le x\le8,\\[1mm]0,&\text{otherwise.}\end{cases}
	\]
	
	\SolutionStage{5}{Tail Probability}
	Since $f_X(6)=\frac16$ and $f_X(8)=\frac5{24}$, the right-tail region is a
	width-$2$ trapezoid:
	\[
	P(X>6)=\frac12(8-6)\left(\frac16+\frac5{24}\right)=\frac38.
	\]
	\SolvedGraph{Tail graph: $P(X>6)$}{\COneGraph[x]{0}{8}{(x+2)/48}{6}{8}{0,2,6,8}{0.24}}
	
	\SolutionStage{6}{Interval Probability}
	This is obtained directly from the trapezoid between $2$ and $6$.  Since
	$f_X(2)=\frac1{12}$ and $f_X(6)=\frac16$,
	\[
	P(2\le X\le6)=\frac12(6-2)\left(\frac1{12}+\frac16\right)=\frac12.
	\]
	\SolvedGraph{Band graph: $P(2\le X\le6)$}{\COneGraph[x]{0}{8}{(x+2)/48}{2}{6}{0,2,6,8}{0.24}}

	\section{Triangular distribution}
	
	\SolutionStage{1}{Support and Shape}
	The random variable $Q$ has support $[2,10]$.  The context supplies zero
	height at both endpoints, a steady rise to the most common value at $q=5$,
	and a steady fall thereafter.  Denote the unknown peak height by $H$; the
	density is zero outside the support.
	\SolvedGraph{Shape from the context}{%
		\CompactDensityCoordinatesGraph{1}{11}{0.30}{2,5,10}{2,5,10}
		{0,0.22}{q}{$0$,$H$}{(1,0) (2,0) (5,0.22) (10,0) (11,0)}%
	}
	
	\SolutionStage{2}{Total Area = 1}
	The triangle must have area $1$:
	\[
	1=\frac12(10-2)H=4H,\qquad H=\frac14.
	\]
	\SolvedGraph{Completed density graph}{%
		\CompactDensityCoordinatesGraph{1}{11}{0.30}{2,4,5,7,8,10}{2,4,5,7,8,10}
		{0,0.25}{q}{$0$,$\frac14$}{(1,0) (2,0) (5,0.25) (10,0) (11,0)}%
	}
	
	\SolutionStage{3}{Straight Line Rule}
	For the rising segment, use $(2,0)$ and $(5,\frac14)$:
	\[
	m_1=\frac{\frac14-0}{5-2}=\frac1{12},\qquad
	b_1=y_1-m_1x_1=0-\frac1{12}(2)=-\frac16,
	\quad y=m_1q+b_1=\frac{q-2}{12}.
	\]
	For the falling segment, use $(5,\frac14)$ and $(10,0)$:
	\[
	m_2=\frac{0-\frac14}{10-5}=-\frac1{20},\qquad
	b_2=y_1-m_2x_1=\frac14-\left(-\frac1{20}\right)5=\frac12,
	\quad y=m_2q+b_2=\frac{10-q}{20}.
	\]
	The adjacent rules agree at $q=5$: both give $\frac14$.
	
	\SolutionStage{4}{Complete pdf}
	\[
	f_Q(q)=\begin{cases}
		\dfrac{q-2}{12},&2\le q\le5,\\[1mm]
		\dfrac{10-q}{20},&5<q\le10,\\[1mm]
		0,&\text{otherwise.}
	\end{cases}
	\]
	As checked above, the pieces are continuous at $q=5$.
	
	\SolutionStage{5}{Tail Probability}
	Since $f_Q(7)=\frac3{20}$, the right-tail region is a triangle:
	\[
	P(Q>7)=\frac12(10-7)\frac3{20}=\frac9{40}=22.5\%.
	\]
	\SolvedGraph{Tail graph: $P(Q>7)$}{\COneGraph[q]{2}{10}{(x<=5)*((x-2)/12)+(x>5)*((10-x)/20)}{7}{10}{2,4,5,7,8,10}{0.30}}
	
	\SolutionStage{6}{Interval Probability}
	Use the complement of the two outside tail triangles.  Since
	$f_Q(4)=\frac16$ and $f_Q(8)=\frac1{10}$,
	\[
	\begin{aligned}P(4\le Q\le8)
		&=1-\frac12(4-2)\frac16-\frac12(10-8)\frac1{10}\\
		&=\frac{11}{15}\approx73.3\%.
	\end{aligned}
	\]
	\SolvedGraph{Band graph: $P(4\le Q\le8)$}{\COneGraph[q]{2}{10}{(x<=5)*((x-2)/12)+(x>5)*((10-x)/20)}{4}{8}{2,4,5,7,8,10}{0.30}}

	\section{Piecewise-uniform distribution}
	
	\SolutionStage{1}{Support and Shape}
	The random variable $W$ has support $[0,5]$, with breakpoint $2$.  The context
	supplies a constant height $k$ on $[0,2)$ and twice that height, $2k$, on
	$[2,5]$; the density is zero outside the support.
	\SolvedGraph{Shape from the context}{%
		\CompactDensityCoordinatesGraph{-0.5}{5.5}{0.30}{0,2,5}{0,2,5}
		{0,0.10,0.20}{w}{$0$,$k$,$2k$}
		{(-0.5,0) (0,0) (0,0.10) (2,0.10) (2,0.20) (5,0.20) (5,0) (5.5,0)}%
	}
	
	\SolutionStage{2}{Total Area = 1}
	The two rectangles must have total area $1$:
	\[
	1=2k+(5-2)(2k)=8k,\qquad k=\frac18,\qquad2k=\frac14.
	\]
	\SolvedGraph{Completed density graph}{%
		\CompactDensityCoordinatesGraph{-0.5}{5.5}{0.30}{0,1,2,4,5}{0,1,2,4,5}
		{0,0.125,0.25}{w}{$0$,$\frac18$,$\frac14$}
		{(-0.5,0) (0,0) (0,0.125) (2,0.125) (2,0.25) (5,0.25) (5,0) (5.5,0)}%
	}
	
	\SolutionStage{3}{Straight Line Rule}
	Both segments are horizontal, so each has $m=0$; no nontrivial line
	calculation is needed.  Their constant rules are $f_W(w)=\frac18$ on
	$[0,2)$ and $f_W(w)=\frac14$ on $[2,5]$.
	
	\SolutionStage{4}{Complete pdf}
	\[
	f_W(w)=\begin{cases}
		\dfrac18,&0\le w<2,\\[1mm]
		\dfrac14,&2\le w\le5,\\[1mm]
		0,&\text{otherwise.}
	\end{cases}
	\]
	
	\SolutionStage{5}{Tail Probability}
	The right-tail region is a width-$1$, height-$\frac14$ rectangle:
	\[
	P(W>4)=(5-4)\frac14=\frac14.
	\]
	\SolvedGraph{Tail graph: $P(W>4)$}{\CtwoProbGraph[w]{0}{5}{(x<2)*(1/8)+(x>=2)*(1/4)}{4}{5}{0,1,2,4,5}{0.30}}
	
	\SolutionStage{6}{Interval Probability}
	Split the interval at the breakpoint $2$ and add the two rectangles:
	\[
	P(1\le W\le4)=(2-1)\frac18+(4-2)\frac14=\frac18+\frac12=\frac58.
	\]
	\SolvedGraph{Band graph: $P(1\le W\le4)$}{\CtwoProbGraph[w]{0}{5}{(x<2)*(1/8)+(x>=2)*(1/4)}{1}{4}{0,1,2,4,5}{0.30}}

	\section{Piecewise-linear distribution}
	
	\SolutionStage{1}{Support and Shape}
	The random variable $D$ has support $[0,6]$.  The context supplies a steady
	rise from zero to an unknown height $H$ at $d=2$, that same constant height
	through $d=4$, and a steady fall to zero at $d=6$.  Thus the symbolic vertices
	are $(0,0),(2,H),(4,H),(6,0)$, and the density is zero outside the support.
	\SolvedGraph{Shape from the context}{%
		\CompactDensityCoordinatesGraph{-0.5}{6.5}{0.30}{0,2,4,6}{0,2,4,6}
		{0,0.22}{d}{$0$,$H$}{(-0.5,0) (0,0) (2,0.22) (4,0.22) (6,0) (6.5,0)}%
	}
	
	\SolutionStage{2}{Total Area = 1}
	The two triangles and middle rectangle must have total area $1$:
	\[
	1=\frac12(2)H+(4-2)H+\frac12(6-4)H=4H,
	\qquad H=\frac14.
	\]
	\SolvedGraph{Completed density graph}{%
		\CompactDensityCoordinatesGraph{-0.5}{6.5}{0.30}{0,1,2,4,5,6}{0,1,2,4,5,6}
		{0,0.25}{d}{$0$,$\frac14$}{(-0.5,0) (0,0) (2,0.25) (4,0.25) (6,0) (6.5,0)}%
	}
	
	\SolutionStage{3}{Straight Line Rule}
	For the first segment, use $(0,0)$ and $(2,\frac14)$:
	\[
	m_1=\frac{\frac14-0}{2-0}=\frac18,\qquad
	b_1=y_1-m_1x_1=0-\frac18(0)=0,
	\quad y=m_1d+b_1=\frac d8.
	\]
	For the horizontal segment, use $(2,\frac14)$ and $(4,\frac14)$:
	\[
	m_2=\frac{\frac14-\frac14}{4-2}=0,\qquad
	b_2=y_1-m_2x_1=\frac14-0(2)=\frac14,
	\quad y=m_2d+b_2=\frac14.
	\]
	For the last segment, use $(4,\frac14)$ and $(6,0)$:
	\[
	m_3=\frac{0-\frac14}{6-4}=-\frac18,\qquad
	b_3=y_1-m_3x_1=\frac14-\left(-\frac18\right)4=\frac34,
	\quad y=m_3d+b_3=\frac{6-d}{8}.
	\]
	At $d=2$ the first two rules both give $\frac14$, and at $d=4$ the last two
	rules both give $\frac14$.
	
	\SolutionStage{4}{Complete pdf}
	\[
	f_D(d)=\begin{cases}
		\dfrac d8,&0\le d\le2,\\[1mm]
		\dfrac14,&2<d\le4,\\[1mm]
		\dfrac{6-d}{8},&4<d\le6,\\[1mm]
		0,&\text{otherwise.}
	\end{cases}
	\]
	The breakpoint checks above verify continuity.
	
	\SolutionStage{5}{Tail Probability}
	Since $f_D(5)=\frac18$, the right-tail region is a triangle:
	\[
	P(D>5)=\frac12(6-5)\frac18=\frac1{16}=6.25\%.
	\]
	\SolvedGraph{Tail graph: $P(D>5)$}{\COneGraph[d]{0}{6}{(x<=2)*(x/8)+(x>2)*(x<=4)*(1/4)+(x>4)*((6-x)/8)}{5}{6}{0,1,2,4,5,6}{0.30}}
	
	\SolutionStage{6}{Interval Probability}
	Use the complement of the two congruent outside tail triangles:
	\[
	P(1\le D\le5)=1-\frac12(1)\frac18-\frac12(1)\frac18
	=\frac78=87.5\%.
	\]
	\SolvedGraph{Band graph: $P(1\le D\le5)$}{\COneGraph[d]{0}{6}{(x<=2)*(x/8)+(x>2)*(x<=4)*(1/4)+(x>4)*((6-x)/8)}{1}{5}{0,1,2,4,5,6}{0.30}}
