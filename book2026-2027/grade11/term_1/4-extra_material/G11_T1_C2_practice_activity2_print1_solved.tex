	\vspace{-4mm}
	
	\noindent\textbf{Line through two points.}  For $x_1\ne x_2$,
	\[
		m=\frac{y_2-y_1}{x_2-x_1},\qquad
		b=y_1-mx_1,\qquad
		y=mx+b.
	\]

	\begin{enumerate}[
		leftmargin=*,
		itemsep=2pt,
		parsep=0pt,
		topsep=2pt
		]
		
		\item \textbf{Waiting for TransMilenio after class.}  On weekday evenings,
		a university student leaves class and waits at a TransMilenio station near
		the Universidad Nacional.  The wait always lasts between two and ten
		minutes, and every waiting time in that span is equally plausible.
		Determine the complete probability density rule for the waiting time,
		including what happens outside the stated span.  Then find the chance that
		the student waits longer than seven minutes and the chance that the wait
		lasts from three through seven minutes.
		
		\item \textbf{An evening commute across Bogot\'a.}  A young apprentice
		travels by SITP from Kennedy to a training programme in Chapinero during
		the evening rush.  Traffic can add anywhere from no delay to eight minutes
		to the usual journey.  A delay of any length in that span is possible, and
		longer delays become steadily more common.  An eight-minute delay is five
		times as common as no delay.  Determine the complete probability density
		rule for the added travel time, including what happens outside the stated
		span.  Then find the chance of more than six minutes of added time and the
		chance of between two and six minutes of added time.
		
		\item \textbf{Weekly study at a public library.}  Members of a Bogot\'a
		youth scholarship programme record the time they study each week at the
		Biblioteca P\'ublica Virgilio Barco.  Weekly totals range from two to ten
		hours.  Totals become steadily more common from two hours up to a most
		common total of five hours, then steadily less common through ten hours;
		there are no totals right at either limit.  Determine the complete
		probability density rule for weekly study time, including what happens
		outside the stated span.  Then find the chance of studying for more than
		seven hours and the chance of studying from four through eight hours.
		
		\item \textbf{A Sunday on the Ciclov\'ia.}  A group of young adults tracks
		how long its members spend cycling and meeting friends on Bogot\'a's Sunday
		Ciclov\'ia.  A recorded visit can last from zero to five hours.  All
		durations up to two hours are equally common.  All durations from two to
		five hours are also equally common, but they occur twice as often as
		durations in the shorter band.  Determine the complete probability density
		rule for the visit length, including what happens outside the stated span.
		Then find the chance that a visit lasts more than four hours and the chance
		that it lasts from one through four hours.
		
		\item \textbf{Time at a youth entrepreneurship fair.}  Young adults attend
		a Bogot\'a entrepreneurship fair at Corferias for up to six hours.
		Immediate departures are not observed.  Departures become
		steadily more common during the first two hours, remain equally common at
		their highest level from two through four hours, and then become steadily
		less common until six hours, when none are observed.  Determine the
		complete probability density rule for the length of a visit, including what
		happens outside the stated span.  Then find the chance of staying longer
		than five hours and the chance of staying from one through five hours.
		
	\end{enumerate}

	\section*{Solutions}
	
	\subsection*{1. Uniform distribution}
	\textbf{Support, shape, and normalization.}  On $[2,10]$, the density is a
	rectangle of unknown height $h$.  Total area $1$ gives
	\[
		1=(10-2)h=8h,\qquad h=\frac18.
	\]
	Therefore the complete pdf is
	\[
		f_T(t)=
		\begin{cases}
			\dfrac18,&2\le t\le10,\\[1mm]
			0,&\text{otherwise.}
		\end{cases}
	\]
	\SolvedGraph{General graph (unshaded)}{%
		\CompactDensityCoordinatesGraph
			{1}{11}{0.15}
			{2,3,7,10}{2,3,7,10}
			{0,0.125}{t}{$0$,$\frac18$}
			{(1,0) (2,0) (2,0.125) (10,0.125) (10,0) (11,0)}%
	}
	\textbf{Tail probability.}  The right tail is a rectangle:
	\[
		P(T>7)=(10-7)\frac18=\frac38=37.5\%.
	\]
	\SolvedGraph{Tail graph: $P(T>7)$}{%
		\CtwoProbGraph[t]{2}{10}{1/8}{7}{10}{2,3,7,10}{0.16}%
	}
	\textbf{Interval probability.}  The requested interval is also a rectangle:
	\[
		P(3\le T\le7)=(7-3)\frac18=\frac12.
	\]
	\SolvedGraph{Band graph: $P(3\le T\le7)$}{%
		\CtwoProbGraph[t]{2}{10}{1/8}{3}{7}{2,3,7,10}{0.16}%
	}

	\subsection*{2. Linear distribution}
	\textbf{Support, shape, and normalization.}  On $[0,8]$, let the positive
	endpoint heights be $h$ and $5h$.  The entire density is a trapezoid, so
	\[
		1=\frac12(8)(h+5h)=24h,
		\qquad h=\frac1{24},\qquad 5h=\frac5{24}.
	\]
	The line passes through $(0,1/24)$ and $(8,5/24)$.  Its slope and intercept
	are
	\[
		m=\frac{\frac5{24}-\frac1{24}}{8-0}=\frac1{48},
		\qquad b=\frac1{24}.
	\]
	Hence
	\[
		f_X(x)=
		\begin{cases}
			\dfrac{x+2}{48},&0\le x\le8,\\[1mm]
			0,&\text{otherwise.}
		\end{cases}
	\]
	Both endpoint heights are positive, so this nonconstant line is not
	triangular.
	\SolvedGraph{General graph (unshaded)}{%
		\CompactDensityCoordinatesGraph
			{-1}{9}{0.24}
			{0,2,6,8}{0,2,6,8}
			{0,0.0416667,0.2083333}{x}{$0$,$\frac1{24}$,$\frac5{24}$}
			{(-1,0) (0,0) (0,0.0416667) (8,0.2083333) (8,0) (9,0)}%
	}

	\textbf{Tail probability.}  Since $f_X(6)=1/6$ and $f_X(8)=5/24$, the tail
	is a width-$2$ trapezoid:
	\[
		P(X>6)=\frac12(8-6)\left(\frac16+\frac5{24}\right)=\frac38.
	\]
	\SolvedGraph{Tail graph: $P(X>6)$}{%
		\COneGraph[x]{0}{8}{(x+2)/48}{6}{8}{0,2,6,8}{0.24}%
	}
	\textbf{Interval probability.}  Here $f_X(2)=1/12$ and $f_X(6)=1/6$:
	\[
		P(2\le X\le6)
		=\frac12(6-2)\left(\frac1{12}+\frac16\right)=\frac12.
	\]
	\SolvedGraph{Band graph: $P(2\le X\le6)$}{%
		\COneGraph[x]{0}{8}{(x+2)/48}{2}{6}{0,2,6,8}{0.24}%
	}

	\subsection*{3. Triangular distribution}
	\textbf{Support, shape, and normalization.}  The support is $[2,10]$, the
	endpoint heights are $0$, and the mode is $5$.  If the peak height is $H$,
	then
	\[
		1=\frac12(10-2)H=4H,\qquad H=\frac14.
	\]
	The rising segment joins $(2,0)$ to $(5,1/4)$, and the falling segment joins
	$(5,1/4)$ to $(10,0)$.  Their slopes are
	\[
		m_1=\frac{\frac14-0}{5-2}=\frac1{12},
		\qquad
		m_2=\frac{0-\frac14}{10-5}=-\frac1{20}.
	\]
	Using the zero-height endpoints gives the complete pdf
	\[
		f_Q(q)=
		\begin{cases}
			\dfrac{q-2}{12},&2\le q\le5,\\[1mm]
			\dfrac{10-q}{20},&5<q\le10,\\[1mm]
			0,&\text{otherwise.}
		\end{cases}
	\]
	The pieces meet at $f_Q(5)=1/4$.
	\SolvedGraph{General graph (unshaded)}{%
		\CompactDensityCoordinatesGraph
			{1}{11}{0.30}
			{2,4,5,7,8,10}{2,4,5,7,8,10}
			{0,0.25}{q}{$0$,$\frac14$}
			{(1,0) (2,0) (5,0.25) (10,0) (11,0)}%
	}

	\textbf{Tail probability.}  Since $f_Q(7)=3/20$, the right tail is a
	triangle:
	\[
		P(Q>7)=\frac12(10-7)\frac3{20}=\frac9{40}=22.5\%.
	\]
	\SolvedGraph{Tail graph: $P(Q>7)$}{%
		\COneGraph[q]{2}{10}
			{(x<=5)*((x-2)/12)+(x>5)*((10-x)/20)}
			{7}{10}{2,4,5,7,8,10}{0.30}%
	}
	\textbf{Interval probability.}  At the outside boundaries,
	$f_Q(4)=1/6$ and $f_Q(8)=1/10$.  Subtract the two tail triangles:
	\[
		\begin{aligned}
		P(4\le Q\le8)
		&=1-\frac12(4-2)\frac16-\frac12(10-8)\frac1{10}\\
		&=\frac{11}{15}\approx73.3\%.
		\end{aligned}
	\]
	\SolvedGraph{Band graph: $P(4\le Q\le8)$}{%
		\COneGraph[q]{2}{10}
			{(x<=5)*((x-2)/12)+(x>5)*((10-x)/20)}
			{4}{8}{2,4,5,7,8,10}{0.30}%
	}

	\subsection*{4. Piecewise-uniform distribution}
	\textbf{Support, heights, and normalization.}  The support is $[0,5]$, with
	breakpoint $2$.  The two constant heights are $k$ and $2k$.  Adding their
	rectangular areas gives
	\[
		1=2k+(5-2)(2k)=8k,\qquad k=\frac18.
	\]
	Thus
	\[
		f_W(w)=
		\begin{cases}
			\dfrac18,&0\le w<2,\\[1mm]
			\dfrac14,&2\le w\le5,\\[1mm]
			0,&\text{otherwise.}
		\end{cases}
	\]
	\SolvedGraph{General graph (unshaded)}{%
		\CompactDensityCoordinatesGraph
			{-0.5}{5.5}{0.30}
			{0,1,2,4,5}{0,1,2,4,5}
			{0,0.125,0.25}{w}{$0$,$\frac18$,$\frac14$}
			{(-0.5,0) (0,0) (0,0.125) (2,0.125) (2,0.25)
			 (5,0.25) (5,0) (5.5,0)}%
	}
	\textbf{Tail probability.}  The tail is a width-$1$, height-$1/4$
	rectangle:
	\[
		P(W>4)=(5-4)\frac14=\frac14.
	\]
	\SolvedGraph{Tail graph: $P(W>4)$}{%
		\CtwoProbGraph[w]{0}{5}
			{(x<2)*(1/8)+(x>=2)*(1/4)}
			{4}{5}{0,1,2,4,5}{0.30}%
	}
	\textbf{Interval probability.}  Split at the breakpoint and add two
	rectangles:
	\[
		P(1\le W\le4)
		=(2-1)\frac18+(4-2)\frac14
		=\frac18+\frac12=\frac58.
	\]
	\SolvedGraph{Band graph: $P(1\le W\le4)$}{%
		\CtwoProbGraph[w]{0}{5}
			{(x<2)*(1/8)+(x>=2)*(1/4)}
			{1}{4}{0,1,2,4,5}{0.30}%
	}

	\subsection*{5. Piecewise-linear distribution}
	\textbf{Support, heights, and normalization.}  The support is $[0,6]$, with
	breakpoints $2$ and $4$ and heights $0,H,H,0$ at $0,2,4,6$.  The total area
	is a triangle, a rectangle, and a triangle:
	\[
		1=\frac12(2)H+(4-2)H+\frac12(6-4)H=4H,
		\qquad H=\frac14.
	\]
	The first segment joins $(0,0)$ to $(2,1/4)$, so
	\[
		m_1=\frac{\frac14-0}{2-0}=\frac18,
		\qquad f_D(d)=\frac d8.
	\]
	The middle segment joins $(2,1/4)$ to $(4,1/4)$ and is constant at $1/4$.
	The last segment joins $(4,1/4)$ to $(6,0)$, so
	\[
		m_3=\frac{0-\frac14}{6-4}=-\frac18,
		\qquad f_D(d)=\frac{6-d}{8}.
	\]
	Therefore the complete connected pdf is
	\[
		f_D(d)=
		\begin{cases}
			\dfrac d8,&0\le d\le2,\\[1mm]
			\dfrac14,&2<d\le4,\\[1mm]
			\dfrac{6-d}{8},&4<d\le6,\\[1mm]
			0,&\text{otherwise.}
		\end{cases}
	\]
	The pieces meet at height $1/4$ at both breakpoints.
	\SolvedGraph{General graph (unshaded)}{%
		\CompactDensityCoordinatesGraph
			{-0.5}{6.5}{0.30}
			{0,1,2,4,5,6}{0,1,2,4,5,6}
			{0,0.25}{d}{$0$,$\frac14$}
			{(-0.5,0) (0,0) (2,0.25) (4,0.25) (6,0) (6.5,0)}%
	}

	\textbf{Tail probability.}  Since $f_D(5)=1/8$, the right tail is a
	triangle:
	\[
		P(D>5)=\frac12(6-5)\frac18=\frac1{16}=6.25\%.
	\]
	\SolvedGraph{Tail graph: $P(D>5)$}{%
		\COneGraph[d]{0}{6}
			{(x<=2)*(x/8)+(x>2)*(x<=4)*(1/4)+(x>4)*((6-x)/8)}
			{5}{6}{0,1,2,4,5,6}{0.30}%
	}
	\textbf{Interval probability.}  The complement consists of two congruent
	tail triangles:
	\[
		P(1\le D\le5)
		=1-\frac12(1)\frac18-\frac12(1)\frac18
		=\frac78=87.5\%.
	\]
	\SolvedGraph{Band graph: $P(1\le D\le5)$}{%
		\COneGraph[d]{0}{6}
			{(x<=2)*(x/8)+(x>2)*(x<=4)*(1/4)+(x>4)*((6-x)/8)}
			{1}{5}{0,1,2,4,5,6}{0.30}%
	}
