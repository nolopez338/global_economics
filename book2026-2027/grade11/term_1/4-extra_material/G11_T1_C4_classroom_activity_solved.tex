	\noindent
	\textbf{Exponential distribution.}
	For a waiting time $T$ with rate $\lambda>0$,
	\vspace{-3mm}
	\[
	T\sim\operatorname{Exponential}(\lambda).
	\]
	\vspace{-17mm}
\begin{center}
	\begin{tabular}{c@{\qquad\qquad}c}
		\begin{minipage}{0.43\textwidth}
			\[
			\begin{array}{c|c}
				\textbf{Situation} & \textbf{Equation}\\[3pt]
				\hline
				\text{Less than} & P(T<t)=1-e^{-\lambda t}\\[5pt]
				\text{Greater than} & P(T>t)=e^{-\lambda t}\\[5pt]
				\text{Between} & P(a<T<b)=e^{-\lambda a}-e^{-\lambda b}
			\end{array}
			\]
		\end{minipage}
		&
		\begin{minipage}{0.43\textwidth}
			\[
			\begin{array}{c|c}
				\textbf{Given information} & \textbf{Equation for }\lambda\\[3pt]
				\hline
				P(T>t)=p &
				\displaystyle\lambda=-\frac{\ln(p)}{t}\\[9pt]
				P(T<t)=p &
				\displaystyle\lambda=-\frac{\ln(1-p)}{t}
			\end{array}
			\]
		\end{minipage}
	\end{tabular}
\end{center}

\subsection*{C4.1 - Finding a probability}\phantomsection
\label{subsec:c4-probability-transmilenio}

At a TransMilenio station, $\lambda=0.20$ buses/minute. Find
$P(T>6)$.

\[
\begin{array}{c|c}
	\textbf{General} & P(T>t)=e^{-\lambda t}\\[6pt]
	\textbf{Context} & P(T>6),\quad \lambda=0.20
\end{array}
\]

Therefore,
\[
P(T>6)=e^{-(0.20)(6)}
=e^{-1.2}
\]

\[
\boxed{P(T>6)\approx0.3010}
\]

\subsection*{C4.7 - Finding the rate parameter}\phantomsection
\label{subsec:c4-find-rate-support}

The company estimates that there is a $30\%$ probability that the next
support call takes more than 8 minutes to arrive. Find $\lambda$.

\[
\begin{array}{c|c}
	\textbf{General} & P(T>t)=e^{-\lambda t}\\[6pt]
	\textbf{Context} & P(T>8)=0.30
\end{array}
\]

Therefore,
\[
0.30=e^{-8\lambda}
\quad\longrightarrow\quad
\ln(0.30)=-8\lambda
\]

\[
\longrightarrow\quad
\lambda=-\frac{\ln(0.30)}{8}
\]

\[
\boxed{\lambda\approx0.1505\text{ calls/minute}}
\]
