	\begin{enumerate}[
		leftmargin=*,
		itemsep=2pt,
		parsep=0pt,
		topsep=2pt
		]
		
		\item \textbf{Randomized drone inspection angle.} A university research
		drone is used to inspect a circular construction site. During each
		inspection, its camera points in a direction $D$ measured in radians
		clockwise from a fixed reference marker, with $D\in[0,2\pi)$.
		Because the drone controller randomly selects the initial orientation with
		no preferred direction, a circular-uniform model is appropriate. The
		direction $2\pi$ is equivalent to $0$ because they represent the same
		physical orientation.
		
		The construction team is particularly interested in directions close to
		the fixed reference marker. Determine the complete periodic probability
		density rule for the camera direction, including what happens outside the
		stated cycle. Then find the probability that the camera direction lies
		within $\frac{\pi}{4}$ radians of the reference marker, interpreting the
		interval around $0$ periodically.
		
		\item \textbf{Periodic triangular energy-demand time.} A university campus
		monitors the time $T$ at which its electricity demand reaches its daily
		peak. The time is measured in hours on $[0,24)$, with midnight represented
		by both $0$ h and $24$ h. Therefore, time of day is periodic with period
		$24$ h.
		
		For this planning model, the electricity demand peak is least likely at
		midnight. The density increases linearly from zero at midnight to its
		maximum at 10:00, then decreases linearly back to zero at the next
		midnight. A circular-triangular pdf represents this daily pattern.
		
		The campus energy team is interested in peak times close to midnight.
		Determine the complete periodic probability density rule for the peak
		time, including what happens outside the displayed day. Then find the
		probability that the daily peak occurs within 5 hours of midnight,
		interpreting the interval around midnight periodically.
		
	\end{enumerate}

	\section*{Solutions}
	
	% ============================================================
	% PROBLEM 1
	% ============================================================
	
	\section{Periodic uniform distribution}
	
	\setlength{\columnseprule}{0.4pt}
	\begin{multicols}{2}
		
		\SolutionStage{1}{Support and Shape}
		The random variable $D$ represents the drone camera direction in radians.
		One complete cycle is represented by $[0,2\pi)$, and the direction is
		periodic because a full rotation returns the camera to the same physical
		orientation. The controller selects all directions with equal likelihood,
		so the density is a rectangle of unknown height $h$ over the complete cycle
		and is zero outside the displayed support.
		
		\SolvedGraph{Shape from the context}{%
			\UniformShapeGraph
		}
		
		\columnbreak
		
		\SolutionStage{2}{Total Area = 1}
		The rectangular pdf must have total area $1$ over one complete cycle:
		\[
		1=(2\pi)h,
		\qquad
		h=\frac1{2\pi}.
		\]
		
		\SolvedGraph{Completed periodic density graph}{%
			\UniformCompleteGraph
		}
		
	\end{multicols}
	\setlength{\columnseprule}{0pt}
	
	
	\SolutionStage{3}{Straight Line Rule}
	The density is constant throughout the support, so the graph is horizontal.Therefore its slope is $m=0$, and the constant rule is $f_D(d)=\frac1{2\pi}$ on the support. The periodic nature of the direction is expressed by $f_D(d+2\pi)=f_D(d)$.
	
	
	\SolutionStage{4}{Complete pdf}
	The complete pdf can be written on one displayed cycle as
	\[
	f_D(d)=
	\begin{cases}
		\dfrac1{2\pi},&0\le d<2\pi,\\[1mm]
		0,&\text{otherwise}.
	\end{cases}
	\]
	
	The displayed endpoints $0$ and $2\pi$ represent the same physical
	direction, and the periodic extension satisfies
	\[
	f_D(d+2\pi)=f_D(d).
	\]
	
	\SolutionStage{5}{Interval Around the Reference Marker}
	The interval is
	
	$$
	-\frac{\pi}{4}\le D\le\frac{\pi}{4}
	\qquad\longrightarrow\qquad
	\left[0,\frac{\pi}{4}\right]\cup\left[\frac{7\pi}{4},2\pi\right).
	$$
	
	The two pieces have total width
	
	$$
	\frac{\pi}{4}+\frac{\pi}{4}=\frac{\pi}{2}
	\qquad\longrightarrow\qquad
	P\left(-\frac{\pi}{4}\le D\le\frac{\pi}{4}\right)
	=\frac{\pi}{2}\frac1{2\pi}
	=\frac14=25\%.
	$$
	
	Thus, $25\%$ of randomized camera directions lie within $\frac{\pi}{4}$ radians of the reference marker.
	
	\SolvedGraph{Periodic interval around $0$}{%
		\UniformWrappedProbabilityGraph
	}
	
	% ============================================================
	% PROBLEM 2
	% ============================================================
	
	\section{Periodic triangular distribution}
	
	\setlength{\columnseprule}{0.4pt}
	\begin{multicols}{2}
		
		\SolutionStage{1}{Support and Shape}
		The random variable $T$ represents the time of the daily electricity-demand
		peak in hours. One complete cycle is represented by $[0,24)$, and the period
		is $24$ h because the clock repeats every day. The density is zero at
		midnight, rises steadily to its maximum at $10$ h, and then falls steadily to
		zero at the next midnight. Therefore, the graph is a triangle with unknown
		maximum height $H$ and mode at $t=10$.
		
		\SolvedGraph{Shape from the context}{%
			\TriangularShapeGraph
		}
		
		\columnbreak
		
		\SolutionStage{2}{Total Area = 1}
		The triangular pdf must have total area $1$:
		\[
		1=\frac12(24)H,
		\qquad
		H=\frac1{12}.
		\]
		
		The completed triangle therefore has maximum height $1/12$ at $t=10$.
		
		\SolvedGraph{Completed periodic density graph}{%
			\TriangularCompleteGraph
		}
		
	\end{multicols}
	\setlength{\columnseprule}{0pt}
	
	\SolutionStage{3}{Straight Line Rules}
	The rising section connects $(0,0)$ and $\left(10,\frac1{12}\right)$:
	
	$$
	(0,0),\left(10,\frac1{12}\right)
	\qquad\longrightarrow\qquad
	m=\frac{\frac1{12}-0}{10-0}=\frac1{120},\quad b=0
	\qquad\longrightarrow\qquad
	f_T(t)=\frac{t}{120}.
	$$
	
	The falling section connects $\left(10,\frac1{12}\right)$ and $(24,0)$:
	
	$$
	\left(10,\frac1{12}\right),(24,0)
	\qquad\longrightarrow\qquad
	m=\frac{0-\frac1{12}}{24-10}=-\frac1{168}
	\qquad\longrightarrow\qquad
	f_T(t)=\frac{24-t}{168}.
	$$
	
	\setlength{\columnseprule}{0.4pt}
	\begin{multicols}{2}
		
		\SolutionStage{4}{Complete pdf}
		The complete pdf on one displayed day is
		\[
		f_T(t)=
		\begin{cases}
			\dfrac{t}{120},&0\le t\le10,\\[2mm]
			\dfrac{24-t}{168},&10<t<24,\\[2mm]
			0,&\text{otherwise}.
		\end{cases}
		\]
		
		\columnbreak
		
		The periodic extension is
		\[
		f_T(t+24)=f_T(t).
		\]
		
		
		
		The two triangular sections have areas
		\[
		\frac12(10)\frac1{12}
		+
		\frac12(14)\frac1{12}
		=
		\frac5{12}+\frac7{12}
		=1,
		\]
		confirming that the pdf is correctly normalized.
		
	\end{multicols}
	\setlength{\columnseprule}{0pt}
	
	\SolutionStage{5}{Interval Around Midnight}
	The interval is
	
	$$
	-5\le T\le5
	\qquad\longrightarrow\qquad
	[0,5]\cup[19,24).
	$$
	
	For $[0,5]$:
	
	$$
	f_T(5)=\frac5{120}=\frac1{24}
	\qquad\longrightarrow\qquad
	A_1=\frac12(5)\frac1{24}=\frac5{48}.
	$$
	
	For $[19,24)$:
	
	$$
	f_T(19)=\frac{24-19}{168}=\frac5{168}
	\qquad\longrightarrow\qquad
	A_2=\frac12(5)\frac5{168}=\frac{25}{336}.
	$$
	
	Therefore,
	
	$$
	P(-5\le T\le5)
	=\frac5{48}+\frac{25}{336}
	=\frac{35}{336}+\frac{25}{336}
	=\frac{60}{336}
	=\frac5{28}
	\approx17.86\%.
	$$
	
	Thus, approximately $17.9\%$ of days have their modeled electricity-demand peak within $5$ hours of midnight.

	
	\SolvedGraph{Periodic interval around midnight}{%
		\TriangularWrappedProbabilityGraph
	}
