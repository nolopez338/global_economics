\section{Electricity bills}

\begin{frame}[t]{Which electricity bills are we studying?}
  A utility studies the \textbf{amount charged per bill}.

  \medskip
  \textbf{Predict.} Which bills might the utility want to describe?
  Which bills could it use to estimate the mean charge?

  \bigskip
  \visible<2->{\textbf{Target population.}
    \HighlightStroke{G11T2C1Population}{All May electricity bills}
    for small shops in one district.}\par
  \medskip
  \visible<3->{\textbf{Sample.}
    \HighlightStroke{G11T2C1Sample}{40 randomly selected bills} from that list.}
\end{frame}

\begin{frame}[t]{From electricity bills to a mean}
  \textbf{Predict.} Which group does each mean describe?

  \begin{center}
    \begin{tabularx}{\linewidth}{@{}*{2}{>{\raggedright\arraybackslash}X}@{}}
      \toprule
      \textbf{Population parameter} & \textbf{Sample statistic} \\
      \midrule
      \leavevmode\visible<2->{$\HighlightStroke{G11T2C1PopulationMean}{\mu}$ is the
        unknown mean charge across \HighlightStroke{G11T2C1Population}{all May bills}
        for small shops in the district.} &
      \leavevmode\visible<3->{$\HighlightStroke{G11T2C1SampleMean}{\bar{x}}
        =\HighlightStroke{G11T2C1SampleMean}{210}$ thousand COP is the
        mean charge across the \HighlightStroke{G11T2C1Sample}{40 selected bills}.} \\
      \bottomrule
    \end{tabularx}
  \end{center}

  \visible<4->{The sample mean $\HighlightStroke{G11T2C1SampleMean}{\bar{x}}$
    estimates the population mean $\HighlightStroke{G11T2C1PopulationMean}{\mu}$.}
\end{frame}

\begin{frame}[t]{Quick check. Redraw or redefine?}
  \textbf{Predict.} Which population mean does each draw estimate?

  \begin{center}
    \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{.41\linewidth}>{\raggedright\arraybackslash}X@{}}
      \toprule
      \textbf{New draw} & \leavevmode\visible<2->{\textbf{Suggested answer}} \\
      \midrule
      \HighlightStroke{G11T2C1Sample}{40 bills} from the
      \HighlightStroke{G11T2C1Population}{original May list} &
      \leavevmode\visible<2->{A fresh draw can give a different
        $\HighlightStroke{G11T2C1SampleMean}{\bar{x}_{may}}$ while estimating the
        same $\HighlightStroke{G11T2C1PopulationMean}{\mu_{may}}$ for the original district.} \\
      \addlinespace
      \HighlightStroke{G11T2C1Sample}{40 bills} from the combined May list
      for small shops in \HighlightStroke{G11T2C1Population}{two districts} &
      \leavevmode\visible<3->{This draw's mean $\HighlightStroke{G11T2C1SampleMean}{\bar{x}_{2D}}$
        estimates $\HighlightStroke{G11T2C1PopulationMean}{\mu_{2D}}$, the population mean across both districts.} \\
      \bottomrule
    \end{tabularx}
  \end{center}

  \visible<4->{The two population means could have the same numerical value.}
\end{frame}

\section{Market survey}

\begin{frame}[t]{A market association's sales survey (C1)}
  A market association wants to describe all \HighlightStroke{G11T2C1Population}{1,200 stalls}
  licensed for June. It randomly selects \HighlightStroke{G11T2C1Sample}{60 stalls}.

  \medskip
  In the survey, mean June sales revenue is
  \HighlightStroke{G11T2C1SampleMean}{4.8} million COP and the standard
  deviation is \HighlightStroke{G11T2C1SampleSD}{1.2} million COP.
  \HighlightStroke{G11T2C1Sample}{18 sampled stalls} accept digital payment.

  \bigskip
  \textbf{Task 1.} Which stalls make up the population? Which make up the sample?

  \medskip
  \visible<2->{\textbf{Population.}
    \HighlightStroke{G11T2C1Population}{All 1,200 June-licensed stalls}.}\par
  \medskip
  \visible<2->{\textbf{Sample.}
    The \HighlightStroke{G11T2C1Sample}{60 randomly selected stalls}.}
\end{frame}

\begin{frame}[t]{Match each quantity to its symbol}
  \textbf{Task 1, continued.} Which sample statistic estimates each
  population parameter? What does each symbol describe?

  \begin{center}
    \begin{tabularx}{\linewidth}{@{}*{2}{>{\raggedright\arraybackslash}X}@{}}
      \toprule
      \textbf{Population parameter} & \textbf{Sample statistic} \\
      \midrule
      \leavevmode\visible<2->{$\HighlightStroke{G11T2C1PopulationMean}{\mu}$,
        mean June sales across all licensed stalls} &
      \leavevmode\visible<2->{$\HighlightStroke{G11T2C1SampleMean}{\bar{x}_{1}}
        =\HighlightStroke{G11T2C1SampleMean}{4.8}$ million COP,
        mean June sales across the 60} \\
      \addlinespace
      \leavevmode\visible<3->{$\HighlightStroke{G11T2C1PopulationSD}{\sigma}$,
        standard deviation of June sales across all licensed stalls} &
      \leavevmode\visible<3->{$\HighlightStroke{G11T2C1SampleSD}{s}
        =\HighlightStroke{G11T2C1SampleSD}{1.2}$ million COP,
        standard deviation of June sales across the 60} \\
      \addlinespace
      \leavevmode\visible<4->{$\HighlightStroke{G11T2C1PopulationProportion}{p}$,
        fraction of all June-licensed stalls accepting digital payment} &
      \leavevmode\visible<4->{$\HighlightStroke{G11T2C1SampleProportion}{\hat{p}}
        =\HighlightStroke{G11T2C1SampleProportion}{0.30}$,
        fraction accepting digital payment among the 60} \\
      \bottomrule
    \end{tabularx}
  \end{center}
\end{frame}

\begin{frame}[t]{The sample share accepting digital payment}
  \textbf{Predict.} What does each count in the fraction describe?

  \[
    \HighlightStroke{G11T2C1SampleProportion}{\hat{p}}
    =\HighlightStroke{G11T2C1SampleProportion}{\frac{18}{60}}
    =\HighlightStroke{G11T2C1SampleProportion}{0.30}
    =\HighlightStroke{G11T2C1SampleProportion}{30\%}
  \]

  \visible<2->{\textbf{Numerator.}
    \HighlightStroke{G11T2C1Sample}{18 sampled stalls} accept digital payment.}\par
  \medskip
  \visible<3->{\textbf{Denominator.}
    \HighlightStroke{G11T2C1Sample}{All 60 sampled stalls} are counted.}\par
  \medskip
  \visible<3->{The sample proportion is unitless.}
\end{frame}

\begin{frame}[t]{A fresh draw from the June list}
  \textbf{Population.} \HighlightStroke{G11T2C1Population}{All 1,200 June-licensed stalls}.
  The association takes a \HighlightStroke{G11T2C1Sample}{fresh random sample}
  from the same list to see how survey results vary across draws.

  \medskip
  \begin{itemize}
    \item Mean June sales revenue is \HighlightStroke{G11T2C1SampleMean}{5.1} million COP.
    \item Standard deviation is \HighlightStroke{G11T2C1SampleSD}{1.0} million COP.
    \item Digital-payment share is
      $\HighlightStroke{G11T2C1SampleProportion}{0.25}
      =\HighlightStroke{G11T2C1SampleProportion}{25\%}$.
  \end{itemize}

  \bigskip
  \textbf{Task 2.} Revisit your answers to Task 1. What changes, what stays
  fixed, and which differences can you calculate?
\end{frame}

\begin{frame}[t]{Compare the two June draws}
  \textbf{Population.} \HighlightStroke{G11T2C1Population}{All 1,200 June-licensed stalls}.

  \medskip
  \textbf{Predict.} Which sample statistics changed across the two draws?

  \begin{center}
    \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}Xcc@{}}
      \toprule
      \textbf{Sample statistic} &
      \textbf{\HighlightStroke{G11T2C1Sample}{First draw}} &
      \textbf{\HighlightStroke{G11T2C1Sample}{Second draw}} \\
      \midrule
      June sales mean (million COP) &
      \visible<2->{$\HighlightStroke{G11T2C1SampleMean}{\bar{x}_{1}}=\HighlightStroke{G11T2C1SampleMean}{4.8}$} &
      \visible<2->{$\HighlightStroke{G11T2C1SampleMean}{\bar{x}_{2}}=\HighlightStroke{G11T2C1SampleMean}{5.1}$} \\
      June sales standard deviation (million COP) &
      \visible<2->{$\HighlightStroke{G11T2C1SampleSD}{s}=\HighlightStroke{G11T2C1SampleSD}{1.2}$} &
      \visible<2->{$\HighlightStroke{G11T2C1SampleSD}{s}=\HighlightStroke{G11T2C1SampleSD}{1.0}$} \\
      Digital-payment share (unitless) &
      \visible<2->{$\HighlightStroke{G11T2C1SampleProportion}{\hat{p}}=\HighlightStroke{G11T2C1SampleProportion}{0.30}$} &
      \visible<2->{$\HighlightStroke{G11T2C1SampleProportion}{\hat{p}}=\HighlightStroke{G11T2C1SampleProportion}{0.25}$} \\
      \bottomrule
    \end{tabularx}
  \end{center}

  \visible<3->{All three statistics differ in this fresh draw.}
\end{frame}

\begin{frame}[t]{Second draw minus first draw}
  \textbf{Predict.} What are the three differences between sample statistics?

  \begin{center}
    \begin{tabular}{@{}ll@{}}
      \toprule
      \textbf{Sample statistic} & \textbf{Difference between draws} \\
      \midrule
      Mean &
      \visible<2->{$\HighlightStroke{G11T2C1SampleMean}{\bar{x}_{2}}-\HighlightStroke{G11T2C1SampleMean}{\bar{x}_{1}}
        =\HighlightStroke{G11T2C1SampleMean}{5.1}
        -\HighlightStroke{G11T2C1SampleMean}{4.8}=+0.3$ million COP} \\
      Standard deviation &
      \visible<3->{$\HighlightStroke{G11T2C1SampleSD}{1.0}
        -\HighlightStroke{G11T2C1SampleSD}{1.2}=-0.2$ million COP} \\
      Digital-payment share &
      \visible<4->{$\HighlightStroke{G11T2C1SampleProportion}{0.25}
        -\HighlightStroke{G11T2C1SampleProportion}{0.30}=-0.05$,
        or $-5$ percentage points} \\
      \bottomrule
    \end{tabular}
  \end{center}

  \textbf{Predict.} What stays fixed?

  \medskip
  \visible<5->{The population parameters $\HighlightStroke{G11T2C1PopulationMean}{\mu}$,
    $\HighlightStroke{G11T2C1PopulationSD}{\sigma}$ and
    $\HighlightStroke{G11T2C1PopulationProportion}{p}$ stay fixed for the
    \HighlightStroke{G11T2C1Population}{same June population}.}
\end{frame}
